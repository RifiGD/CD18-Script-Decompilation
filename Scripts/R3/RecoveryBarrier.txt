private alias 61 : TYPE_RECOVERYBARRIER



event ObjectUpdate
	switch object.state
	case 0
		object.frame = 0
		if object[-4].value2 == 3
			object.state = 1
		end if
		break
	case 1
		object.frame = 1
		if object.value0 < 30
			object.value0++
		else
			object.state = 2
		end if
		break
	case 2
		object.frame = 2
		if object[-4].value2 == 0
			object.state = 3
		end if
		break
	case 3
		object.frame = 3
		if object.value0 < 30
			object.value0++
		else
			object.state = 0
		end if
		break
	end switch
	if object.state == 2
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, -8, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == 1
				object[currentPlayer].yvel = -262144
			end if
		next
	end if
end event


event ObjectDraw
	DrawSprite(object.frame)
end event

	LoadSpriteSheet("R3/Objects2.gif")

	SpriteFrame(-16, -8, 32, 16, 51, 50)
end event

end event

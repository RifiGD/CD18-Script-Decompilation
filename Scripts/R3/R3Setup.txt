private function R3Setup_drawFunctionPresent
	GetTableValue(temp0, object.palIndexA, R3Setup_palCycleF)
	object.palIndexA++
	object.palIndexA %= 7
	SetPaletteEntry(0, 175, temp0)

	GetTableValue(temp0, object.palIndexB, R3Setup_palCycleG)
	object.palIndexB++
	object.palIndexB %= 41
	SetPaletteEntry(0, 188, temp0)

	GetTableValue(temp0, object.palIndexC, R3Setup_palCycleH)
	object.palIndexC++
	object.palIndexC %= 63
	SetPaletteEntry(0, 189, temp0)

	GetTableValue(temp0, object.palIndexD, R3Setup_palCycleA)
	object.palIndexD++
	object.palIndexD %= 5
	SetPaletteEntry(0, 191, temp0)
end function


private function R3Setup_drawFunctionPast
	//There is nothing here lol
end function


private function R3Setup_drawFunctionGoodFuture
	GetTableValue(temp0, object.palIndexA, R3Setup_palCycleB)
	object.palIndexA++
	object.palIndexA %= 7
	SetPaletteEntry(0, 172, temp0)
end function


private function R3Setup_drawFunctionBadFuture
	GetTableValue(temp0, object.palIndexA, R3Setup_palCycleC)
	object.palIndexA++
	object.palIndexA &= 7
	SetPaletteEntry(0, 170, temp0)

	object.palTimerA++
	if object.palTimerA == 5
		object.palTimerA = 0
		GetTableValue(temp0, object.palIndexB, R3Setup_palCycleD)
		object.palIndexB++
		object.palIndexB %= 33
		SetPaletteEntry(0, 189, temp0)
	end if

	object.palTimerB++
	if object.palTimerB == 10
		object.palTimerB = 0
		GetTableValue(temp0, object.palIndexC, R3Setup_palCycleE)
		object.palIndexC++
		object.palIndexC &= 15
		SetPaletteEntry(0, 187, temp0)
	end if
end function

event ObjectUpdate
	CallFunction(object.drawFunction)
	
	if object.turnDirection == false
		if object.bumperOscillation < 0x300000
			object.bumperOscillation += 0x10000
		else
			object.turnDirection = true
		end if
	else
		if object.bumperOscillation > -0x300000
			object.bumperOscillation -= 0x10000
		else
			object.turnDirection = false
		end if
	end if
	
	if hud.currentTimePeriod != TIME_PAST
		R3Setup_rotoDrumTimer++
		R3Setup_rotoDrumTimer %= 20
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp2 = player[currentPlayer].xpos
		temp2 >>= 16
		temp4 = temp2
		temp2 -= 8
		temp4 += 8
		temp3 = player[currentPlayer].ypos
		temp3 >>= 16
		if player[currentPlayer].yvel >= 0
			temp3 += player[currentPlayer].collisionBottom
			temp3 -= 2
		else
			temp3 += player[currentPlayer].collisionTop
			temp3 += 2
		end if

		temp5 = temp3
		temp7 = 0
		Get16x16TileInfo(temp0, temp2, temp3, TILEINFO_ANGLEA)
		switch temp0
		case 1
			temp7 = 1
			break
		case 3
			temp7 = 3
			Get16x16TileInfo(temp6, temp2, temp3, TILEINFO_DIRECTION)
			break
		case 5
			temp7 = 2
			break
		end switch

		Get16x16TileInfo(temp0, temp4, temp5, TILEINFO_ANGLEA)
		switch temp0
		case 1
			temp7 = 1
			break
		case 3
			temp7 = 3
			Get16x16TileInfo(temp6, temp4, temp5, TILEINFO_DIRECTION)
			break
		case 5
			temp7 = 2
			break
		end switch

		switch temp7
		case 1
			if player[currentPlayer].yvel >= 0
				player[currentPlayer].gravity = 1
				PlaySfx(SfxName[Bumper 2], 0)
				if player[currentPlayer].state == Player_State_Fly
					player[currentPlayer].yvel = -0x40000
				else
					player[currentPlayer].yvel = -0x70000
				end if
				if player[currentPlayer].state == Player_State_Hurt
					player[currentPlayer].state = Player_State_Air
					player[currentPlayer].timer = 0
				end if
			else
				if player[currentPlayer].gravity == 1
					player[currentPlayer].yvel = 0x40000
					PlaySfx(SfxName[Bumper 2], 0)
				end if
			end if
			break

		case 2
			if player[currentPlayer].gravity == 1
				PlaySfx(SfxName[Mini Bumper], 0)
				if player[currentPlayer].yvel >= 0
					player[currentPlayer].yvel = -0x40000
				else
					player[currentPlayer].yvel = 0x40000
				end if
			end if
			break

		case 3
			if player[currentPlayer].gravity == 1
				PlaySfx(SfxName[Bumper 2], 0)
				if player[currentPlayer].yvel >= 0
					if player[currentPlayer].state == Player_State_Fly
						player[currentPlayer].yvel = -0x40000
					else
						player[currentPlayer].yvel = -0x70000
					end if
				else
					player[currentPlayer].yvel = 0x40000
				end if
				if temp6 == 0
					if player[currentPlayer].xvel > 0
						FlipSign(object[currentPlayer].xvel)
						player[currentPlayer].xvel *= 5
						player[currentPlayer].xvel >>= 2
						if player[currentPlayer].xvel > -0x20000
							player[currentPlayer].xvel = -0x20000
						end if
					else
						if player[currentPlayer].xvel > -0x20000
							player[currentPlayer].xvel = -0x20000
						end if
					end if
				else
					if player[currentPlayer].xvel < 0
						FlipSign(player[currentPlayer].xvel)
						player[currentPlayer].xvel *= 5
						player[currentPlayer].xvel >>= 2
						if player[currentPlayer].xvel < 0x20000
							player[currentPlayer].xvel = 0x20000
						end if
					else
						if player[currentPlayer].xvel < 0x20000
							player[currentPlayer].xvel = 0x20000
						end if
					end if
				end if
				player[currentPlayer].speed = player[currentPlayer].xvel
			end if
			break

		end switch

		temp3 = player[currentPlayer].ypos
		temp3 >>= 16
		temp5 = temp3
		temp3 -= 6
		temp5 += 6
		temp2 = player[currentPlayer].xpos
		temp2 >>= 16

		if player[currentPlayer].xvel > 0
			temp2 += 8
		else
			temp2 -= 8
		end if
		temp4 = temp2

		temp7 = 0
		Get16x16TileInfo(temp0, temp2, temp3, TILEINFO_ANGLEA)
		switch temp0
		case 3
			temp7 = 1
			break
		case 5
			temp7 = 2
			break
		case 7
			temp7 = 1
			break
		end switch

		Get16x16TileInfo(temp0, temp4, temp5, TILEINFO_ANGLEA)
		switch temp0
		case 3
			temp7 = 1
			break
		case 5
			temp7 = 2
			break
		case 7
			temp7 = 1
			break
		end switch

		switch temp7
		case 1
			if player[currentPlayer].xvel > 0
				player[currentPlayer].xvel = -0x50000
			else
				player[currentPlayer].xvel = 0x50000
			end if
			player[currentPlayer].speed = player[currentPlayer].xvel
			PlaySfx(SfxName[Bumper 2], 0)
			break
		case 2
			if player[currentPlayer].xvel > 0
				player[currentPlayer].xvel = -0x40000
			else
				player[currentPlayer].xvel = 0x40000
			end if
			player[currentPlayer].speed = player[currentPlayer].xvel
			PlaySfx(SfxName[Mini Bumper], 0)
			break
		end switch
	next
end event


event ObjectDraw
	object.timer++
	if object.timer > 3
		tileLayer[0].deformationOffsetW++
		tileLayer[1].deformationOffset++
		object.timer = 0
	end if
end event


event ObjectStartup
	object[SLOT_SETUP].type = TypeName[R3 Setup]

	object[SLOT_SETUP].priority = PRIORITY_ACTIVE
	object[SLOT_SETUP].drawOrder = 0

	switch hud.currentTimePeriod
	case TIME_PRESENT
		object[SLOT_SETUP].drawFunction = R3Setup_drawFunctionPresent
		break

	case TIME_PAST
		object[SLOT_SETUP].drawFunction = R3Setup_drawFunctionPast
		break

	case TIME_GOOD_FUTURE
		object[SLOT_SETUP].drawFunction = R3Setup_drawFunctionGoodFuture
		break

	case TIME_BAD_FUTURE
		object[SLOT_SETUP].drawFunction = R3Setup_drawFunctionBadFuture
		break

	end switch

	SetLayerDeformation(DEFORM_BG, 128, 6, 0, 0, 0)
	SetLayerDeformation(DEFORM_FG_WATER, 64, 4, 0, 0, 0)

	stage.waterLevel = 1472

	if hud.currentTimePeriod == TIME_PAST
		R3Setup_rotoDrumTimer = 1
	else
		R3Setup_rotoDrumTimer = 0
	end if

end event

// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

//---Aliases//
private alias object.value0	 :	object.timer
private alias object.value1	 :	object.yoffset

//---States---//
private alias 0	:	PISTON_DETECT_PLAYER
private alias 1	:	PISTON_PREPARE_FIRE
private alias 2	:	PISTON_EMERGE
private alias 3	:	PISTON_FIRE
private alias 4	:	PISTON_HIDE

// Function declarations
reserve function Piston_DebugDraw
reserve function Piston_DebugSpawn


public function Piston_DebugDraw
	DrawSprite(1)
	DrawSprite(0)
end function


public function Piston_DebugSpawn
	CreateTempObject(TypeName[Piston], 0, object.xpos, object.ypos)
	object[tempObjectPos].frame = 1
end function


event ObjectUpdate
	switch object.state
	case PISTON_DETECT_PLAYER
		BoxCollisionTest(C_TOUCH, object.entityPos, -112, -256, 112, 256, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object.timer < 80
				object.timer++
				object.frame = 1
			else
				object.timer = 6
				object.state = PISTON_PREPARE_FIRE
			end if
		end if
		break
	case PISTON_PREPARE_FIRE
		if object.timer < 33
			object.timer++
			object.frame = object.timer
			object.frame /= 3
		else
			object.frame = 1
			object.timer = 0
			object.state = PISTON_EMERGE
		end if
		break
	case PISTON_EMERGE
		if object.yoffset < 2097152
			object.yoffset += 524288
		else
			object.state = PISTON_FIRE
		end if
		break
	case PISTON_FIRE
		if object.timer < 8
			if object.timer == 0
				switch hud.currentTimePeriod
				case TIME_PRESENT
				case TIME_BAD_FUTURE
					CreateTempObject(TypeName[Fireball], 0, object.xpos, object.ypos)
					object[tempObjectPos].xpos -= 1572864
					object[tempObjectPos].ypos -= 1310720
					object[tempObjectPos].value0 = 64
					object[tempObjectPos].xvel = -65536
					object[tempObjectPos].inkEffect = 2
					object[tempObjectPos].alpha = 255
					object[tempObjectPos].frame = 0

					CreateTempObject(TypeName[Fireball], 0, object.xpos, object.ypos)
					object[tempObjectPos].xpos += 1572864
					object[tempObjectPos].ypos -= 1310720
					object[tempObjectPos].value0 = 64
					object[tempObjectPos].xvel = 65536
					object[tempObjectPos].inkEffect = 2
					object[tempObjectPos].alpha = 255
					object[tempObjectPos].frame = 1
					
				end switch
			end if
			object.timer++
		else
			object.timer = 0
			object.state = PISTON_HIDE
		end if
		break
	case PISTON_HIDE
		if object.yoffset > 0
			object.yoffset -= 262144
		else
			object.state = PISTON_DETECT_PLAYER
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp0 = -786432
		temp0 -= object.yoffset
		temp0 >>= 16
		BoxCollisionTest(C_SOLID, object.entityPos, -16, -12, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -16, temp0, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object.yoffset == 2097152
				if object.state == PISTON_EMERGE
					object[currentPlayer].yvel = -458752
					object[currentPlayer].value1 = 0
					object[currentPlayer].gravity = 1
					object[currentPlayer].animation = ANI_BOUNCING
					object[currentPlayer].state = Player_State_Air
				end if
			end if
		end if
	next
end event


event ObjectDraw
	temp0 = object.ypos
	temp0 -= object.yoffset
	DrawSpriteXY(object.frame, object.xpos, temp0)

#platform: USE_DECOMP
	// fixes a bug where the piston draws above the foreground
	object.drawOrder = 3
#endplatform

	DrawSprite(0)
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects.gif")

	SpriteFrame(-16, -4, 32, 8, 214, 34)		// #0 - Piston Head
	SpriteFrame(-16, -12, 32, 40, 1, 34)		// #1 - Piston

	switch object[arrayPos0].propertyValue
	case TIME_PRESENT
	case TIME_BAD_FUTURE
		SpriteFrame(-16, -12, 32, 40, 34, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 34, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 34, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 34, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 34, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		break
	case TIME_PAST
	case TIME_GOOD_FUTURE
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		SpriteFrame(-16, -12, 32, 40, 1, 34)
		break
	end switch

	foreach (TypeName[Piston], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].frame = 1
	next

	SetTableValue(TypeName[Piston], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(Piston_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(Piston_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++

end event

	DrawSprite(0)

	if editor.showGizmos == true
		temp0   = Object.YPos
		temp0  -= 0x200000
		DrawArrow(Object.XPos, Object.YPos, Object.XPos, temp0, 255, 0, 0, 0)
	end if
end event

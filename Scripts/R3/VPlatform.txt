private alias 45 : TYPE_VPLATFORM

// Function declarations
reserve function VPlatform_DebugDraw
reserve function VPlatform_DebugSpawn


public function VPlatform_DebugDraw
	DrawSprite(0)
end function


public function VPlatform_DebugSpawn
	CreateTempObject(TypeName[V Platform], 0, object.xpos, object.ypos)
	object[tempObjectPos].ypos &= -65536
end function


event ObjectUpdate
	object.angle = oscillation
	object.angle *= 3
	if object.propertyValue == 1
		object.angle += 256
	end if
	Sin(object.value2, object.angle)
	object.value2 *= 9984
	object.value2 += object.ypos
	object.value2 &= -65536
	object.value2 -= object.value1
	temp0 = object.ypos
	object.ypos = object.value1
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			object[currentPlayer].ypos += object.value2
		end if
	next
	object.ypos = temp0
end event


event ObjectDraw
	Sin(object.value1, object.angle)
	object.value1 *= 9984
	object.value1 += object.ypos
	object.value1 &= -65536
	DrawSpriteXY(0, object.xpos, object.value1)
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects.gif")
	SpriteFrame(-32, -16, 64, 32, 100, 1)
	SetTableValue(TypeName[V Platform], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(VPlatform_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(VPlatform_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

event RSDKDraw
	DrawSprite(0)
end event

end event

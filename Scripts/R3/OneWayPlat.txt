		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -8, 32, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
	next
	if object[0].ypos < object.ypos
		if object.value0 < 4194304
			object.value0 += 1048576
		end if
	else
		if object.value0 > 0
			object.value0 -= 1048576
		end if
	end if
end event

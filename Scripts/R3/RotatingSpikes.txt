public function RotatingSpikes_DebugDraw

public function RotatingSpikes_DebugSpawn
	CreateTempObject(TypeName[Rotating Spikes], 0, object.xpos, object.ypos)
	switch object[tempObjectPos].propertyValue
	case 0
		object[tempObjectPos].speed = 2
		if hud.currentTimePeriod > 1
			object[tempObjectPos].value2 = 4
		else
			object[tempObjectPos].value2 = 3
		end if
		break
	case 1
		object[tempObjectPos].speed = -2
		if hud.currentTimePeriod > 1
			object[tempObjectPos].value2 = 4
		else
			object[tempObjectPos].value2 = 3
		end if
		break
	case 2
		object[tempObjectPos].speed = 2
		object[tempObjectPos].value2 = 2
		break
	case 3
		object[tempObjectPos].speed = -2
		object[tempObjectPos].value2 = 2
		break
	end switch
end function


// ========================

event ObjectUpdate
	object.value0 += object.speed
	if object.value0 < 0
		object.value0 += 512
	end if
	object.value0 &= 511
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp0 = 0
		temp1 = 16
		temp2 = object.xpos
		temp3 = object.ypos
		while temp0 < object.value2
			Cos(object.xpos, object.value0)
			object.xpos *= temp1
			object.xpos <<= 7
			object.xpos += temp2
			Sin(object.ypos, object.value0)
			object.ypos *= temp1
			object.ypos <<= 7
			object.ypos += temp3
			BoxCollisionTest(C_TOUCH, object.entityPos, -8, -8, 8, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				CallFunction(Player_Hit)
			end if
			temp0++
			temp1 += 16
		loop
		object.xpos = temp2
		object.ypos = temp3
	next
end event


event ObjectDraw
	DrawSprite(0)
	temp0 = 0
	temp1 = 16
	while temp0 < object.value2
		Cos(temp2, object.value0)
		temp2 *= temp1
		temp2 <<= 7
		temp2 += object.xpos
		Sin(temp3, object.value0)
		temp3 *= temp1
		temp3 <<= 7
		temp3 += object.ypos
		DrawSpriteXY(1, temp2, temp3)
		temp0++
		temp1 += 16
	loop
end event

event ObjectUpdate
	if object.value0 == 1
		if object.value1 < 16
			object.value1++
			temp0 = object.value1
			temp0 &= 3
			if temp0 == 3
				object.value2 = 65536
			else
				object.value2 = 0
			end if
		else
			object.value2 = 0
		end if
	else
		if object.value1 > 0
			object.value1--
			temp0 = object.value1
			temp0 &= 3
			if temp0 == 3
				object.value2 = -65536
			else
				object.value2 = 0
			end if
		else
			object.value2 = 0
			object.ypos = object.value3
		end if
	end if
	object.value0 = 0
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -17, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			object.value0 = 1
			object[currentPlayer].ypos += object.value2
			object[currentPlayer].ypos += 65536
		end if
	next
	object.ypos += object.value2
end event

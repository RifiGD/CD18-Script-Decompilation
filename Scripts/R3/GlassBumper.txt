event ObjectUpdate
	if object.state == 1
		object.value0++
		if object.value0 > 15
			object.type = TypeName[Blank Object]
		end if
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object[currentPlayer].animation != ANI_HURT
			if object.state == 0
				BoxCollisionTest(C_TOUCH, object.entityPos, -14, -14, 14, 14, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == 1
					PlaySfx(SfxName[Glass Smash], 0)
					CreateTempObject(TypeName[Object Score], 0, object.xpos, object.ypos)
					player.score += 100
					object.state = 1
					object.priority = 1
					temp0 = object[currentPlayer].xpos
					temp0 -= object.xpos
					temp1 = object[currentPlayer].ypos
					temp1 -= object.ypos
					ATan2(temp2, temp0, temp1)
					Cos256(temp0, temp2)
					Sin256(temp1, temp2)
					temp0 *= 1792
					temp1 *= 1792
					object[currentPlayer].xvel = temp0
					object[currentPlayer].yvel = temp1
					object[currentPlayer].speed = object[currentPlayer].xvel
					object[currentPlayer].gravity = 1
					object[currentPlayer].value1 = 0
					temp0 = object.ixpos
					temp1 = object.iypos
					temp0 -= 8
					temp1 -= 8
					Set16x16TileInfo(0, temp0, temp1, 0)
					temp0 += 16
					Set16x16TileInfo(0, temp0, temp1, 0)
					temp0 -= 16
					temp1 += 16
					Set16x16TileInfo(0, temp0, temp1, 0)
					temp0 += 16
					Set16x16TileInfo(0, temp0, temp1, 0)
				end if
			end if
		end if
	next
end event

end event

private alias 51 : TYPE_FIREBALL



event ObjectUpdate
	object.xpos += object.xvel
	if object.xvel > 0
		object.xvel += 4096
	else
		object.xvel -= 4096
	end if
	if object.outOfBounds == 1
		object.type = TypeName[Blank Object]
	else
		temp0 = screen.yoffset
		temp0 -= 8
		temp0 <<= 16
		if object.ypos < temp0
			object.type = TypeName[Blank Object]
		end if
	end if
	if object.value0 > 0
		object.value0--
		if object.value0 < 32
			object.alpha = object.value0
			object.alpha <<= 3
		end if
	else
		object.type = TypeName[Blank Object]
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -8, -6, 8, 6, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			CallFunction(Player_FireHit)
		end if
	next
end event


event ObjectDraw
	DrawSpriteFX(object.frame, FX_INK, object.xpos, object.ypos)
end event

end event

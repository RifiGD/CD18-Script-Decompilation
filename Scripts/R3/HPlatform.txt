private alias 46 : TYPE_HPLATFORM

// Function declarations
reserve function HPlatform_DebugDraw
reserve function HPlatform_DebugSpawn


public function HPlatform_DebugDraw
	DrawSprite(0)
end function


public function HPlatform_DebugSpawn
	CreateTempObject(TypeName[H Platform], 0, object.xpos, object.ypos)
	object[tempObjectPos].xpos &= -65536
	object[tempObjectPos].ypos &= -65536
end function


event ObjectUpdate
	object.angle = oscillation
	object.angle *= 3
	if object.propertyValue == 1
		object.angle += 256
	end if
	Cos(object.value4, object.angle)
	object.value4 *= 9984
	object.value4 += object.xpos
	object.value4 &= -65536
	object.value4 -= object.value3
	if object.value0 == 1
		if object.value1 < 16
			object.value1++
			temp0 = object.value1
			temp0 &= 3
			if temp0 == 3
				object.value2 = 65536
			else
				object.value2 = 0
			end if
		else
			object.value2 = 0
		end if
	else
		if object.value1 > 0
			object.value1--
			temp0 = object.value1
			temp0 &= 3
			if temp0 == 3
				object.value2 = -65536
			else
				object.value2 = 0
			end if
		else
			object.value2 = 0
		end if
	end if
	object.value0 = 0
	temp0 = object.xpos
	object.xpos = object.value3
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			object.value0 = 1
			object[currentPlayer].xpos += object.value4
			object[currentPlayer].ypos += object.value2
		end if
	next
	object.xpos = temp0
	object.ypos += object.value2
end event


event ObjectDraw
	Cos(object.value3, object.angle)
	object.value3 *= 9984
	object.value3 += object.xpos
	object.value3 &= -65536
	DrawSpriteXY(0, object.value3, object.ypos)
end event


event ObjectStartup
	LoadSpriteSheet("R3/Objects.gif")
	SpriteFrame(-32, -16, 64, 32, 100, 1)
	SetTableValue(TypeName[H Platform], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(HPlatform_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(HPlatform_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

event ObjectUpdate
	switch object.state
	case 1
		if object.ypos > 84934656
			object.ypos -= 163840
		else
			object.ypos = 84934656
			object.state = 2
		end if
		object.frame++
		object.frame %= 9
		break
	case 2
		temp0 = object.value0
		temp0 &= 3
		if temp0 == 0
			Rand(temp0, 96)
			temp0 -= 48
			temp0 <<= 16
			temp0 += object.xpos
			Rand(temp1, 40)
			temp1 -= 20
			temp1 <<= 16
			temp1 += object.ypos
			CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
			PlaySfx(SfxName[Explosion], 0)
		end if
		object.value0++
		if object.value0 == 120
			object.type = TypeName[Blank Object]
		end if
		break
	end switch
	if object.state == 0
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, -48, -16, 48, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.state = 1
				object[currentPlayer].yvel = 0
				PlaySfx(SfxName[Explosion], 0)
			end if
		next
	end if
end event

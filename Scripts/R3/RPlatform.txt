event ObjectUpdate
	if object.outOfBounds == 1
		object.ypos = object.value3
		object.state = 6
		object.value1 = 0
	end if
	switch object.state
	case 0
		object.value2 = 0
		if object.value0 == 1
			object.state = 1
			if object.priority != 4
				object.priority = 1
			end if
			object.yvel = 0
			object.value4 = 0
		end if
		break
	case 1
		if object.value4 < 8
			object.value4++
		else
			object.value4 = 0
			object.state++
		end if
		break
	case 2
		if object.yvel > -262144
			object.yvel -= 4096
		else
			object.state++
		end if
		break
	case 3
		if object.value4 < 36
			object.value4++
		else
			object.value4 = 0
			object.state++
		end if
		break
	case 4
		if object.yvel < 0
			object.yvel += 4096
		else
			object.yvel = 0
			object.state++
		end if
		break
	case 5
		if object.value0 == 1
			if object.value1 < 16
				object.value1++
				temp0 = object.value1
				temp0 &= 3
				if temp0 == 3
					object.yvel = 65536
				else
					object.yvel = 0
				end if
			else
				object.yvel = 0
			end if
		else
			if object.value1 > 0
				object.value1--
				temp0 = object.value1
				temp0 &= 3
				if temp0 == 3
					object.yvel = -65536
				else
					object.yvel = 0
				end if
			else
				object.yvel = 0
			end if
		end if
		break
	case 6
		if object.outOfBounds == 1
			object.state = 0
			if object.priority != 4
				object.priority = 0
			end if
		end if
		break
	end switch
	object.value0 = 0
	temp0 = object.ypos
	temp0 += object.yvel
	object.ypos &= -65536
	object.value2 = temp0
	object.value2 &= -65536
	object.value2 -= object.ypos
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			object.value0 = 1
			object[currentPlayer].ypos += object.value2
		end if
	next
	object.ypos = temp0
end event

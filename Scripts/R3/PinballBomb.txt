event ObjectUpdate
	object.yvel += 8192
	if object.yvel > 262144
		object.drawOrder = 3
	end if
	object.xpos += object.xvel
	object.ypos += object.yvel
	object.value0++
	object.value0 &= 3
	if object.xvel > 0
		ObjectTileCollision(1, 8, 0, 0)
	else
		ObjectTileCollision(3, -8, 0, 0)
	end if
	if checkResult == true
		FlipSign(object.xvel)
	end if
	ObjectTileCollision(0, 0, 8, 0)
	if checkResult == true
		FlipSign(object.yvel)
		object.yvel >>= 1
		if object.value3 < 1
			object.value3++
		else
			PlaySfx(SfxName[Explosion], 0)
			ResetObjectEntity(object.entityPos, TypeName[Explosion], 0, object.xpos, object.ypos)
		end if
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -8, -8, 8, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			PlaySfx(SfxName[Explosion], 0)
			ResetObjectEntity(object.entityPos, TypeName[Explosion], 0, object.xpos, object.ypos)
			object.priority = 1
			temp0 = object[currentPlayer].xpos
			temp0 -= object.xpos
			temp1 = object[currentPlayer].ypos
			temp1 -= object.ypos
			ATan2(temp2, temp0, temp1)
			Cos256(temp0, temp2)
			Sin256(temp1, temp2)
			temp0 *= 1792
			temp1 *= 1792
			object[currentPlayer].xvel = temp0
			object[currentPlayer].yvel = temp1
			object[currentPlayer].speed = object[currentPlayer].xvel
			object[currentPlayer].gravity = 1
			object[currentPlayer].value1 = 0
		end if
	next
end event


event ObjectDraw
	if object.value0 == 0
		DrawSprite(0)
	else
		DrawSprite(1)
	end if
end event

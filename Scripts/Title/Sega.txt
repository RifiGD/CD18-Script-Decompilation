private alias 1 : TYPE_SEGA

public alias 256 : GROUP_PLAYERS
public alias arrayPos6 : currentPlayer
public alias arrayPos7 : playerCount



event ObjectUpdate
	switch object.state
	case 0
		object.state = 1
		object.value0 = 480
		object.alpha = 256
		object.inkEffect = 2
		SetScreenFade(0, 0, 0, 255)
		object.ixpos = screen.xcenter
		break
	case 1
		if object.value0 > 0
			object.value0 -= 8
		else
			object.state = 2
		end if
		SetScreenFade(0, 0, 0, object.value0)
		break
	case 2
		object.value0++
		if object.value0 == 160
			object.value0 = 0
			object.state = 3
		end if
		if object.value0 == 2
			PlaySfx(SfxName[Sega Chant], 0)
		end if
		break
	case 3
		if object.alpha > 0
			object.alpha -= 8
		else
			if object.value0 == 16
				ResetObjectEntity(object.entityPos, TypeName[Sonic], 0, object.xpos, object.ypos)
				object.inkEffect = 2
				object.alpha = 0
				object.value0 = 0
				object.value3 = 96
			else
				object.value0++
			end if
		end if
	end switch
end event


event ObjectDraw
	DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 96, 255)
	if object.alpha > 8
		DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("Title/Title.gif")
	SpriteFrame(-49, -16, 98, 31, 106, 1)
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

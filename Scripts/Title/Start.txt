event ObjectUpdate
	if object.state == 0
		if options.gameMode == 0
			CallFunction(Start_HandleCheatCodes)
		end if
	end if
	switch object.state
	case START_SETUP
		//This still has touchscreen support?
		CheckTouchRect(0, 0, screen.xsize, 240)
		if keyPress[0].start == true
			checkResult = false
		end if
		if keyPress[0].buttonA == true
			checkResult = false
		end if
		if keyPress[0].buttonB == true
			checkResult = false
		end if
		if keyPress[0].buttonC == true
			checkResult = false
		end if
		if checkResult > -1
			object.state = START_NEWGAME
			StopMusic()
			PlaySfx(SfxName[Select], 0)
		else
			object.value4++
			if object.value4 == 1550
				object.state = 5
			end if
		end if
		break
	case START_NEWGAME
		screen.fade += 4
		if screen.fade == 384
		
			if options.soundtrack == 0
			// Interestingly, video playback was turned into a Native Function for this port.
				CallNativeFunction2(PlayVideo, "Opening", 0)
			else
				CallNativeFunction2(PlayVideo, "OpeningUS", 0)
			end if

			// Off to Palmtree Panic you go!
			stage.activeList 	= REGULAR_STAGE
			stage.listPos 		= 0
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_STAGESELECT
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 3
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_SOUNDTEST
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 4
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_DAGARDEN
		screen.fade += 4
		if screen.fade == 384
			options.gameMode 	= 0
			stage.listPos 		= 5
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	case START_DEMO
		screen.fade += 4
		if screen.fade == 384
			fadeColor = 0
			player.score = 0
			player.lives = 3
			player.scoreBonus 		= 50000
			generator.destroyed 	= false
			metalSonic.destroyed 	= false
			goodFuture.enabled 		= false
			goodFuture.list 		= 0
			goodFuture.count 		= 0
			options.gameMode 		= 0
			options.attractMode 	= true
			
			switch options.demoNumber
			case DEMO_R11A
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 0
				break
			case DEMO_SS1
				stage.activeList 	= SPECIAL_STAGE
				stage.listPos 		= 0
				break
			case DEMO_R43C
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 28
				break
			case DEMO_SS6
				stage.activeList 	= SPECIAL_STAGE
				stage.listPos 		= 5
				break
			case DEMO_R82A
				stage.activeList 	= REGULAR_STAGE
				stage.listPos 		= 64
				break
			end switch
			options.demoNumber++
			options.demoNumber %= 5
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, screen.fade)
		break
	end switch
end event


event ObjectDraw
	object.animationTimer++
	if object.animationTimer > 59
		object.animationTimer = 0
	end if
	if object.animationTimer > 29
		if options.touchControls == false
			DrawSprite(0)
		else
			DrawSprite(1)
		end if
	end if
end event

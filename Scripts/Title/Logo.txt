private alias 3 : TYPE_LOGO



event ObjectDraw
	temp0 = screen.xcenter
	temp0 <<= 1
	DrawSpriteScreenXY(16, 0, 100)
	DrawSpriteScreenXY(17, temp0, 100)
	switch object.state
	case 0
		SetScreenFade(255, 255, 255, object.value0)
		object.direction = 0
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)
		object.direction = 1
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)
		DrawSpriteScreenXY(2, screen.xcenter, 120)
		temp0 = screen.xcenter
		temp0 -= 110
		DrawSpriteScreenXY(1, temp0, 135)
		temp0 += 121
		DrawSpriteScreenXY(4, temp0, 110)
		if object.value0 > 0
			object.value0 -= 8
		else
			object.state = 1
			object.frame = 4
		end if
		break
	case 1
		object.direction = 0
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)
		object.direction = 1
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)
		DrawSpriteScreenXY(2, screen.xcenter, 120)
		if object.frame > 7
			if object.frame < 13
				temp0 = screen.xcenter
				temp0 -= 43
				DrawSpriteScreenXY(3, temp0, 48)
			end if
		end if
		temp0 = screen.xcenter
		temp0 -= 110
		DrawSpriteScreenXY(1, temp0, 135)
		temp0 += 121
		DrawSpriteScreenXY(object.frame, temp0, 110)
		object.value0++
		if object.value0 > 1
			object.value0 = 0
			object.frame++
			if object.frame > 15
				object.state = 2
				object[+1].type = TypeName[Start]
				object[+1].ixpos = screen.xcenter
				object[+1].iypos = 200
			end if
		end if
		break
	case 2
		object.direction = 0
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)
		object.direction = 1
		DrawSpriteScreenFX(0, FX_FLIP, screen.xcenter, 40)
		DrawSpriteScreenXY(2, screen.xcenter, 120)
		temp0 = screen.xcenter
		temp0 -= 110
		DrawSpriteScreenXY(1, temp0, 135)
		temp0 += 121
		DrawSpriteScreenXY(4, temp0, 110)
		break
	end switch
end event

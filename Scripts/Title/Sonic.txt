event ObjectDraw
	DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 96, 255)
	switch object.state
	case 0
		if object.alpha > 8
			DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
		end if
		if object.value0 < 256
			object.value0 += 8
			if object.value0 < 256
				object.alpha = object.value0
			else
				object.alpha = 255
			end if
		else
			if object.value0 < 272
				object.value0++
			else
				object.value0 = 0
				object.state = 1
				lampPostID = 0
				PlayMusic(options.soundtrack)
			end if
		end if
		break
	case 1
		temp0 = object.value0
		temp0 >>= 2
		DrawSprite(temp0)
		temp0 = 11
		temp0 += screen.xcenter
		if object.value0 > 15
			DrawSpriteScreenXY(5, temp0, 72)
		end if
		object.value0++
		if object.value0 > 19
			ResetObjectEntity(object.entityPos, TypeName[Logo], 0, object.xpos, object.ypos)
			object.value0 = 256
			arrayPos0 = object.entityPos
			arrayPos0 -= 2
			object[arrayPos0].type = TypeName[Background]
			object[arrayPos0].drawOrder = 0
			stage[2].activeLayer = 9
			arrayPos0++
			object[arrayPos0].type = TypeName[Clouds]
		end if
		break
	end switch
end event

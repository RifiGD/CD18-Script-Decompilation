event ObjectUpdate
	switch object.state
	case 0
		if object.value0 < 8
			object.value0++
		else
			object.value0 = 288
			SetActivePalette(1, 0, screen.ysize)
			object.state++
		end if
		SetScreenFade(0, 0, 0, 255)
		break
	case 1
		if object.value0 > 0
			object.value0 -= 16
		else
			PlayMusic(options.soundtrack)
			object.state++
		end if
		SetScreenFade(0, 0, 0, object.value0)
		break
	case 2
		if object.value0 < 60
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 3
		if keyPress[0].buttonA == true
			object.state = 4
		end if
		if keyPress[0].buttonB == true
			object.state = 4
		end if
		if keyPress[0].buttonC == true
			object.state = 4
		end if
		if keyPress[0].start == true
			object.state = 4
		end if
		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		if checkResult > -1
			object.state = 4
		end if
		break
	case 4
		if object.value0 < 288
			object.value0 += 8
			music.volume -= 5
		else
			SetActivePalette(0, 0, screen.ysize)
			StopMusic()
			options.gameMode = 0
			stage.activeList = PRESENTATION_STAGE
			stage.listPos = 3
			LoadStage()
		end if
		temp0 = object.value0
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break
	end switch
end event

	DrawSpriteScreenXY(0, screen.xcenter, screen.ycenter)
end event

private alias 3 : TYPE_SECRETIMAGE



event ObjectUpdate
	switch object.state
	case 0
		if object.value0 < 8
			object.value0++
		else
			object.value0 = 288
			object.state++
		end if
		SetScreenFade(0, 0, 0, 255)
		break
	case 1
		if object.value0 > 0
			object.value0 -= 16
		else
			PlayMusic(options.soundtrack)
			object.state++
		end if
		SetScreenFade(0, 0, 0, object.value0)
		break
	case 2
		if object.value0 < 60
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 3
		if keyPress[0].buttonA == 1
			object.state = 4
		end if
		if keyPress[0].buttonB == 1
			object.state = 4
		end if
		if keyPress[0].buttonC == 1
			object.state = 4
		end if
		if keyPress[0].start == 1
			object.state = 4
		end if
		CheckTouchRect(0, 0, screen.xsize, screen.ysize)
		if checkResult > -1
			object.state = 4
		end if
		break
	case 4
		if object.value0 < 288
			object.value0 += 8
			music.volume -= 5
		else
			StopMusic()
			options.gameMode = 0
			stage.activeList = PRESENTATION_STAGE
			stage.listPos = 0
			LoadStage()
		end if
		temp0 = object.value0
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break
	end switch
end event


event ObjectDraw
	switch object.propertyValue
	case 0
		DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)
		DrawSpriteScreenXY(0, screen.xcenter, screen.ycenter)
		break
	case 1
		DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)
		DrawSpriteScreenXY(1, screen.xcenter, screen.ycenter)
		break
	case 2
		DrawRect(0, 0, screen.xsize, screen.ysize, 238, 238, 204, 255)
		DrawSpriteScreenXY(2, screen.xcenter, screen.ycenter)
		break
	case 3
		DrawRect(0, 0, screen.xsize, screen.ysize, 0, 0, 0, 255)
		DrawSpriteScreenXY(3, screen.xcenter, screen.ycenter)
		break
	case 4
		DrawRect(0, 0, screen.xcenter, screen.ysize, 0, 34, 221, 255)
		DrawRect(screen.xcenter, 0, screen.xcenter, screen.ysize, 0, 34, 204, 255)
		DrawSpriteScreenXY(4, screen.xcenter, screen.ycenter)
		DrawSpriteScreenXY(5, screen.xcenter, screen.ycenter)
		break
	end switch
end event

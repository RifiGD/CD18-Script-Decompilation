event ObjectUpdate
	switch object.state
	case 0
		if object.value0 < 8
			object.value0++
		else
			object.value0 = 254
			object.state++
		end if
		SetScreenFade(0, 0, 0, 255)
		break
	case 1
		if object.value0 > 0
			SetScreenFade(0, 0, 0, object.value0)
			object.value0 -= 8
		else
			object.state++
		end if
		break
	case 2
		if keyPress[0].up == true
			object.value2 = 0
			object.value1--
			if object.value1 < 0
				object.value1 = 69
			end if
		else
			if keyPress[0].down == true
				object.value2 = 0
				object.value1++
				if object.value1 > 69
					object.value1 = 0
				end if
			end if
		end if
		if keyDown[0].up == true
			object.value2++
			if object.value2 == 8
				object.value2 = 0
				object.value1--
				if object.value1 < 0
					object.value1 = 69
				end if
			end if
		else
			if keyDown[0].down == true
				object.value2++
				if object.value2 == 8
					object.value2 = 0
					object.value1++
					if object.value1 > 69
						object.value1 = 0
					end if
				end if
			end if
		end if
		if keyPress[0].start == true
			object.state = 3
		end if
		if keyPress[0].buttonB == true
			object.state = 4
		end if
		if engine.deviceType == MOBILE
			temp0 = screen.xcenter
			temp0 -= 138
			temp1 = screen.xcenter
			temp1 -= 76
			CheckTouchRect(temp0, 20, temp1, 72)
			if checkResult > -1
				keyDown[0].up = 1
				object.value2++
				if object.value2 == 8
					object.value2 = 0
					object.value1--
					if object.value1 < 0
						object.value1 = 69
					end if
				end if
			end if
			CheckTouchRect(temp0, 162, temp1, 214)
			if checkResult > -1
				keyDown[0].down = 1
				object.value2++
				if object.value2 == 8
					object.value2 = 0
					object.value1++
					if object.value1 > 69
						object.value1 = 0
					end if
				end if
			end if
			temp0 = screen.xcenter
			temp0 += 28
			temp1 = screen.xcenter
			temp1 += 68
			CheckTouchRect(temp0, 162, temp1, 214)
			if checkResult > -1
				object.value3 = 1
			else
				CheckTouchRect(0, 0, screen.xsize, screen.ysize)
				if checkResult < 0
					if object.value3 == 1
						object.value3 = 0
						PlaySfx(SfxName[Select], 0)
						object.state = 4
					end if
				else
					if object.value3 == 1
						object.value3 = 0
					end if
				end if
			end if
			temp0 += 48
			temp1 += 48
			CheckTouchRect(temp0, 162, temp1, 214)
			if checkResult > -1
				object.value3 = 2
			else
				CheckTouchRect(0, 0, screen.xsize, screen.ysize)
				if checkResult < 0
					if object.value3 == 2
						object.value3 = 0
						PlaySfx(SfxName[Select], 0)
						object.state = 3
					end if
				else
					if object.value3 == 2
						object.value3 = 0
					end if
				end if
			end if
		end if
		break
	case 3
		if object.value0 < 288
			object.value0 += 8
		else
			options.gameMode = 0
			player.lives = 3
			stage.activeList = REGULAR_STAGE
			stage.listPos = object.value1
			LoadStage()
		end if
		temp0 = object.value0
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break
	case 4
		if object.value0 < 288
			object.value0 += 8
		else
			options.gameMode = 0
			stage.activeList = PRESENTATION_STAGE
			stage.listPos = 0
			LoadStage()
		end if
		temp0 = object.value0
		if temp0 > 255
			temp0 = 255
		end if
		SetScreenFade(0, 0, 0, temp0)
		break
	end switch
end event


event ObjectDraw
	temp0 = 0
	while temp0 < 240
		temp1 = 0
		while temp1 < screen.xsize
			DrawSpriteScreenXY(15, temp1, temp0)
			temp1 += 224
		loop
		temp0 += 48
	loop
	DrawSpriteScreenXY(16, screen.xcenter, 80)
	temp0 = screen.xcenter
	temp0 -= 48
	temp1 = object.value1
	temp1 /= 10
	switch temp1
	case 0
		DrawSpriteScreenXY(1, temp0, 112)
		break
	case 1
		DrawSpriteScreenXY(3, temp0, 112)
		break
	case 2
		DrawSpriteScreenXY(4, temp0, 112)
		break
	case 3
		DrawSpriteScreenXY(5, temp0, 112)
		break
	case 4
		DrawSpriteScreenXY(6, temp0, 112)
		break
	case 5
		DrawSpriteScreenXY(7, temp0, 112)
		break
	case 6
		DrawSpriteScreenXY(8, temp0, 112)
		break
	end switch
	temp0 += 8
	DrawSpriteScreenXY(14, temp0, 112)
	temp0 += 8
	temp1 = object.value1
	temp1 %= 10
	temp1 >>= 2
	temp1++
	DrawSpriteScreenXY(temp1, temp0, 112)
	temp0 += 8
	DrawSpriteScreenXY(14, temp0, 112)
	temp0 += 8
	temp1 = object.value1
	temp1 %= 10
	switch temp1
	case 0
	case 4
		DrawSpriteScreenXY(10, temp0, 112)
		break
	case 1
	case 5
		DrawSpriteScreenXY(11, temp0, 112)
		break
	case 2
	case 6
	case 8
		DrawSpriteScreenXY(12, temp0, 112)
		break
	case 3
	case 7
	case 9
		DrawSpriteScreenXY(13, temp0, 112)
		break
	end switch
	if engine.deviceType == MOBILE
		temp0 -= 96
		if keyDown[0].up == true
			DrawSpriteScreenXY(18, temp0, 48)
		else
			DrawSpriteScreenXY(17, temp0, 48)
		end if
		if keyDown[0].down == true
			DrawSpriteScreenXY(20, temp0, 192)
		else
			DrawSpriteScreenXY(19, temp0, 192)
		end if
		temp0 += 160
		if object.value3 == 1
			DrawSpriteScreenXY(24, temp0, 192)
		else
			DrawSpriteScreenXY(23, temp0, 192)
		end if
		temp0 += 48
		if object.value3 == 2
			DrawSpriteScreenXY(22, temp0, 192)
		else
			DrawSpriteScreenXY(21, temp0, 192)
		end if
	end if
end event

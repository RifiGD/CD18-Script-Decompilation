event ObjectUpdate
	switch object.state
	case 0
		object.state = 1
		object.value0 = 384
		SetScreenFade(0, 0, 0, 255)
		PlayMusic(options.soundtrack)
		break
	case 1
		if object.value0 > 0
			object.value0 -= 8
		else
			object.state = 2
		end if
		SetScreenFade(0, 0, 0, object.value0)
		break
	case 2
		if keyPress[0].buttonB == 1
			StopMusic()
			object.state = 6
		end if
		if keyPress[0].buttonA == 1
			object.state = 3
			object[33].xpos = screen.xsize
			object[33].xpos += 96
			object[33].xpos <<= 16
			object[33].state = 1
			object[33].drawOrder = 5
			object[33].priority = 1
			object[35].propertyValue = 4
			object[36].propertyValue = 5
			object[37].propertyValue = 6
		end if
		break
	case 3
		if object[33].state == 0
			if keyPress[0].left == 1
				object[33].state = 2
			end if
			if keyPress[0].right == 1
				object[33].state = 4
			end if
			if keyPress[0].buttonA == 1
				object.state = 4
				StopMusic()
			end if
			if keyPress[0].start == 1
				object.state = 4
				StopMusic()
			end if
			if keyPress[0].buttonC == 1
				object.state = 4
				StopMusic()
			end if
		end if
		if keyPress[0].buttonB == 1
			StopMusic()
			object.state = 6
		end if
		break
	case 4
		if object.value0 < 288
			object.value0 += 8
		else
			object[33].xpos = screen.xsize
			object[33].xpos += 96
			object[33].xpos <<= 16
			object[33].state = 0
			object[33].priority = 0
			object[35].propertyValue = 0
			object[36].propertyValue = 2
			object[37].propertyValue = 7
			CallFunction(MenuControl_Function0)
			object.state++
		end if
		SetScreenFade(208, 255, 224, object.value0)
		break
	case 5
		if object.value0 > 0
			object.value0 -= 8
		else
			object.state = 2
		end if
		SetScreenFade(208, 255, 224, object.value0)
		break
	case 6
		if object.value0 < 384
			object.value0 += 8
		else
		// CD11 Leftover:
		// -> listPos 1 is the MENU in the GameConfig.
		// However, in CD18, it just redirects to the Title Screen
			stage.listPos = BACK_TO_MENU
			LoadStage()
		end if
		SetScreenFade(0, 0, 0, object.value0)
		break
	end switch
end event


event ObjectStartup
	SetMusicTrack("JP/DAGarden.ogg", 0, 1)
	SetMusicTrack("US/DAGarden.ogg", 1, 117382)
	object[0].type = TypeName[Menu Control]
	object[0].drawOrder = 1
	object[0].priority = 1
	if engine.deviceType == MOBILE
		object[34].drawOrder = 5
		object[35].drawOrder = 5
		object[36].drawOrder = 5
		object[37].drawOrder = 5
		temp0 = screen.xsize
		temp0 >>= 2
		temp1 = temp0
		temp1 >>= 1
		object[34].ixpos = temp1
		temp1 += temp0
		object[35].ixpos = temp1
		temp1 += temp0
		object[36].ixpos = temp1
		temp1 += temp0
		object[37].ixpos = temp1
	else
		object[34].type = TypeName[Blank Object]
		object[35].type = TypeName[Blank Object]
		object[36].type = TypeName[Blank Object]
		object[37].type = TypeName[Blank Object]
	end if
end event

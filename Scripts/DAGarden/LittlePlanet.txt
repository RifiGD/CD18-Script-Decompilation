event ObjectUpdate
	if object[0].state == 2
		if keyDown[0].left == 1
			if object.xpos > -1024
				object.xpos--
			end if
		end if
		if keyDown[0].right == 1
			if object.xpos < 1024
				object.xpos++
			end if
		end if
		if keyDown[0].up == 1
			object.ypos--
		end if
		if keyDown[0].down == 1
			object.ypos++
		end if
		if keyDown[0].start == 1
			if object.value4 == 0
				if object.value3 > -256
					object.value3 -= 2
				end if
			else
				if object.value3 < 256
					object.value3 += 2
				end if
			end if
		end if
		if keyPress[0].start == 1
			object.value4++
			object.value4 &= 1
			object.value3 = 0
		end if
		if keyPress[0].buttonC == 1
			object.value5++
			object.value5 &= 1
		end if
		if keyDown[0].buttonC == 1
			if object.value5 == 0
				if object.value1 > 48
					object.value1 -= 8
				end if
			else
				if object.value1 < 16384
					object.value1 += 8
				end if
			end if
		end if
	end if
	object.angle += object.value3
	if object.angle < 0
		object.angle += 8192
	end if
	object.angle &= 8191
	object.rotation = object.angle
	object.rotation >>= 4
end event

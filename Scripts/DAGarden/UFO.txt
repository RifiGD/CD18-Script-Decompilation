private alias 12 : TYPE_UFO



event ObjectUpdate
	if object.direction == 0
		object.xpos += object.xvel
	else
		object.xpos -= object.xvel
	end if
	switch object.state
	case 0
	case 3
		object.frame = 0
		object.angle += 8
		object.angle &= 511
		Sin(object.ypos, object.angle)
		object.ypos <<= 9
		object.ypos += object.value1
		break
	case 1
		object.ypos += object.yvel
		object.frame = 1
		object.value1 = object.ypos
		if object.ypos > 13631488
			object.state = 2
		end if
		break
	case 2
		object.ypos -= object.yvel
		object.frame = 2
		object.value1 = object.ypos
		if object.ypos < 2097152
			object.state = 1
		end if
		break
	end switch
	if object.value0 > 0
		object.value0--
	else
		object.value1 = object.ypos
		Rand(object.state, 3)
		Rand(object.value0, 90)
		Rand(object.xvel, 6)
		object.xvel++
		object.xvel <<= 14
		Rand(object.yvel, 6)
		object.yvel++
		object.yvel <<= 15
	end if
	if object.outOfBounds == 1
		object.type = TypeName[Blank Object]
	end if
end event


event ObjectDraw
	temp0 = object.frame
	temp0 += object.propertyValue
	DrawSprite(temp0)
end event

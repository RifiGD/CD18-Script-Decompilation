private alias 11 : TYPE_SHOOTINGSTAR



event ObjectUpdate
	switch object.state
	case 0
		object.xvel = -81920
		object.yvel = 131072
		object.state++
		break
	case 1
		object.xpos += object.xvel
		object.ypos += object.yvel
		break
	end switch
	if object.outOfBounds == 1
		object.type = TypeName[Blank Object]
	end if
end event


event ObjectDraw
	temp0 = object.animationSpeed
	temp0 >>= 2
	temp0 += object.propertyValue
	DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
	object.animationSpeed++
	object.animationSpeed %= 24
end event

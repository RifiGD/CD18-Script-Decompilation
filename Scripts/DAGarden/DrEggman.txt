event ObjectUpdate
	if object.direction == 1
		object.xpos += 32768
	else
		object.xpos -= 32768
	end if
	switch object.state
	case 0
		object.value1 = object.ypos
		object.state++
		break
	case 1
		object.angle += object.propertyValue
		object.angle &= 511
		Sin(object.ypos, object.angle)
		object.ypos <<= 11
		object.ypos += object.value1
		object.value1 -= 16384
		break
	end switch
	if object.outOfBounds == 1
		object.type = TypeName[Blank Object]
	end if
end event


event ObjectDraw
	temp0 = object.animationTimer
	temp0 >>= 1
	temp0 += 2
	DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
	object.animationTimer++
	object.animationTimer %= 12
	temp0 = object.animationSpeed
	temp0 >>= 5
	DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
	object.animationSpeed++
	object.animationSpeed &= 63
end event

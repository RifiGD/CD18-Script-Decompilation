private alias 8 : TYPE_BLUEBIRD



event ObjectUpdate
	if object.direction == 0
		object.xpos += 32768
	else
		object.xpos -= 32768
	end if
	switch object.state
	case 0
		object.value1 = object.ypos
		object.state++
		break
	case 1
		object.angle += object.propertyValue
		object.angle &= 511
		Sin(object.ypos, object.angle)
		object.ypos <<= 10
		object.ypos += object.value1
		break
	end switch
	if object.outOfBounds == 1
		object.type = TypeName[Blank Object]
	end if
end event


event ObjectDraw
	temp0 = object.animationSpeed
	temp0 >>= 3
	DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
	object.animationSpeed++
	object.animationSpeed &= 15
end event


event ObjectStartup
	LoadSpriteSheet("DAGarden/Objects.gif")
	SpriteFrame(-8, -9, 16, 16, 466, 577)
	SpriteFrame(-8, -8, 16, 16, 483, 577)
end event

event RSDKDraw
	DrawSprite(0)
end event

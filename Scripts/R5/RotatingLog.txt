event ObjectUpdate
	object.angle++
	object.angle &= 511
	temp1 = 96
	Cos(object.value3, object.angle)
	object.value3 *= temp1
	object.value3 <<= 7
	object.value3 += object.xpos
	object.value3 &= -65536
	object.value3 -= object.value1
	Sin(object.value4, object.angle)
	object.value4 *= temp1
	object.value4 <<= 7
	object.value4 += object.ypos
	object.value4 &= -65536
	object.value4 -= object.value2
	temp0 = object.xpos
	temp1 = object.ypos
	object.xpos = object.value1
	object.ypos = object.value2
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_SOLID, object.entityPos, -8, -8, 8, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			object[currentPlayer].xpos += object.value3
			object[currentPlayer].ypos += object.value4
		end if
	next
	object.xpos = temp0
	object.ypos = temp1
end event


event ObjectDraw
	temp1 = 96
	Cos(object.value1, object.angle)
	object.value1 *= temp1
	object.value1 <<= 7
	object.value1 += object.xpos
	object.value1 &= -65536
	Sin(object.value2, object.angle)
	object.value2 *= temp1
	object.value2 <<= 7
	object.value2 += object.ypos
	object.value2 &= -65536
	DrawSpriteXY(0, object.value1, object.value2)
end event

	case TIME_PRESENT
		SpriteFrame(-8, -8, 16, 16, 35, 1)

		break
	case TIME_GOOD_FUTURE

		break
	case TIME_BAD_FUTURE
		SpriteFrame(-8, -8, 16, 16, 83, 158)
		break
	end switch
	foreach (TypeName[Rotating Log], arrayPos0, ALL_ENTITIES)
		if object[arrayPos0].propertyValue == 1
			arrayPos1 = arrayPos0
			object[arrayPos1].angle = 504
			arrayPos1++
			object[arrayPos1].angle = 8
			arrayPos1++
			object[arrayPos1].angle = 120
			arrayPos1++
			object[arrayPos1].angle = 136
			arrayPos1++
			object[arrayPos1].angle = 248
			arrayPos1++
			object[arrayPos1].angle = 264
			arrayPos1++
			object[arrayPos1].angle = 376
			arrayPos1++
			object[arrayPos1].angle = 392
		end if
	next
	SetTableValue(TypeName[Rotating Log], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(RotatingLog_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(RotatingLog_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)

end event

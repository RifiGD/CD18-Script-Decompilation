event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			switch object.propertyValue
			case 0
				if object[currentPlayer].speed >= 0
					if object[currentPlayer].state != Player_State_TubeRoll
						PlaySfx(SfxName[Charge], 0)
					end if
					object[currentPlayer].direction = 0
					object[currentPlayer].state = Player_State_TubeRoll
					object[currentPlayer].animation = ANI_JUMPING
					object[currentPlayer].value17 = 131072
					if object[currentPlayer].speed < 131072
						object[currentPlayer].speed = 131072
					end if
				else
					if object[currentPlayer].speed > -131072
						object[currentPlayer].speed = -131072
					end if
					object[currentPlayer].state = Player_State_Roll
					object[currentPlayer].animation = ANI_JUMPING
				end if
				break
			case 1
				if object[currentPlayer].speed <= 0
					if object[currentPlayer].state != Player_State_TubeRoll
						PlaySfx(SfxName[Charge], 0)
					end if
					object[currentPlayer].state = Player_State_TubeRoll
					object[currentPlayer].animation = ANI_JUMPING
					object[currentPlayer].value17 = 524288
					if object[currentPlayer].speed > -131072
						object[currentPlayer].speed = -131072
					end if
				else
					if object[currentPlayer].speed < 131072
						object[currentPlayer].speed = 131072
					end if
					object[currentPlayer].state = Player_State_Roll
					object[currentPlayer].animation = ANI_JUMPING
				end if
				break
			case 2
				object[currentPlayer].gravity = 1
				object[currentPlayer].collisionMode = 0
				object[currentPlayer].xvel = 0
				object[currentPlayer].speed = 0
				break
			case 3
				if R5Setup_value14 == 0
					if object[currentPlayer].ypos < object.ypos
						if object[currentPlayer].state != Player_State_Static
							PlaySfx(SfxName[Release], 0)
						end if
						object[currentPlayer].state = Player_State_Static
						if object[currentPlayer].animation != ANI_JUMPING
							object[currentPlayer].animation = ANI_JUMPING
						end if
						object[currentPlayer].value5 = 240
						object[currentPlayer].speed = 0
						object[currentPlayer].value1 = 0
						object[currentPlayer].xpos = object.xpos
						object[currentPlayer].xvel = 0
						object[currentPlayer].yvel = 655360
					end if
				else
					if object[currentPlayer].state == Player_State_Static
						object[currentPlayer].state = Player_State_Air
					end if
				end if
				break
			case 4
				if R5Setup_value14 == 3
					if object[currentPlayer].ypos > object.ypos
						if warp.destination > 0
							if warp.timer == 0
								warp.timer = 1
								ResetObjectEntity(3, TypeName[Warp Star], 0, object[currentPlayer].xpos, object[currentPlayer].ypos)
								object[3].value0 = 7
								object[3].drawOrder = 4
							end if
						end if
						if object[currentPlayer].state != Player_State_Static
							PlaySfx(SfxName[Release], 0)
						end if
						object[currentPlayer].state = Player_State_Static
						if object[currentPlayer].animation != ANI_JUMPING
							object[currentPlayer].animation = ANI_JUMPING
						end if
						object[currentPlayer].value5 = 240
						object[currentPlayer].speed = 0
						object[currentPlayer].value1 = 0
						object[currentPlayer].xpos = object.xpos
						object[currentPlayer].xvel = 0
						object[currentPlayer].yvel = -524288
					end if
				else
					if object[currentPlayer].state == Player_State_Static
						object[currentPlayer].state = Player_State_Air
					end if
				end if
				break
			case 5
				if object[currentPlayer].yvel < 0
					object[currentPlayer].gravity = 0
					object[currentPlayer].state = Player_State_TubeRoll
					object[currentPlayer].animation = ANI_JUMPING
					object[currentPlayer].collisionMode = 3
					object[currentPlayer].angle = 64
					object[currentPlayer].speed = -655360
					object[currentPlayer].value17 = 655360
				end if
				break
			case 6
				if object[currentPlayer].yvel > 0
					object[currentPlayer].gravity = 0
					object[currentPlayer].state = Player_State_TubeRoll
					object[currentPlayer].animation = ANI_JUMPING
					object[currentPlayer].collisionMode = 3
					object[currentPlayer].angle = 64
					object[currentPlayer].speed = object[currentPlayer].yvel
					object[currentPlayer].value17 = 655360
				end if
				break
			case 7
				object[currentPlayer].gravity = 0
				object[currentPlayer].state = Player_State_TubeRoll
				object[currentPlayer].animation = ANI_JUMPING
				if object[currentPlayer].collisionMode != 3
					object[currentPlayer].collisionMode = 3
					object[currentPlayer].angle = 64
				end if
				if R5Setup_value14 == 3
					object[currentPlayer].speed = -851968
					object[currentPlayer].value17 = 851968
				else
					object[currentPlayer].speed = 655360
					object[currentPlayer].value17 = 655360
				end if
				break
			case 8
				if R5Setup_value14 == 3
					object[currentPlayer].gravity = 0
					object[currentPlayer].state = Player_State_TubeRoll
					object[currentPlayer].animation = ANI_JUMPING
					object[currentPlayer].collisionMode = 3
					object[currentPlayer].angle = 64
					object[currentPlayer].speed = -851968
					object[currentPlayer].value17 = 851968
					if object.value0 == 0
						PlaySfx(SfxName[Release], 0)
					end if
				end if
				break
			end switch
			object.value0 = 1
		else
			object.value0 = 0
		end if
	next
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

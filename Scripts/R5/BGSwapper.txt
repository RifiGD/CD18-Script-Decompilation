private alias 41 : TYPE_BGSWAPPER



event ObjectUpdate
	if object.propertyValue == 0
		if object[0].xpos > object.xpos
			stage[0].activeLayer = 2
		else
			stage[0].activeLayer = 1
		end if
	else
		if object[0].xpos > object.xpos
			stage[0].activeLayer = 1
		else
			stage[0].activeLayer = 2
		end if
	end if
end event


event ObjectStartup
	temp0 = 16384
	temp1 = 0
	foreach (TypeName[BG Swapper], arrayPos0, ALL_ENTITIES)
		if object[arrayPos0].ixpos < temp0
			temp0 = object[arrayPos0].ixpos
		end if
		if object[arrayPos0].ixpos > temp1
			temp1 = object[arrayPos0].ixpos
		end if
		object[arrayPos0].priority = 3
	next
	temp2 = object[0].xpos
	temp2 >>= 16
	if temp2 > temp0
		if temp2 > temp1
			stage[0].activeLayer = 1
		else
			stage[0].activeLayer = 2
		end if
	else
		stage[0].activeLayer = 1
	end if
end event

event RSDKDraw
	DrawSprite(0)
end event

end event

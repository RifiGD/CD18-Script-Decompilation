// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 1 : TIME_PAST

private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	switch object.state
	case 0
		if object[0].xpos > object.xpos
			object[+3].value2 = stage.curXBoundary2
			stage.newXBoundary1 = object.ixpos
			stage.newXBoundary1 += 328
			stage.newXBoundary1 -= screen.xcenter
			stage.newXBoundary2 = stage.newXBoundary1
			stage.newXBoundary2 += screen.xcenter
			object[+3].value1 = stage.newXBoundary2
			object[+3].value1 += 160
			object[+3].value1 <<= 16
			stage.newXBoundary2 += screen.xcenter
			stage.newYBoundary2 = object.iypos
			object.state = 1
			object.priority = 1
			R5Setup_value13 = 0
		end if
		break
	case 1
		temp0 = object.xpos
		temp0 += 3145728
		if object[0].xpos > temp0
			object.state = 2
		end if
		break
	case 2
		if object.value0 < 64
			object.value0 += 2
			object.ypos -= 131072
		else
			object.value0 = 0
			object.state = 3
		end if
		break
	case 3
		if object.value0 < 192
			object.value0 += 2
			object.xpos += 131072
		else
			object.value0 = 0
			object.state = 4
			object.priority = 0
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state < 3
			BoxCollisionTest(C_SOLID, object.entityPos, -24, -32, 0, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		else
			BoxCollisionTest(C_TOUCH, object.entityPos, -64, -32, 0, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object[currentPlayer].xpos = object[currentPlayer].collisionRight
				object[currentPlayer].xpos <<= 16
				object[currentPlayer].xpos += object.xpos
			end if
		end if
		BoxCollisionTest(C_TOUCH, object.entityPos, 0, -32, 24, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			CallFunction(Player_Hit)
			if object[currentPlayer].state == Player_State_GotHit
				object[+6].value3 = 49152
			end if
		end if
	next
end event


event ObjectDraw
	DrawSprite(0)
end event

	end if
end event

	object.angle &= 511

		object.value2 = object.angle
		object.value2 *= -16384
		object.value2 += object.xpos
		object.value2 &= -65536
		object.value2 -= object.value0
	else
		object.value2 = 0
	end if
	Cos(object.value3, object.angle)
	object.value3 *= 12288
	object.value3 += object.ypos
	object.value3 &= -65536
	object.value3 -= object.value1
	temp0 = object.xpos
	temp1 = object.ypos
	object.xpos = object.value0
	object.ypos = object.value1

		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -16, 32, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

		end if

		if checkResult == 1
			object[currentPlayer].xpos += object.value2
			object[currentPlayer].ypos += object.value3
		end if
	next
	object.xpos = temp0
	object.ypos = temp1
end event

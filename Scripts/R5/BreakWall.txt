private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE


// Function declarations
reserve function BreakableWall_DebugDraw
reserve function BreakableWall_DebugSpawn


public function BreakableWall_DebugDraw
	DrawSprite(0)
end function


public function BreakableWall_DebugSpawn
	CreateTempObject(TypeName[Breakable Wall], 0, object.xpos, object.ypos)
end function


event ObjectUpdate
	if object.propertyValue > 0
		object.xpos += object.xvel
		object.ypos += object.yvel
		object.yvel += 16384
		if object.outOfBounds == 1
			object.type = TypeName[Blank Object]
		end if
	else
		if object.state == 1
			BoxCollisionTest(C_TOUCH, object.entityPos, -384, -256, 384, 256, 0, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == 0
				object.state = 0
				if object.priority != 4
					object.priority = 0
				end if
			end if
		else
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				if object[currentPlayer].animation == ANI_JUMPING
					BoxCollisionTest(C_TOUCH, object.entityPos, -18, -48, 18, 48, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
					if checkResult == 1
						object.state = 1
						if object.priority != 4
							object.priority = 1
						end if
						PlaySfx(SfxName[Explosion], 0)
						temp1 = object.xpos
						temp1 -= 524288
						temp2 = object.xpos
						temp2 += 524288
						temp3 = object.ypos
						temp3 -= 1572864
						CreateTempObject(TypeName[Breakable Wall], 1, temp1, temp3)
						object[tempObjectPos].xvel = -65536
						object[tempObjectPos].yvel = -131072
						object[tempObjectPos].drawOrder = 5
						CreateTempObject(TypeName[Breakable Wall], 2, temp2, temp3)
						object[tempObjectPos].xvel = 65536
						object[tempObjectPos].yvel = -131072
						object[tempObjectPos].drawOrder = 5
						temp3 += 1048576
						CreateTempObject(TypeName[Breakable Wall], 3, temp1, temp3)
						object[tempObjectPos].xvel = -131072
						object[tempObjectPos].yvel = -65536
						object[tempObjectPos].drawOrder = 5
						CreateTempObject(TypeName[Breakable Wall], 4, temp2, temp3)
						object[tempObjectPos].xvel = 131072
						object[tempObjectPos].yvel = -65536
						object[tempObjectPos].drawOrder = 5
						temp3 += 1048576
						CreateTempObject(TypeName[Breakable Wall], 5, temp1, temp3)
						object[tempObjectPos].xvel = -262144
						object[tempObjectPos].yvel = 0
						object[tempObjectPos].drawOrder = 5
						CreateTempObject(TypeName[Breakable Wall], 6, temp2, temp3)
						object[tempObjectPos].xvel = 262144
						object[tempObjectPos].yvel = 0
						object[tempObjectPos].drawOrder = 5
						temp3 += 1048576
						CreateTempObject(TypeName[Breakable Wall], 7, temp1, temp3)
						object[tempObjectPos].xvel = -131072
						object[tempObjectPos].yvel = 65536
						object[tempObjectPos].drawOrder = 5
						CreateTempObject(TypeName[Breakable Wall], 8, temp2, temp3)
						object[tempObjectPos].xvel = 131072
						object[tempObjectPos].yvel = 65536
						object[tempObjectPos].drawOrder = 5
					end if
				else
					BoxCollisionTest(C_SOLID, object.entityPos, -16, -48, 16, 48, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				end if
			next
		end if
	end if
end event


event ObjectDraw
	if object.state == 0
		DrawSprite(object.propertyValue)
	end if
end event

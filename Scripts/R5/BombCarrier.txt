		temp0 += 2097152
		if object[0].xpos > temp0
			CreateTempObject(TypeName[Fade Music], 0, 0, 0)
			object.state = 1
		end if
		break
	case 2
		if object[+3].state < 6
			if object.value0 < 180
				object.value0++
			else
				object.state = 4
				PlaySfx(SfxName[Bomb Carrier], 0)
				if object[+4].state < 8
					object[+3].state = 1
				else
					object[+3].state = 5
					object[+3].value0 = 0
					object[+3].frame = 0
				end if
			end if

			ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
		end if
		break
	case 3
		if object.xpos < object[+1].xpos
			object.xpos += 122880
		else
			object.state = 2
			object.xpos = object[+1].xpos
			object.value0 = 0
			if object[+4].state > 8
				if object[+4].state > 9
					ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
				end if
				if object[+4].value5 > 640
					ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
				end if
			end if
		end if
		break
	case 4
		if object.xpos > object[0].xpos
			object.xpos -= 49152
			if object.xpos <= object[0].xpos
				object.state = 5
				object.xpos = object[0].xpos
				if object[+4].state < 8
					object[+3].state = 0
				end if
				object.value0 = 0
			end if
		else
			object.state = 5
			object.xpos = object[0].xpos
			object[+3].state = 0
			object.value0 = 0
		end if
		temp0 = object[+1].xpos

			object.value1 = 384
		else
			object.value1 += 8
			object.value1 &= 511
		end if
		object[-1].xpos = object.xpos
		Sin(object[-1].ypos, object.value1)
		object[-1].ypos <<= 9
		object[-1].ypos += object.ypos
		object[-1].ypos += 1835008
		break
	case 5

			object.value0++
		else
			object.state = 6

	case 6

			object.value0++
		else

		end if
		break
	end switch

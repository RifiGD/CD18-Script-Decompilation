	case 0

		break
	case 1

		break
	case 2

		temp0 >>= 3
		temp0 += 2
		DrawSprite(temp0)
		object.frame++

		end if

		end if

		object.value0 &= 7

			object.value0 = 0
			object.frame = 0
			object.state = 4
		end if
		break
	case 4
		DrawSprite(0)

			object.value0++
		else
			object.value0 = 0
			object.state = 3
		end if

	case 5
		DrawSprite(1)

			object.value0++
		else
			object.value0 = 0
			object.frame = 0
			object.state = 3
		end if
		break
	case 6
		temp0 = object.frame
		temp0 /= 6
		temp0 += 6
		DrawSprite(temp0)
		object.frame++
		object.frame %= 24
		temp0 = object.value0

		end if

	end switch
end event

// Function declarations
reserve function DropRock_DebugDraw
reserve function DropRock_DebugSpawn


public function DropRock_DebugDraw
	DrawSprite(0)
end function


public function DropRock_DebugSpawn
	CreateTempObject(TypeName[Drop Rock], 0, object.xpos, object.ypos)
	object[tempObjectPos].value0 = object[tempObjectPos].ypos
end function


event ObjectUpdate
	switch object.state
	case 1
		if object.yvel < 524288
			object.yvel += 16384
		end if
		object.ypos += object.yvel
		if object.yvel > 131072
			ObjectTileCollision(0, 0, 12, 0)
			if checkResult == true
				PlaySfx(SfxName[Explosion], 0)
				object.state = 2
				CreateTempObject(TypeName[Drop Rock], 1, object.xpos, object.ypos)
				object[tempObjectPos].state = 3
				object[tempObjectPos].xpos -= 524288
				object[tempObjectPos].ypos -= 524288
				object[tempObjectPos].xvel = -65536
				object[tempObjectPos].yvel = -262144
				object[tempObjectPos].drawOrder = 5
				CreateTempObject(TypeName[Drop Rock], 2, object.xpos, object.ypos)
				object[tempObjectPos].state = 3
				object[tempObjectPos].xpos += 524288
				object[tempObjectPos].ypos -= 524288
				object[tempObjectPos].xvel = 65536
				object[tempObjectPos].yvel = -262144
				object[tempObjectPos].drawOrder = 5
				CreateTempObject(TypeName[Drop Rock], 3, object.xpos, object.ypos)
				object[tempObjectPos].state = 3
				object[tempObjectPos].xpos -= 524288
				object[tempObjectPos].ypos += 524288
				object[tempObjectPos].xvel = -131072
				object[tempObjectPos].yvel = -131072
				object[tempObjectPos].drawOrder = 5
				CreateTempObject(TypeName[Drop Rock], 4, object.xpos, object.ypos)
				object[tempObjectPos].state = 3
				object[tempObjectPos].xpos += 524288
				object[tempObjectPos].ypos += 524288
				object[tempObjectPos].xvel = 131072
				object[tempObjectPos].yvel = -131072
				object[tempObjectPos].drawOrder = 5
				CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
				object[tempObjectPos].drawOrder = 5
			end if
		end if
		break
	case 2
		if object.outOfBounds == 1
			object.ypos = object.value0
			if object.priority != 4
				object.priority = 0
			end if
			object.state = 0
		end if
		break
	case 3
		object.xpos += object.xvel
		object.yvel += 16384
		object.ypos += object.yvel
		if object.outOfBounds == 1
			object.type = TypeName[Blank Object]
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		switch object.state
		case 0
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			BoxCollisionTest(C_TOUCH, object.entityPos, -4, 0, 4, 96, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.value0 = object.ypos
				if object.priority != 4
					object.priority = 1
				end if
				object.yvel = 0
				object.state = 1
			end if
		case 1
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -16, 12, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == 4
				PlaySfx(SfxName[Explosion], 0)
				object.state = 2
				CallFunction(Player_Hit)
				CreateTempObject(TypeName[Drop Rock], 1, object.xpos, object.ypos)
				object[tempObjectPos].state = 3
				object[tempObjectPos].xpos -= 524288
				object[tempObjectPos].ypos -= 524288
				object[tempObjectPos].xvel = -65536
				object[tempObjectPos].yvel = -262144
				object[tempObjectPos].drawOrder = 5
				CreateTempObject(TypeName[Drop Rock], 2, object.xpos, object.ypos)
				object[tempObjectPos].state = 3
				object[tempObjectPos].xpos += 524288
				object[tempObjectPos].ypos -= 524288
				object[tempObjectPos].xvel = 65536
				object[tempObjectPos].yvel = -262144
				object[tempObjectPos].drawOrder = 5
				CreateTempObject(TypeName[Drop Rock], 3, object.xpos, object.ypos)
				object[tempObjectPos].state = 3
				object[tempObjectPos].xpos -= 524288
				object[tempObjectPos].ypos += 524288
				object[tempObjectPos].xvel = -131072
				object[tempObjectPos].yvel = -196608
				object[tempObjectPos].drawOrder = 5
				CreateTempObject(TypeName[Drop Rock], 4, object.xpos, object.ypos)
				object[tempObjectPos].state = 3
				object[tempObjectPos].xpos += 524288
				object[tempObjectPos].ypos += 524288
				object[tempObjectPos].xvel = 131072
				object[tempObjectPos].yvel = -196608
				object[tempObjectPos].drawOrder = 5
				CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
				object[tempObjectPos].drawOrder = 5
			end if
			break
		end switch
	next
end event


event ObjectDraw
	switch object.state
	case 0
	case 1
		DrawSprite(0)
		break
	case 3
		DrawSprite(object.propertyValue)
		break
	end switch
end event


event ObjectStartup
	LoadSpriteSheet("R5/Objects.gif")
	SpriteFrame(-16, -16, 32, 32, 92, 18)
	SpriteFrame(-8, -8, 16, 16, 92, 18)
	SpriteFrame(-8, -8, 16, 16, 108, 18)
	SpriteFrame(-8, -8, 16, 16, 92, 34)
	SpriteFrame(-8, -8, 16, 16, 108, 34)
	SetTableValue(TypeName[Drop Rock], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(DropRock_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(DropRock_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

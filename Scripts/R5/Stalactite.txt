private alias 67 : TYPE_STALACTITE

// Function declarations
reserve function Stalactite_DebugDraw
reserve function Stalactite_DebugSpawn


public function Stalactite_DebugDraw
	DrawSprite(0)
end function


public function Stalactite_DebugSpawn
	CreateTempObject(TypeName[Stalactite], 0, object.xpos, object.ypos)
	object[tempObjectPos].value1 = object[tempObjectPos].ypos
end function


event ObjectUpdate
	switch object.state
	case 0
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, -4, 0, 4, 96, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == 1
				if object.priority != 4
					object.priority = 1
				end if
				object.yvel = 0
				object.state = 1
			end if
		next
		break
	case 1
		if object.yvel < 524288
			object.yvel += 16384
		end if
		object.ypos += object.yvel
		if object.outOfBounds == 1
			object.state = 2
			object.ypos = object.value1
		end if
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, -8, -24, 8, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == 1
				CallFunction(Player_Hit)
			end if
		next
		break
	case 2
		if object.outOfBounds == 1
			if object.priority != 4
				object.priority = 0
			end if
			object.state = 0
		end if
		break
	end switch
end event


event ObjectDraw
	switch object.state
	case 0
	case 1
		DrawSprite(0)
		break
	end switch
end event


event ObjectStartup
	LoadSpriteSheet("R5/Objects.gif")
	switch object[9].propertyValue
	case 0
		SpriteFrame(-8, -24, 16, 48, 172, 207)
		break
	case 1
		SpriteFrame(-8, -24, 16, 48, 189, 207)
		break
	case 2
		SpriteFrame(-8, -24, 16, 48, 172, 207)
		break
	case 3
		SpriteFrame(-8, -24, 16, 48, 155, 207)
		break
	end switch
	foreach (TypeName[Stalactite], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].value1 = object[arrayPos0].ypos
	next
	SetTableValue(TypeName[Stalactite], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(Stalactite_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(Stalactite_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

event RSDKDraw
	DrawSprite(0)
end event

event ObjectUpdate
	if object.propertyValue == 0
		switch object.state
		case 0
			Sin(object.ypos, object.value0)
			object.ypos <<= 11
			object.ypos += object.value1
			object.xpos -= 65536
			object.value0 -= 8
			if object.value0 == 0
				object.state = 1
				object.direction = 1
			end if
			break
		case 1
			Sin(object.ypos, object.value0)
			FlipSign(object.ypos)
			object.ypos <<= 11
			object.ypos += object.value1
			object.xpos += 65536
			object.value0 += 8
			if object.value0 == 2048
				object.state = 0
				object.direction = 0
			end if
			break
		end switch
	else
		switch object.state
		case 0
			Sin(object.ypos, object.value0)
			object.ypos <<= 11
			object.ypos += object.value1
			object.xpos -= 32768
			object.value0 -= 2
			if object.value0 == 0
				object.state = 1
				object.direction = 1
			end if
			break
		case 1
			Sin(object.ypos, object.value0)
			FlipSign(object.ypos)
			object.ypos <<= 11
			object.ypos += object.value1
			object.xpos += 32768
			object.value0 += 2
			if object.value0 == 1024
				object.state = 0
				object.direction = 0
			end if
			break
		end switch
	end if
	object.animationTimer++
	object.animationTimer %= 6
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -20, -10, 20, 10, currentPlayer, object[currentPlayer].value41, object[currentPlayer].value39, object[currentPlayer].value42, object[currentPlayer].value40)
		if checkResult == true
			CallFunction(Player_BadnikBreak)
		end if
	next
	CallFunction(StageSetup_CheckGoodFuture)
end event

		object.frame += 2
		DrawSpriteFX(object.frame, FX_FLIP, object.xpos, object.ypos)
	end if
end event

private alias 45 : TYPE_KAMEMUSISHOT



event ObjectUpdate
	object.yvel += 8192
	object.xpos += object.xvel
	object.ypos += object.yvel
	if object.outOfBounds == 1
		object.type = TypeName[Blank Object]
	else
		temp0 = screen.yoffset
		temp0 -= 8
		temp0 <<= 16
		if object.ypos < temp0
			object.type = TypeName[Blank Object]
		end if
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -4, -4, 4, 4, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			CallFunction(Player_ProjectileHit)
		end if
	next
	object.animationTimer++
	if object.animationTimer > 2
		object.animationTimer = 0
		object.frame++
		object.frame &= 3
	end if
end event


event ObjectDraw
	DrawSprite(object.frame)
end event


event ObjectStartup
	LoadSpriteSheet("R1/Objects.gif")
	SpriteFrame(-4, -4, 8, 8, 166, 34)
	SpriteFrame(-4, -4, 8, 8, 175, 34)
	SpriteFrame(-6, -6, 12, 12, 101, 93)
	SpriteFrame(-6, -6, 12, 12, 114, 93)
end event

event RSDKDraw
	DrawSprite(0)
end event

private alias 64 : TYPE_KUMOKUMOWEB



event ObjectUpdate
	if object.state == 0
		object.xpos += object.xvel
		if object.outOfBounds == 1
			object.type = TypeName[Blank Object]
		else
			temp0 = screen.yoffset
			temp0 -= 8
			temp0 <<= 16
			if object.ypos < temp0
				object.type = TypeName[Blank Object]
			end if
		end if
	else
		if object.value0 < 120
			object.value0++
			currentPlayer = object.value1
			if object[currentPlayer].speed > 16384
				object[currentPlayer].speed = 16384
			end if
			if object[currentPlayer].speed < -16384
				object[currentPlayer].speed = -16384
			end if
		else
			object.type = TypeName[Blank Object]
		end if
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -6, -6, 6, 6, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			object.state = 1
			object.value1 = currentPlayer
		end if
	next
end event


event ObjectDraw
	if object.state == 1
		currentPlayer = object.value1
		object.xpos = object[currentPlayer].xpos
		object.ypos = object[currentPlayer].ypos
	end if
	temp0 = object.frame
	temp0 /= 3
	DrawSprite(temp0)
	object.frame++
	object.frame %= 6
end event

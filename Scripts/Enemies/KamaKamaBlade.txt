	object.ypos += object.yvel
	object.frame++
	object.frame %= 24
	switch object.state
	case 0
		arrayPos0 = object.value1
		if object[arrayPos0].type != TypeName[Kama Kama]
			object.type = TypeName[Blank Object]
		end if
		if object.propertyValue > 0
			object.propertyValue--
		else
			object.state = 1
			PlaySfx(SfxName[Blades], 0)
			if object.direction == 0
				object.xvel = -196608
			else
				object.xvel = 196608
			end if
		end if
		break
	case 2
		object.yvel += 12288
		break
	end switch
	if object.outOfBounds == 1
		object.type = TypeName[Blank Object]
	else
		temp0 = screen.yoffset
		temp0 -= 8
		temp0 <<= 16

			object.type = TypeName[Blank Object]
		end if
	end if

		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, -8, -6, 8, 6, currentPlayer, object[currentPlayer].value41, object[currentPlayer].value39, object[currentPlayer].value42, object[currentPlayer].value40)

				CheckEqual(object[currentPlayer].animation, ANI_JUMPING)
				temp0 = checkResult

				temp0 |= checkResult
				if temp0 == 1
					object.state = 2
					object.yvel = -524288
					if object.direction == 0
						if object[currentPlayer].xvel > 32768
							FlipSign(object.xvel)
						end if
					else
						if object[currentPlayer].xvel < -32768
							FlipSign(object.xvel)
						end if
					end if
				else
					CallFunction(Player_Hit)
				end if
			end if
		next
	end if
end event

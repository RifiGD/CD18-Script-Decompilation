reserve function TailsObject_ProcessTailSprite


public function TailsObject_ProcessTailSprite
	switch object.value12
	case 0
		if object.frame == 0
			object.value11++
			object.value11 %= 40
		end if
		break
	case 1
	case 3
	case 4
		object.value11++
		object.value11 %= 40
		break
	case 7
	case 9
	case 10
	case 43
		object.value11++
		object.value11 %= 12
		break
	case 17
		object.value11++
		object.value11 %= 30
		break
	end switch
end function

			object.rotation = 0

			object.visible = 1

				object[+playerCount].propertyValue = 0
			end if

			if object.gravity == 0
				object.value34 = 0
			end if
			CallFunction(object.state)
			ProcessAnimation()

			if object.animation == ANI_JUMPING
				screen[0].adjustCameraY = object.value29
			else
				if screen[0].adjustCameraY == object.value29
					screen[0].adjustCameraY = 0
					object.iypos += object.value29
				end if
			end if
			if object.value33 == 0
				object.value43 = object.gravity
				object.value43 ^= 1
				ProcessObjectMovement()
				checkResult |= object.value43
				object.value43 ^= 1
				if checkResult == true
					object.value18 = 0
					if object.animation == ANI_JUMPING

							if object.state != Player_State_Roll
								if object.state != Player_State_TubeRoll

event ObjectDraw
	if object.animation != object.prevAnimation
		object.prevAnimation = object.animation
		object.frame = 0
		object.animationTimer = 0
		object.animationSpeed = 0
	end if
	if object.value12 != object.animation
		if object.animation > 4
			object.value11 = 0
		end if
		if object.value12 > 4
			object.value11 = 0
		end if
		object.value12 = object.animation
	end if
	
	// this should fix what is known as "Four-Tailed Freak"
#platform: USE_DECOMP
	if Mini_PlayerFlag == false
#endplatform
		switch object.value12
		case 0
			if object.frame == 0
				temp0 = object.value11
				temp0 >>= 3
				if object.visible == 1
					DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
				end if
			end if
			break
		case 1
		case 3
		case 4
		case 37
			temp0 = object.value11
			temp0 >>= 3
			if object.visible == 1
				DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
			end if
			break
		case 7
		case 9
			temp0 = object.value11
			temp0 >>= 2
			temp0 += 11
			if object.visible == 1
				DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
			end if
			break
		case 10
		case 43
			temp0 = object.value11
			temp0 >>= 2
			CheckEqual(object.xvel, 0)
			temp1 = checkResult
			CheckEqual(object.yvel, 0)
			temp1 &= checkResult
			temp2 = object.rotation
			if temp1 == 0
				ATan2(object.rotation, object.xvel, object.yvel)
				object.rotation += 16
				object.rotation &= 255
				object.rotation >>= 5
				switch object.rotation
				case 0
				case 8
					temp0 += 5
					object.rotation = 0
					break
				case 1
					temp0 += 8
					if object.direction == 0
						object.rotation = 64
					else
						object.rotation = 0
					end if
					break
				case 2
					temp0 += 5
					object.rotation = 64
					break
				case 3
					temp0 += 8
					if object.direction == 0
						object.rotation = 128
					else
						object.rotation = 64
					end if
					break
				case 4
					temp0 += 5
					object.rotation = 128
					break
				case 5
					temp0 += 8
					if object.direction == 0
						object.rotation = 192
					else
						object.rotation = 128
					end if
					break
				case 6
					temp0 += 5
					object.rotation = 192
					break
				case 7
					temp0 += 8
					if object.direction == 0
						object.rotation = 0
					else
						object.rotation = 192
					end if
					break
				end switch
				if object.direction == 1
					object.rotation += 128
				end if
			else
				temp0 += 5
				object.rotation = 0
			end if
			object.rotation <<= 1
			if object.visible == 1
				DrawSpriteFX(temp0, FX_ROTATE, object.xpos, object.ypos)
			end if
			object.rotation = temp2
			break
		case 17
			temp0 = object.value11
			temp0 /= 10
			temp0 += 11
			if object.visible == 1
				DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
			end if
			break
		end switch
#platform: USE_DECOMP
	end if
#endplatform
	DrawObjectAnimation()
end event

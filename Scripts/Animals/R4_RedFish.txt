event ObjectUpdate
	switch object.state
	case 0
		if object.angle < 512
			object.angle += 2
			object.xpos += 65536
		else
			object.state = 1
			object.direction = 1
		end if
		break
	case 1
		if object.angle > 0
			object.angle -= 2
			object.xpos -= 65536
		else
			object.state = 0
			object.direction = 0
		end if
		break
	end switch
	object.value0++
	if object.value0 > 19
		object.value0 = 0
		object.frame++
		object.frame &= 1
	end if
end event


event ObjectDraw
	if metalSonic.destroyed == 1
		Sin(temp1, object.angle)
		temp1 <<= 12
		temp1 += object.ypos
		DrawSpriteFX(object.frame, FX_FLIP, object.xpos, temp1)
	end if
	if hud.currentTimePeriod == TIME_GOOD_FUTURE
		Sin(temp1, object.angle)
		temp1 <<= 12
		temp1 += object.ypos
		DrawSpriteFX(object.frame, FX_FLIP, object.xpos, temp1)
	end if
end event

end event

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST

private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	object.ypos += object.yvel
	object.yvel += 11264
	switch object.state
	case 0
		object.xpos += 65536
		if object.yvel > 262144
			ObjectTileCollision(0, 0, 8, 0)
			if checkResult == 1
				object.yvel = -327680
				object.value0++
				if object.value0 == 2
					object.value0 = 0
					object.state = 1
					object.direction = 1
				end if
			end if
		end if
		break
	case 1
		object.xpos -= 65536
		if object.yvel > 262144
			ObjectTileCollision(0, 0, 8, 0)
			if checkResult == 1
				object.yvel = -327680
				object.value0++
				if object.value0 == 2
					object.value0 = 0
					object.state = 0
					object.direction = 0
				end if
			end if
		end if
		break
	end switch
end event


event ObjectDraw
	if metalSonic.destroyed == 1
		if object.yvel < 0
			DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
		else
			DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
		end if
	else
		if hud.currentTimePeriod == TIME_GOOD_FUTURE
			if object.yvel < 0
				DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
			else
				DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
			end if
		end if
	end if
end event

event ObjectUpdate
	object.ypos += object.yvel
	object.yvel += 10240
	switch object.state
	case 0
		object.xpos += 65536
		if object.yvel > 0
			ObjectTileCollision(0, 0, 8, 0)
			if checkResult == true
				object.direction = 0
				object.yvel = -327680
				object.state = 1
			end if
		end if
		break
	case 1
		object.xpos -= 65536
		if object.yvel > 0
			ObjectTileCollision(0, 0, 8, 0)
			if checkResult == true
				object.direction = 1
				object.yvel = -327680
				object.state = 0
			end if
		end if
		break
	end switch
end event


event ObjectDraw
	if metalSonic.destroyed == 1
		if object.yvel < 0
			DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
		else
			DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
		end if
	end if
	if hud.currentTimePeriod == TIME_GOOD_FUTURE
		if object.yvel < 0
			DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
		else
			DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
		end if
	end if
end event


event ObjectStartup
	LoadSpriteSheet("Global/Items3.gif")
	arrayPos0 = 32
	while arrayPos0 < 1056
		if object[arrayPos0].type == TypeName[Squirrel]
			object[arrayPos0].yvel = -327680
			object[arrayPos0].direction = 1
		end if
		arrayPos0++
	loop
	SpriteFrame(-10, -8, 24, 16, 150, 35)
	SpriteFrame(-10, -8, 24, 16, 150, 52)
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

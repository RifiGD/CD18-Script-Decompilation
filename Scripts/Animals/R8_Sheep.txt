private alias 69 : TYPE_SHEEP

// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	object.ypos += object.yvel
	object.yvel += 10240
	switch object.state
	case 0
		object.xpos += 65536
		if object.yvel > 0
			ObjectTileCollision(0, 0, 8, 0)
			if checkResult == 1
				object.direction = 1
				object.yvel = -327680
				object.state = 1
			end if
		end if
		break
	case 1
		object.xpos -= 65536
		if object.yvel > 0
			ObjectTileCollision(0, 0, 8, 0)
			if checkResult == 1
				object.direction = 0
				object.yvel = -327680
				object.state = 0
			end if
		end if
		break
	end switch
end event


event ObjectDraw
	if metalSonic.destroyed == 1
		if object.yvel < 0
			DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
		else
			DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
		end if
	end if
	if hud.currentTimePeriod == TIME_GOOD_FUTURE
		if object.yvel < 0
			DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
		else
			DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
		end if
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R8/Objects2.gif")
	foreach (TypeName[Sheep], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].yvel = -327680
	next
	SpriteFrame(-10, -8, 24, 16, 160, 110)
	SpriteFrame(-10, -8, 24, 16, 160, 127)
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad

end event

private alias 68 : TYPE_FLYINGBIRD

// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	switch object.state
	case 0
		if object.angle > 0
			object.angle -= 2
		else
			object.state = 1
			object.direction = 1
		end if
		break
	case 1
		if object.angle < 256
			object.angle += 2
		else
			object.state = 0
			object.direction = 0
		end if
		break
	end switch
	object.value0++
	if object.value0 > 19
		object.value0 = 0
		object.frame++
		object.frame &= 1
	end if
end event


event ObjectDraw
	if metalSonic.destroyed == 1
		Cos(temp0, object.angle)
		temp0 <<= 14
		temp0 += object.xpos
		Sin(temp1, object.angle)
		temp1 <<= 14
		temp1 += object.ypos
		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	end if
	if hud.currentTimePeriod == TIME_GOOD_FUTURE
		Cos(temp0, object.angle)
		temp0 <<= 14
		temp0 += object.xpos
		Sin(temp1, object.angle)
		temp1 <<= 14
		temp1 += object.ypos
		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R8/Objects2.gif")
	SpriteFrame(-8, -8, 16, 16, 143, 110)
	SpriteFrame(-8, -8, 16, 16, 143, 127)
	foreach (TypeName[Flying Bird], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].drawOrder = 5
	next
end event

event RSDKDraw

end event

event ObjectUpdate
	switch object.state
	case 0
		if object.value1 > 0
			object.value1 -= 2
		else
			object.state = 1
			object.direction = 1
		end if
		break
	case 1
		if object.value1 < 256
			object.value1 += 2
		else
			object.state = 0
			object.direction = 0
		end if
		break
	end switch
	object.value0++
	if object.value0 > 19
		object.value0 = 0
		object.frame++
		object.frame &= 1
	end if
end event


event ObjectDraw
	if metalSonic.destroyed == 1
		Cos(temp0, object.value1)
		temp0 <<= 14
		temp0 += object.xpos
		Sin(temp1, object.value1)
		temp1 <<= 14
		temp1 += object.ypos
		DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
	else
		if hud.currentTimePeriod == TIME_GOOD_FUTURE
			Cos(temp0, object.value1)
			temp0 <<= 14
			temp0 += object.xpos
			Sin(temp1, object.value1)
			temp1 <<= 14
			temp1 += object.ypos
			DrawSpriteFX(object.frame, FX_FLIP, temp0, temp1)
		end if
	end if
end event

event ObjectUpdate
	switch object.state
	case 0
		if object.value0 < 60
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 1
		if object.value0 < 74
			object.value0++
			object.value3++
			object.xpos += object.xvel
			object.ypos += object.yvel
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 2
		if object.value0 < 60
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 3
		if object.value0 < 74
			object.value0++
			object.value3--
			object.xpos -= object.xvel
			object.ypos -= object.yvel
		else
			object.value0 = 0
			object.state = 0
		end if
		break
	end switch
	if object.value4 > 0
		object.value4--
	else
		PlaySfx(SfxName[Buzzsaw], 0)
		object.value4 += 48
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -24, -24, 24, 24, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			CallFunction(Player_Hit)
		end if
	next
end event


event ObjectDraw
	temp0 = object.frame
	temp0 >>= 1
	DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
	object.frame++
	object.frame &= 3
	temp0 = object.value3
	temp0 /= 20
	switch temp0
	case 0
		DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
		break
	case 1
		DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
		DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
		break
	case 2
		DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
		DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
		DrawSpriteFX(4, FX_FLIP, object.xpos, object.ypos)
		break
	case 3
		DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
		DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
		DrawSpriteFX(4, FX_FLIP, object.xpos, object.ypos)
		DrawSpriteFX(5, FX_FLIP, object.xpos, object.ypos)
		break
	end switch
end event

	SpriteFrame(-113, -113, 26, 26, 158, 43)
	foreach (TypeName[Buzzsaw], arrayPos0, ALL_ENTITIES)
		if object[arrayPos0].propertyValue > 127
			object[arrayPos0].propertyValue -= 128
			object[arrayPos0].direction = object[arrayPos0].propertyValue
			object[arrayPos0].value3 = 74
			object[arrayPos0].state = 2
		else
			object[arrayPos0].direction = object[arrayPos0].propertyValue
		end if
		switch object[arrayPos0].direction
		case 0
			object[arrayPos0].xvel = 65536
			object[arrayPos0].yvel = 65536
			break
		case 1
			object[arrayPos0].xvel = -65536
			object[arrayPos0].yvel = 65536
			break
		case 2
			object[arrayPos0].xvel = 65536
			object[arrayPos0].yvel = -65536
			break
		case 3
			object[arrayPos0].xvel = -65536
			object[arrayPos0].yvel = -65536
			break
		end switch

end event

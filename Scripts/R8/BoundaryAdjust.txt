private alias 57 : TYPE_BOUNDARYADJUST



event ObjectUpdate
	if object.state == 0
		if object[0].xpos > object.xpos
			temp0 = 3120
			temp0 -= screen.xcenter
			stage.newXBoundary1 = temp0
			temp0 += screen.xsize
			stage.newXBoundary2 = temp0
			temp0 = object.iypos
			temp0 -= screen.ysize
			stage.newYBoundary1 = temp0
			stage.curYBoundary2 = screen.yoffset
			stage.curYBoundary2 += screen.ysize
			if stage.curYBoundary2 < object.iypos
				stage.curYBoundary2 = object.iypos
			end if
			object.state++
		end if
	else
		stage.curYBoundary2 -= 3
		if stage.curYBoundary2 <= object.iypos
			stage.curYBoundary2 = object.iypos
			object.type = TypeName[Blank Object]
		end if
	end if
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

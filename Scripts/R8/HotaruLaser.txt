	case 0
		object.ypos += 524288
		ObjectTileCollision(0, 0, 8, 0)
		if checkResult == true
			object.value1 = object.ypos
			object.state++
		end if
		break
	case 1
		if object.value0 < 2
			object.ypos += 524288
			object.value0++
		else
			object.type = TypeName[Blank Object]
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		arrayPos0 = object.value2
		if object[arrayPos0].type == TypeName[Hotaru]
			BoxCollisionTest(C_TOUCH, object.entityPos, -4, -8, 4, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				CallFunction(Player_Hit)
			end if
		end if
	next
end event

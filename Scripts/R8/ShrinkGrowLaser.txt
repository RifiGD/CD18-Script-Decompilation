			object.ypos += 262144

	case 1

			temp0 = 2
			temp0 += object.propertyValue
			DrawSprite(temp0)
		else
			DrawSprite(4)
		end if
		object.animationTimer++
		object.animationTimer &= 3
		break
	case 2
		if object.animationTimer < 2
			temp0 = 2
			temp0 += object.propertyValue
			DrawSprite(temp0)
			temp0 += 3

		end if

		object.animationTimer &= 3
		break
	end switch
end event

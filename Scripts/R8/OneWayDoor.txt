event ObjectUpdate
	if object.state == 0
		if object.value1 > 0
			object.value1 -= 131072
		end if
	else
		if object.value1 < 2097152
			object.value1 += 131072
		end if
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp0 = object.ypos
		temp1 = object[currentPlayer].ypos
		object.ypos -= object.value1
		BoxCollisionTest(C_SOLID, object.entityPos, -8, -48, 8, -16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult > 1
			object[currentPlayer].ypos = temp1
		end if
		object.ypos = temp0
		object.ypos += object.value1
		BoxCollisionTest(C_SOLID, object.entityPos, -8, -16, 8, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult > 1
			object[currentPlayer].ypos = temp1
		end if
		object.ypos = temp0
		if object.propertyValue == 2
			BoxCollisionTest(C_TOUCH, object.entityPos, -96, -48, 8, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		else
			BoxCollisionTest(C_TOUCH, object.entityPos, -8, -48, 96, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		end if
		object.state = checkResult
	next
end event


event ObjectDraw
	object.frame = object.animationTimer
	object.frame /= 3
	switch object.propertyValue
	case 1
		temp0 = object.value1
		FlipSign(temp0)
		temp0 += object.ypos
		DrawSpriteXY(0, object.xpos, temp0)
		object.frame += 6
		DrawSpriteXY(object.frame, object.xpos, temp0)
		temp0 = object.value1
		temp0 += object.ypos
		DrawSpriteXY(1, object.xpos, temp0)
		object.frame += 2
		DrawSpriteXY(object.frame, object.xpos, temp0)
		break
	case 2
		temp0 = object.value1
		FlipSign(temp0)
		temp0 += object.ypos
		DrawSpriteXY(0, object.xpos, temp0)
		object.frame += 2
		DrawSpriteXY(object.frame, object.xpos, temp0)
		temp0 = object.value1
		temp0 += object.ypos
		DrawSpriteXY(1, object.xpos, temp0)
		object.frame += 2
		DrawSpriteXY(object.frame, object.xpos, temp0)

		temp0 = object.value1

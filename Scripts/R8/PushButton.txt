// ========================

private alias object.propertyValue : object.stood

// Player Aliases
private alias object.ypos    : player.ypos
private alias object.yvel    : player.yvel
private alias object.gravity : player.gravity
private alias object.collisionBottom : player.collisionBottom

// ========================
// Events
// ========================

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.stood == 0
			if player[currentPlayer].yvel >= 0
				BoxCollisionTest(C_SOLID, object.entityPos, -16, -8, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object.stood = true
					object[currentPlayer].ypos += 0x80000
					PlaySfx(SfxName[Switch], 0)
				else
					object.stood = false
				end if
			else
				object.stood = false
			end if
		else
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, -1, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
#platform: USE_DECOMP
				if player[currentPlayer].yvel > -1
#endplatform
					object.stood = true
					player[currentPlayer].gravity = GRAVITY_GROUND
					player[currentPlayer].yvel = 0
					player[currentPlayer].ypos = player[currentPlayer].collisionBottom
					FlipSign(player[currentPlayer].ypos)
					player[currentPlayer].ypos <<= 16
					player[currentPlayer].ypos += object.ypos
#platform: USE_DECOMP
				end if
#endplatform
			else
				object.stood = false

				if player[currentPlayer].yvel > -1
					player[currentPlayer].gravity = 0
					player[currentPlayer].yvel = 0
					player[currentPlayer].ypos = player[currentPlayer].collisionBottom
					FlipSign(player[currentPlayer].ypos)
					player[currentPlayer].ypos -= 8
					player[currentPlayer].ypos <<= 16
					player[currentPlayer].ypos += object.ypos
				end if
			end if
		end if
	next
end event


event ObjectDraw
	DrawSprite(object.stood)
end event


event ObjectStartup
	LoadSpriteSheet("R8/Objects.gif")
	SpriteFrame(-16, -8, 32, 16, 142, 98)
	SpriteFrame(-16, 0, 32, 8, 142, 114)
end event

// ========================
// Editor Events

event RSDKDraw

end event

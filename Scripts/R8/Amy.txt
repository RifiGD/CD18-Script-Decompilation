event ObjectUpdate
	switch object.state
	case 0
		object.frame = object.value0
		object.frame >>= 2
		object.value0++
		object.value0 %= 24
		object.xpos += 81920
		ObjectTileGrip(0, 0, 19, 0)
		temp0 = object[-1].xpos
		temp0 -= 6291456
		if object.xpos > temp0
			object.frame = 6
			object.xpos = temp0
			object.yvel = -196608
			object.state++
		end if
		break
	case 1
		object.yvel += 16384
		object.ypos += object.yvel
		ObjectTileCollision(0, 0, 19, 0)
		if object.yvel > 0
			object.frame = 7
		end if
		if checkResult == true
			object.frame = 6
			object.yvel = -196608
			object.state++
		end if
		break
	case 2
		object.yvel += 16384
		object.ypos += object.yvel
		ObjectTileCollision(0, 0, 19, 0)
		if object.yvel > 0
			object.frame = 7
		end if
		if checkResult == true
			object.yvel = 0
			object.state++
		end if
		break
	case 3
		object.frame = object.value0
		object.frame >>= 2
		object.value0++
		object.value0 %= 24
		object.xpos += 81920
		ObjectTileGrip(0, 0, 19, 0)
		temp0 = object[-1].xpos
		temp0 -= 1048576
		if object.xpos > temp0
			object[+1].type = TypeName[Fade Screen]
			object.state++
		end if
		break
	case 4
		object.frame = object.value0
		object.frame >>= 2
		object.value0++
		object.value0 %= 24
		object.xpos += 81920
		ObjectTileGrip(0, 0, 19, 0)
		break
	end switch
end event

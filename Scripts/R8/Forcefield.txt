event ObjectUpdate
	switch object.state
	case 0
		if object.propertyValue > 168
			object.propertyValue--
		else
			object.state++
		end if
		object.frame = object.animationTimer
		object.frame >>= 2
		object.animationTimer++
		object.animationTimer &= 15
		temp1 = oscillation
		temp1 <<= 3
		Sin(temp0, temp1)
		temp0 >>= 4
		temp0 += 190
		object.alpha = temp0
		object.inkEffect = 2
		object.drawOrder = 4
		break
	case 1
		object.frame = object.animationTimer
		object.frame >>= 2
		object.animationTimer++
		object.animationTimer &= 15
		temp1 = oscillation
		temp1 <<= 3
		Sin(temp0, temp1)
		temp0 >>= 4
		temp0 += 190
		object.alpha = temp0
		break
	case 2
		object.frame = 0
		if object.alpha > 0
			object.alpha -= 2
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.alpha > 0
			temp0 = object.propertyValue
			temp0 -= 8
			BoxCollisionTest(C_SOLID, object.entityPos, temp0, -96, 240, 96, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			FlipSign(temp0)
			BoxCollisionTest(C_SOLID, object.entityPos, -240, -96, temp0, 96, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		end if
	next
end event

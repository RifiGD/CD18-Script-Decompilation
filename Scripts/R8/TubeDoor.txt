private alias 58 : TYPE_TUBEDOOR



event ObjectUpdate
	if object.outOfBounds == 1
		object.state = 0
		object.value1 = 0
		object.priority = 0
	end if
	switch object.state
	case 0
		object.priority = 1
		if object[-1].propertyValue == 1
			if object.propertyValue == 0
				object.state = 1
			end if
		else
			if object[0].state == 1
				if object[0].animation == ANI_JUMPING
					object.state = 3
					object.value1 = 1048576
				end if
			end if
		end if
		break
	case 1
		if object.value1 < 1048576
			object.value1 += 131072
		end if
		if object[0].state == 1
			if object[0].animation == ANI_JUMPING
				if object[0].ypos > object.ypos
					object.state = 3
				end if
			end if
		end if
		break
	case 2
		if object.value1 > 0
			object.value1 -= 131072
		else
			object.state = 0
		end if
		break
	case 3
		temp0 = object.ypos
		temp0 -= object[0].ypos
		if temp0 > 1048576
			object.state = 2
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state == 0
			BoxCollisionTest(C_PLATFORM, object.entityPos, -8, -4, 56, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		else
			BoxCollisionTest(C_PLATFORM, object.entityPos, -8, -4, 4, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			BoxCollisionTest(C_PLATFORM, object.entityPos, 44, -4, 56, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		end if
	next
end event


event ObjectDraw
	temp0 = object.value1
	FlipSign(temp0)
	temp0 += object.xpos
	DrawSpriteXY(2, temp0, object.ypos)
	temp0 = object.value1
	temp0 += object.xpos
	DrawSpriteXY(3, temp0, object.ypos)
	DrawSprite(0)
	DrawSprite(1)
end event

private alias 3 : DEFORM_BG_WATER

// ========================
// Function Declarations

reserve function R4Setup_deformLeft
reserve function R4Setup_deformRight

// ========================
// Static Values
// ========================

public value Water_flashTimer = 0;

// ========================
// Function Definitions
// ========================

private function R4Setup_deformLeft
	stage[arrayPos0].deformationData1 = 1
	arrayPos0++
	stage[arrayPos0].deformationData1 = 1
	arrayPos0++
	stage[arrayPos0].deformationData1 = 2
	arrayPos0++
	stage[arrayPos0].deformationData1 = 2
	arrayPos0++
	stage[arrayPos0].deformationData1 = 3
	arrayPos0++
	stage[arrayPos0].deformationData1 = 3
	arrayPos0++
	stage[arrayPos0].deformationData1 = 3
	arrayPos0++
	stage[arrayPos0].deformationData1 = 3
	arrayPos0++
	stage[arrayPos0].deformationData1 = 2
	arrayPos0++
	stage[arrayPos0].deformationData1 = 2
	arrayPos0++
	stage[arrayPos0].deformationData1 = 1
	arrayPos0++
	stage[arrayPos0].deformationData1 = 1
	arrayPos0++
end function


private function R4Setup_deformRight
	stage[arrayPos0].deformationData1 = -1
	arrayPos0++
	stage[arrayPos0].deformationData1 = -1
	arrayPos0++
	stage[arrayPos0].deformationData1 = -2
	arrayPos0++
	stage[arrayPos0].deformationData1 = -2
	arrayPos0++
	stage[arrayPos0].deformationData1 = -3
	arrayPos0++
	stage[arrayPos0].deformationData1 = -3
	arrayPos0++
	stage[arrayPos0].deformationData1 = -3
	arrayPos0++
	stage[arrayPos0].deformationData1 = -3
	arrayPos0++
	stage[arrayPos0].deformationData1 = -2
	arrayPos0++
	stage[arrayPos0].deformationData1 = -2
	arrayPos0++
	stage[arrayPos0].deformationData1 = -1
	arrayPos0++
	stage[arrayPos0].deformationData1 = -1
	arrayPos0++
end function

// ========================
// Events
// ========================

event ObjectUpdate
	object.deformTimer++
	if object.deformTimer > 1
		tileLayer[0].deformationOffsetW++
		tileLayer[1].deformationOffsetW++
		object.deformTimer = 0
	end if

	// Don't know why it checks for this...
	// It always true...
	if object.palCycleEnabled == true
		object.cycleTimer++
		if object.cycleTimer == 3
			object.cycleTimer = 0

			CopyPalette(3, object.cycleIndex, 0, 176, 16)
			object.cycleIndex += 16
			object.cycleIndex %= object.cycleEndIndex
		end if
	end if

	if Water_flashTimer > 0
		Water_flashTimer--
	end if
end event


event ObjectDraw
	temp0 = stage.waterLevel
	temp0 -= screen.yoffset
	if temp0 < 0
		temp0 = 0
	end if

	if temp0 > screen.ysize
		temp0 = screen.ysize
	end if

	SetActivePalette(0, 0, temp0)
	if Water_flashTimer > 0
		SetActivePalette(2, temp0, screen.ysize)
	else
		SetActivePalette(1, temp0, screen.ysize)
	end if

	player.waterStream = 0
end event


event ObjectStartup
	object[SLOT_ZONESETUP].type = TypeName[R4 Setup]
	object[SLOT_ZONESETUP].priority = PRIORITY_ACTIVE
	object[SLOT_ZONESETUP].drawOrder = 0

	arrayPos0 = 0
	CallFunction(R4Setup_deformLeft)

	arrayPos0 = 128
	CallFunction(R4Setup_deformLeft)

	arrayPos0 = 160
	CallFunction(R4Setup_deformRight)

	arrayPos0 = 256
	while arrayPos0 < 576
		arrayPos1 = arrayPos0
		arrayPos1 -= 256
		stage[arrayPos0].deformationData1 = stage[arrayPos1].deformationData1
		arrayPos0++
	loop

	SetLayerDeformation(DEFORM_BG_WATER, 128, 4, 0, 0, 0)

	switch hud.currentTimePeriod
	case TIME_PRESENT
		LoadPalette("R4A_WaterPal.act", 1, 0, 0, 256)
		if stage.actNum == 2
			LoadPalette("R4A_PalCycle.act", 3, 0, 0, 64)
			object[SLOT_ZONESETUP].palCycleEnabled = true
			object[SLOT_ZONESETUP].cycleEndIndex = 64
		end if
		break

	case TIME_PAST
		LoadPalette("R4B_WaterPal.act", 1, 0, 0, 256)
		if stage.actNum == 2
			LoadPalette("R4B_PalCycle.act", 3, 0, 0, 48)
			object[SLOT_ZONESETUP].palCycleEnabled = true
			object[SLOT_ZONESETUP].cycleEndIndex = 48
		end if
		break

	case TIME_GOOD_FUTURE
		LoadPalette("R4C_WaterPal.act", 1, 0, 0, 256)
		LoadPalette("R4C_PalCycle.act", 3, 0, 0, 64)
		object[SLOT_ZONESETUP].palCycleEnabled = true
		object[SLOT_ZONESETUP].cycleEndIndex = 64
		break

	case TIME_BAD_FUTURE
		LoadPalette("R4D_WaterPal.act", 1, 0, 0, 256)
		LoadPalette("R4D_PalCycle.act", 3, 0, 0, 48)
		object[21].palCycleEnabled = true
		object[21].cycleEndIndex = 48
		break
	end switch

	LoadPalette("ElectricFlash.act", 2, 0, 0, 256)
	Water_flashTimer = 0
end event

event ObjectUpdate
	switch object.propertyValue
	case 0
		if object[0].xpos < 41943040
			stage[0].activeLayer = 2
			object.type = TypeName[Blank Object]
			object[19].type = TypeName[BG Animation]
		end if
		break

	case 1
		if object[0].ypos > object.ypos
			stage[0].activeLayer = 1
			object[19].type = TypeName[BG Animation]
		else
			stage[0].activeLayer = 2
			object[19].type = TypeName[BG Animation]
		end if
		break

	case 2
		if object[0].xpos > object.xpos
			stage.newWaterLevel = 0x264 0000	// Y - 612 in map coords
		else
			stage.newWaterLevel = 0x284 0000	// Y - 644 in map coords
		end if
		break

	case 4
		BoxCollisionTest(C_TOUCH, object.entityPos, -10, -16, 10, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					warp.timer = 0
					object[0].state = Player_State_WaterCurrent
					object[0].animation = ANI_FANROTATE
					object[0].xvel = 0
					object[0].yvel = -262144
					if object[0].xpos < object.xpos
						object[0].xpos += 65536
					else
						object[0].xpos -= 65536
					end if
				end if
			end if
		else
			if object[0].state == Player_State_WaterCurrent
				BoxCollisionTest(C_TOUCH, object.entityPos, -64, -16, -32, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object[0].state = Player_State_Air
					object[0].animation = ANI_WALKING
				end if
				BoxCollisionTest(C_TOUCH, object.entityPos, 32, -16, 64, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object[0].state = Player_State_Air
					object[0].animation = ANI_WALKING
				end if
			end if
		end if
		break

	case 5
		BoxCollisionTest(C_TOUCH, object.entityPos, -10, -16, 10, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					warp.timer = 0
					object[0].state = Player_State_WaterCurrent
					object[0].animation = ANI_FANROTATE
					object[0].xvel = 0
					object[0].yvel = 262144
					if object[0].xpos < object.xpos
						object[0].xpos += 65536
					else
						object[0].xpos -= 65536
					end if
				end if
			end if
		else
			if object[0].state == Player_State_WaterCurrent
				BoxCollisionTest(C_TOUCH, object.entityPos, -64, -16, -32, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object[0].state = Player_State_Ground
					object[0].animation = ANI_WALKING
				end if
				BoxCollisionTest(C_TOUCH, object.entityPos, 32, -16, 64, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object[0].state = Player_State_Ground
					object[0].animation = ANI_WALKING
				end if
			end if
		end if
		break

	case 6
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					object[0].xvel = -262144
					object[0].animation = ANI_FANROTATE
				end if
			end if
		end if
		break

	case 7
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					object[0].xvel = 262144
					object[0].animation = ANI_FANROTATE
				end if
			end if
		end if
		break

	case 8
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					object[0].yvel = -262144
					object[0].animation = ANI_FANROTATE
				end if
			end if
		end if
		break

	case 9
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					object[0].yvel = 262144
					object[0].animation = ANI_FANROTATE
				end if
			end if
		end if
		break

	case 10
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					object[0].xvel = 0
				end if
			end if
		end if
		break

	case 11
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					object[0].yvel = 0
				end if
			end if
		end if
		break

	case 12
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[0].state != Player_State_Death
				if object[0].state != Player_State_Drown
					object[0].yvel = 262144
					object[0].state = Player_State_Air
					object[0].animation = ANI_WALKING
				end if
			end if
		end if
		break

	case 13
		if object[0].xpos > object.xpos
			stage.newWaterLevel = 0x780 0000	// Y - 1920 in map coords
		else
			stage.newWaterLevel = 0x534 0000	// Y - 1332 in map coords
		end if
		break

	end switch
end event

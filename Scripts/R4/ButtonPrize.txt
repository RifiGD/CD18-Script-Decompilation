				object[+3].type = TypeName[Blank Object]
				object[+5].type = TypeName[Blank Object]
				object[+7].type = TypeName[Blank Object]
				object[+9].type = TypeName[Blank Object]
				CreateTempObject(TypeName[Explosion], 0, object[-1].xpos, object[-1].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+1].xpos, object[+1].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+3].xpos, object[+3].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+5].xpos, object[+5].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+7].xpos, object[+7].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+9].xpos, object[+9].ypos)
				break
			case BUTTONPRIZE_SCORE
				object[-3].type = TypeName[Object Score]
				object[-3].propertyValue = 3
				object[-1].type = TypeName[Blank Object]

				object[+3].type = TypeName[Object Score]

				object[+5].type = TypeName[Object Score]

				object[+7].propertyValue = 3
				CreateTempObject(TypeName[Explosion], 0, object[-3].xpos, object[-3].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[-1].xpos, object[-1].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+1].xpos, object[+1].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+3].xpos, object[+3].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+5].xpos, object[+5].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+7].xpos, object[+7].ypos)
				player.score += 5000
				break
			case BUTTONPRIZE_BUBBLES
				object[-5].type = TypeName[Button Prize]
				object[-3].type = TypeName[Button Prize]
				object[-1].type = TypeName[Button Prize]
				object[+1].type = TypeName[Button Prize]
				object[+3].type = TypeName[Button Prize]
				object[+5].type = TypeName[Button Prize]
				object[-5].state = 1
				object[-3].state = 1

				object[+3].state = 1
				object[+5].state = 1
				CreateTempObject(TypeName[Explosion], 0, object[-5].xpos, object[-5].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[-3].xpos, object[-3].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[-1].xpos, object[-1].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+1].xpos, object[+1].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+3].xpos, object[+3].ypos)
				CreateTempObject(TypeName[Explosion], 0, object[+5].xpos, object[+5].ypos)
				break
			case BUTTONPRIZE_BUBBLES_2
				object[-7].type = TypeName[Button Prize]
				object[-5].type = TypeName[Button Prize]
				object[-3].type = TypeName[Button Prize]
				object[-1].type = TypeName[Button Prize]

	case BUTTONPRIZE_CREATE_BUBBLES
		// Crash Details (take your time reading this one):
		// As you can see, it spawns the Air Bubble, then writes some of it's values.
		// HOWEVER, CD18 uses S1/S2's Air Bubble object rather than CD11's. The value/alias layout is different.
		// The crash itself comes from the fact that it writes value2 to the object's current xpos,
		// back then, value 2 was the air bubble's X Velocity.
		// If you head over to Air Bubble, you'll see that the first thing it does in ObjectUpdate is set
		// the currentPlayer array to this value2, which is now a REALLY FUCKING LARGE value,
		// so now, EVERY reference to this array is some BIGASS VALUE, which is far OUTSIDE the max entity count for a single scene.
		// Now the game can rest in value2's piss.

		// I decided to go ahead and do a rewrite of this logic under the USE_DECOMP flag, to match what the v3 version did
		// The crash still exists under USE_ORIGINAL_CODE, I wouldn't like to see it become lost media lol
		object.value0++
		if object.value0 == 60
#platform: USE_ORIGINAL_CODE
			CreateTempObject(TypeName[Air Bubble], 5, object.xpos, object.ypos)
			object[tempObjectPos].value2 = object[tempObjectPos].xpos
			Rand(object[tempObjectPos].value1, 256)
			object[tempObjectPos].value1 <<= 1
			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].value3 = -0x8800
#endplatform

#platform: USE_DECOMP
			CreateTempObject(TypeName[Air Bubble], 6, object.xpos, object.ypos)
			object[tempObjectPos].value1 = object[tempObjectPos].xpos
			Rand(object[tempObjectPos].angle, 256)
			object[tempObjectPos].angle <<= 1
			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].yvel = -0x8800
#endplatform
		end if
		if object.value0 == 120
#platform: USE_ORIGINAL_CODE
			CreateTempObject(TypeName[Air Bubble], 5, object.xpos, object.ypos)
			object[tempObjectPos].value2 = object[tempObjectPos].xpos
			Rand(object[tempObjectPos].value1, 256)
			object[tempObjectPos].value1 <<= 1
			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].value3 = -0x8800
#endplatform

#platform: USE_DECOMP
			CreateTempObject(TypeName[Air Bubble], 6, object.xpos, object.ypos)
			object[tempObjectPos].value1 = object[tempObjectPos].xpos
			Rand(object[tempObjectPos].angle, 256)
			object[tempObjectPos].angle <<= 1
			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].yvel = -0x8800
#endplatform
			object.type = TypeName[Blank Object]
		end if
		break
	case BUTTONPRIZE_CREATE_RINGS
		object.value0++
		if object.value0 == 360
			object.type = TypeName[Blank Object]
			object.priority = 0
		end if
		temp0 = object.value0
		temp0 &= 15
		if temp0 == 0
			CreateTempObject(TypeName[Lose Ring], 0, object.xpos, object.ypos)
			object[tempObjectPos].ypos -= 12058624
			object[tempObjectPos].animationSpeed = 256
			temp1 = object.value0
			temp1 >>= 4
			temp1 %= 10
			switch temp1
			case 0
				object[tempObjectPos].xpos -= 10485760
				break
			case 1
			case 9
				object[tempObjectPos].xpos -= 8388608
				break
			case 2
			case 8
				object[tempObjectPos].xpos -= 6291456
				break
			case 3
			case 7
				object[tempObjectPos].xpos -= 4194304
				break
			case 4
			case 6
				object[tempObjectPos].xpos -= 2097152
				break
			case 5
				break
			end switch
		end if
		break
	end switch
end event

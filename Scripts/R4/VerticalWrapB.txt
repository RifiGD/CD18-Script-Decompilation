private alias 50 : TYPE_VERTICALWRAP



event ObjectUpdate
	if object[0].yvel >= 0
		if screen[0].cameraY >= 2168
			temp1 = object[0].ypos
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				temp0 = object[currentPlayer].ypos
				temp0 -= temp1
				Abs(temp0)
				if temp0 < 9175040
					object[currentPlayer].ypos -= 134217728
				end if
			next
			screen[0].cameraY -= 2048
			screen.yoffset -= 2048
		end if
	end if
	if object[0].yvel <= 0
		if screen[0].cameraY <= 120
			temp1 = object[0].ypos
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				temp0 = object[currentPlayer].ypos
				temp0 -= temp1
				Abs(temp0)
				if temp0 < 9175040
					object[currentPlayer].ypos += 134217728
				end if
			next
			screen[0].cameraY += 2048
			screen.yoffset += 2048
		end if
	end if
end event

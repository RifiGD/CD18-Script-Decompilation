event ObjectUpdate
	object.angle = oscillation
	object.angle += object.propertyValue
	object.angle <<= 1
	Cos(object.value2, object.angle)
	object.value2 <<= 12
	object.value2 += object.xpos
	object.value2 &= -65536
	object.value2 -= object.value1
	object.value3 = 0
	temp0 = object.xpos
	object.xpos = object.value1
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_SOLID, object.entityPos, -64, -16, 64, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		switch checkResult
		case 1
			object[currentPlayer].xpos += object.value2
			break
		case 2
			if object[-1].value3 == 1
				CallFunction(Player_Kill)
			end if
			break
		case 3
			object.value3 = 1
			break
		end switch
	next
	object.xpos = temp0
end event


event ObjectDraw
	Cos(object.value1, object.angle)
	object.value1 <<= 12
	object.value1 += object.xpos
	object.value1 &= -65536
	DrawSpriteXY(0, object.value1, object.ypos)
	DrawSpriteXY(1, object.value1, object.ypos)
	DrawSpriteXY(2, object.value1, object.ypos)
	DrawSpriteXY(3, object.value1, object.ypos)
end event

		LoadSpriteSheet("R4/Objects.gif")
		temp0 = 163

		break

		LoadSpriteSheet("R4/Objects2.gif")
		temp0 = 1

		break

	case TIME_BAD_FUTURE

		break
	end switch

event ObjectUpdate
	object.angle += object.value5
	if object.angle < 0
		object.angle += 512
	end if
	object.angle &= 511
	temp1 = 96
	Cos(object.value3, object.angle)
	object.value3 *= temp1
	object.value3 <<= 7
	object.value3 += object.xpos
	object.value3 &= -65536
	object.value3 -= object.value1
	Sin(object.value4, object.angle)
	object.value4 *= temp1
	object.value4 <<= 7
	object.value4 += object.ypos
	object.value4 &= -65536
	object.value4 -= object.value2
	if object.frame == 1
		temp0 = object.xpos
		temp1 = object.ypos
		object.xpos = object.value1
		object.ypos = object.value2
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -8, 32, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == 1
				object[currentPlayer].xpos += object.value3
				object[currentPlayer].ypos += object.value4
			end if
		next
		object.xpos = temp0
		object.ypos = temp1
	end if
end event

event ObjectUpdate
	if object[-1].propertyValue == 1
		object.propertyValue = 1
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, object.value1, -32, object.value2, 32, currentPlayer, 0, 0, 0, 0)
			if checkResult == true
				if object[currentPlayer].state != Player_State_Hurt
					if object[currentPlayer].state != Player_State_Death
						if object[currentPlayer].state != Player_State_Drown
							if object[currentPlayer].state != Player_State_Static
								object[currentPlayer].state = Player_State_WaterCurrent
								object[currentPlayer].animation = ANI_FANROTATE
								if object[currentPlayer].up == false
									if object[currentPlayer].down == false
										if object[currentPlayer].ypos < object.ypos
											object[currentPlayer].ypos += 65536
										end if
										if object[currentPlayer].ypos > object.ypos
											object[currentPlayer].ypos -= 65536
										end if
									end if
								end if
								object[currentPlayer].xvel = -262144
								object[currentPlayer].speed = -262144
								if player.waterStream == 0
									object[currentPlayer].yvel = 0
									player.waterStream = 1
								end if
							end if
						end if
					end if
				end if
			else
				if object[currentPlayer].state == Player_State_WaterCurrent
					if player.waterStream == 0
						object[currentPlayer].state = Player_State_Air
						object[currentPlayer].controlMode = currentPlayer
						object[currentPlayer].animation = ANI_WALKING
					end if
				end if
			end if
		next
	end if
end event

	CreateTempObject(TypeName[RPlatform], 0, object.xpos, object.ypos)

event ObjectUpdate
	object.angle += 2
	object.angle &= 511
	temp1 = 96
	Cos(object.value3, object.angle)
	object.value3 *= temp1
	object.value3 <<= 7
	object.value3 += object.xpos
	object.value3 &= -65536
	object.value3 -= object.value1
	Sin(object.value4, object.angle)
	object.value4 *= temp1
	object.value4 <<= 7
	object.value4 += object.ypos
	object.value4 &= -65536
	object.value4 -= object.value2
	temp0 = object.xpos
	temp1 = object.ypos
	object.xpos = object.value1
	object.ypos = object.value2
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -8, 32, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[currentPlayer].xpos += object.value3
			object[currentPlayer].ypos += object.value4
		end if
		if object.propertyValue == 0
			BoxCollisionTest(C_SOLID, object.entityPos, 0, -24, 32, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				if object.propertyValue == 0
					PlaySfx(SfxName[Switch], 0)
				end if
				object.propertyValue = 1
				object[currentPlayer].ypos += 524288
				object[currentPlayer].xpos += object.value3
				object[currentPlayer].ypos += object.value4
			else
				object.propertyValue = 0
			end if
		else
			BoxCollisionTest(C_SOLID, object.entityPos, 0, -16, 32, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.propertyValue = 1
				object[currentPlayer].xpos += object.value3
				object[currentPlayer].ypos += object.value4
			else
				object.propertyValue = 0
			end if
		end if
	next
	object.xpos = temp0
	object.ypos = temp1
end event


event ObjectDraw
	DrawSprite(2)
	temp0 = 0
	temp1 = 16
	while temp0 < 5
		Cos(temp2, object.angle)
		temp2 *= temp1
		temp2 <<= 7
		temp2 += object.xpos
		Sin(temp3, object.angle)
		temp3 *= temp1
		temp3 <<= 7
		temp3 += object.ypos
		DrawSpriteXY(3, temp2, temp3)
		temp0++
		temp1 += 16
	loop
	Cos(object.value1, object.angle)
	object.value1 *= temp1
	object.value1 <<= 7
	object.value1 += object.xpos
	object.value1 &= -65536
	Sin(object.value2, object.angle)
	object.value2 *= temp1
	object.value2 <<= 7
	object.value2 += object.ypos
	object.value2 &= -65536
	DrawSpriteXY(4, object.value1, object.value2)
	DrawSpriteXY(object.propertyValue, object.value1, object.value2)
end event

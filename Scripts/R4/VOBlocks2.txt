event ObjectUpdate
	object.angle = oscillation
	object.angle += object.propertyValue
	object.angle <<= 1
	Sin(object.value2, object.angle)
	object.value2 <<= 13
	object.value2 += object.ypos
	object.value2 &= -65536
	object.value2 -= object.value1
	temp0 = object.ypos
	object.ypos = object.value1
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_SOLID, object.entityPos, -16, -64, 16, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[currentPlayer].ypos += object.value2
		end if
	next
	object.ypos = temp0
end event


event ObjectDraw
	Sin(object.value1, object.angle)
	object.value1 <<= 13
	object.value1 += object.ypos
	object.value1 &= -65536
	DrawSpriteXY(0, object.xpos, object.value1)
	DrawSpriteXY(1, object.xpos, object.value1)
	DrawSpriteXY(2, object.xpos, object.value1)
	DrawSpriteXY(3, object.xpos, object.value1)
end event

// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	switch object.state
	case 0
		object.value1 = object.ypos
		if object[-1].propertyValue == 1
			object.state++
		end if
		if object[+1].propertyValue == 1
			object.state++
		end if
		break
	case 1
		if object.value0 < 128
			object.value0++
			object.ypos += 65536
			object.value1 -= 65536
		else
			object.state++
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_SOLID, object.entityPos, -32, -65, 0, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[currentPlayer].ypos += 65536
		end if
		BoxCollisionTest(C_SOLID, object.entityPos, 32, -65, 64, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[currentPlayer].ypos += 65536
		end if
		temp0 = object.ypos
		object.ypos = object.value1
		BoxCollisionTest(C_SOLID, object.entityPos, -64, -64, -32, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		BoxCollisionTest(C_SOLID, object.entityPos, 0, -64, 32, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		object.ypos = temp0
	next
end event

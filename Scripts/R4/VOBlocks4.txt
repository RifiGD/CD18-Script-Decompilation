event ObjectUpdate
	object.angle = oscillation
	object.angle += object.propertyValue
	object.angle <<= 1
	Sin(object.value1, object.angle)
	object.value1 <<= 13
	object.value1 += object.ypos
	object.value1 &= -65536
	object.value1 -= object.value0
	temp0 = object.ypos
	object.ypos = object.value0
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_SOLID, object.entityPos, -16, -64, 16, 62, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 4
			if object[currentPlayer].gravity == 1
				CallFunction(Player_Hit)
			else
				CallFunction(Player_Kill)
			end if
		end if
	next
	object.ypos = temp0
end event


event ObjectDraw
	Sin(object.value0, object.angle)
	object.value0 <<= 13
	object.value0 += object.ypos
	object.value0 &= -65536
	object.spriteSheet = object.value2
	DrawSpriteXY(0, object.xpos, object.value0)
	DrawSpriteXY(1, object.xpos, object.value0)
	DrawSpriteXY(2, object.xpos, object.value0)
	object.spriteSheet = object.value3
	DrawSpriteXY(3, object.xpos, object.value0)
end event

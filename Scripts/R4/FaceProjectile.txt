private alias 70 : TYPE_FACEPROJECTILE



event ObjectUpdate
	object.xpos += object.xvel
	if object.outOfBounds == 1
		object.type = TypeName[Blank Object]
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object[currentPlayer].yvel > -917504
			BoxCollisionTest(C_TOUCH, object.entityPos, -13, -8, 13, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == 1
				if object.direction == 0
					if object[currentPlayer].left == 1
						if object[currentPlayer].speed > -196608
							object[currentPlayer].speed = -196608
						end if
					else
						if object[currentPlayer].right == 1
							object[currentPlayer].speed = -131072
						else
							object[currentPlayer].speed = -196608
						end if
					end if
				else
					if object[currentPlayer].right == 1
						if object[currentPlayer].speed < 196608
							object[currentPlayer].speed = 196608
						end if
					else
						if object[currentPlayer].left == 1
							object[currentPlayer].speed = 131072
						else
							object[currentPlayer].speed = 196608
						end if
					end if
				end if
			end if
		end if
	next
	object.animationTimer++
	object.animationTimer %= 20
end event


event ObjectDraw
	if object.animationTimer > 9
		object.direction += 2
		DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
		object.direction -= 2
	else
		DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R4/Objects2.gif")
	SpriteFrame(-16, -8, 32, 16, 152, 159)
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

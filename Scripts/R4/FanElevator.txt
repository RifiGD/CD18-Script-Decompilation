// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	if object.state == 0
		if object.yvel < 65536
			object.yvel += 8192
		end if
		if object.ypos == object.value1
			temp0 = object[0].ypos
			temp0 -= object.ypos
			if temp0 < 10485760
				object.yvel = 0
			end if
		else
			temp0 = object.ypos
			temp0 -= object.value1
			if temp0 > 44040192
				object.ypos = object.value1
			end if
		end if
	end if
	object.ypos += object.yvel
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -28, 32, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object.yvel = -131072
			if object.state == 0
				PlaySfx(SfxName[Switch], 0)
			end if
			object.state = 1
		else
			if object.yvel < 0
				object.yvel = 0
			end if
			object.state = 0
		end if
	next
end event


event ObjectDraw
	if object.state == 0
		DrawSprite(0)
		DrawSprite(1)
		DrawSpriteFX(0, FX_FLIP, object.xpos, object.ypos)
		DrawSpriteFX(1, FX_FLIP, object.xpos, object.ypos)
	else
		DrawSprite(2)
		DrawSprite(3)
		DrawSpriteFX(2, FX_FLIP, object.xpos, object.ypos)
		DrawSpriteFX(3, FX_FLIP, object.xpos, object.ypos)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R4/Objects3.gif")
	if hud.currentTimePeriod < TIME_BAD_FUTURE
		SpriteFrame(-32, -32, 32, 48, 84, 46)
		SpriteFrame(-32, 16, 32, 16, 110, 144)
		SpriteFrame(-32, -32, 32, 48, 84, 95)
		SpriteFrame(-32, 16, 32, 16, 110, 161)
	else
		SpriteFrame(-32, -32, 32, 48, 84, 46)
		SpriteFrame(-32, 16, 32, 16, 126, 228)
		SpriteFrame(-32, -32, 32, 48, 84, 95)
		SpriteFrame(-32, 16, 32, 16, 158, 228)
	end if
	foreach (TypeName[Fan Elevator], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].direction = 1
		object[arrayPos0].priority = 3
		object[arrayPos0].value1 = object[arrayPos0].ypos
	next
end event

event RSDKDraw
	DrawSprite(0)
end event

	switch object.state
	case 0
		if object.value0 < 60
			object.frame = 1
		else
			temp0 = object.value0
			temp0 -= 60
			temp0 >>= 1
			switch temp0
			case 0
				object.frame = 2
				break
			case 1
			case 2
				object.frame = 1
				break
			case 3
				object.frame = 2
				break
			case 4
			case 5
				object.frame = 1
				break
			case 6
				object.frame = 2
				break
			end switch
		end if
		object.value0++
		if object.value0 == 74
			object.value0 = 0
		end if
		break
	case 1
		if object.value0 < 120
			object.value0++
			temp0 = object.value0
			temp0 &= 3
			if temp0 == 3
				Rand(temp0, 64)
				temp0 -= 32
				temp0 <<= 16
				temp0 += object.xpos
				Rand(temp1, 40)
				temp1 -= 40
				temp1 <<= 16
				temp1 += object.ypos
				CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
				PlaySfx(SfxName[Explosion], 0)
			end if
		else
			object.state = 2
			object.value0 = 256
			object.inkEffect = 2
			object.alpha = 255
			temp0 = object.xpos
			temp0 -= 196608
			temp1 = object.ypos
			temp1 -= 1572864
			CreateTempObject(TypeName[Pod Seed], 1, temp0, temp1)
			object[tempObjectPos].xvel = -131072
			object[tempObjectPos].yvel = -393216
			object[tempObjectPos].value1 = object.entityPos
			temp0 += 65536
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = -87380
			object[tempObjectPos].yvel = -393216
			temp0 += 65536
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = -43690
			object[tempObjectPos].yvel = -393216
			temp0 += 65536
			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = 0
			object[tempObjectPos].yvel = -393216
			temp0 += 65536

			CreateTempObject(TypeName[Pod Seed], 0, temp0, temp1)
			object[tempObjectPos].xvel = 131072
			object[tempObjectPos].yvel = -393216
		end if
		break
	case 2
		if object.value0 > 0
			object.value0 -= 16
			object.alpha = object.value0
		end if
		break
	case 3
		if object[0].controlMode == -1
			object[0].left = 0
			object[0].right = 0
			object[0].up = 0
			object[0].down = 0
			object[0].jumpHold = 0
			PlayMusic(1)
			object.state = 4
			object[30].type = TypeName[Act Finish]
			object[30].drawOrder = 6
			object[30].priority = 1
		end if
		break
	case 4
		if object[0].controlMode == -1
			object[0].right = 0
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state == 0
			if object[currentPlayer].animation == ANI_JUMPING
				BoxCollisionTest(C_TOUCH, object.entityPos, -28, -46, 28, 6, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					stage.timeEnabled = 0
					object.state = 1
					object.frame = 3

					FlipSign(object[currentPlayer].yvel)
				end if
			end if
		end if
	next
end event


event ObjectDraw
	if object.state < 2
		DrawSprite(0)
		DrawSprite(object.frame)
	else
		DrawSpriteFX(0, FX_INK, object.xpos, object.ypos)
		DrawSpriteFX(object.frame, FX_INK, object.xpos, object.ypos)
	end if
end event

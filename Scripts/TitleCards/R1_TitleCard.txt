event ObjectDraw
	player[currentPlayer].timer = 0

	switch object.state
	case TitleCard_State_FadeIn
		player[currentPlayer].controlMode = CONTROLMODE_NONE
		player[currentPlayer].up = false
		player[currentPlayer].down = false
		player[currentPlayer].left = false
		player[currentPlayer].right = false
		player[currentPlayer].jumpPress = false
		player[currentPlayer].jumpHold = false
		
		if warp.xpos < 1
			player[currentPlayer].speed = 0
		end if

		if object.timer > 0
			if object.timer == 256
				PlayMusic(0)
				if warp.xpos > 0
					player[currentPlayer].xpos = warp.xpos
					player[currentPlayer].ypos = warp.ypos
					player[currentPlayer].xvel = warp.xvel
					player[currentPlayer].yvel = warp.yvel
					player[currentPlayer].speed = warp.xvel
					player[currentPlayer].state = warp.state
					stage.milliSeconds = recMilliSeconds
					stage.seconds = recSeconds
					stage.minutes = recMinutes
				end if
				warp.timer = 0
				warp.frame = 0
			end if
			SetScreenFade(object.fadeR, object.fadeG, object.fadeB, object.timer)
			object.timer -= 8
		else
			fadeColor = 0
			CheckEqual(warp.xpos, 0)
			temp0 = checkResult
			CheckLower(lampPostID, 32)
			temp0 &= checkResult
			if temp0 == 1
				object.state = TitleCard_State_SlideIn
				object.timer = 0
			else
				player[currentPlayer].controlMode = 0
				stage.timeEnabled = true
				ResetObjectEntity(20, TypeName[Blank Object], 0, 0, 0)
				warp.xpos = 0
			end if
		end if
		break

	case TitleCard_State_SlideIn
		temp0 = 144
		temp0 += object.charPos
		if object.textPos > temp0
			object.textPos -= 8
		else
			if object.barPos == 0
				object.state = TitleCard_State_Wait
			end if
		end if

		if object.barPos < 0
			object.barPos += 8
		end if
		break

	case TitleCard_State_Wait
		if object.timer == 90
			object.state = TitleCard_State_SlideOut
		else
			object.timer++
		end if
		break

	case TitleCard_State_SlideOut
		if object.textPos < 408
			object.textPos += 16
			object.barPos -= 16
		else
			player[currentPlayer].controlMode = CONTROLMODE_P1
			stage.timeEnabled = true
			ResetObjectEntity(20, TypeName[Blank Object], 0, 0, 0)
			warp.xpos = 0
		end if
		break

	end switch

	if object.type > 0
		temp0 = 136
		temp0 += object.charPos
		DrawSpriteScreenXY(0, temp0, object.barPos)
		DrawSpriteScreenXY(stage.actNum, object.textPos, 80)
		DrawSpriteScreenXY(4, object.textPos, 80)
		DrawSpriteScreenXY(5, object.textPos, 80)
		DrawSpriteScreenXY(6, object.textPos, 80)
		DrawSpriteScreenXY(7, object.textPos, 80)
		temp0 = 8
		while temp0 < object.textSize
			DrawSpriteScreenXY(temp0, object.textPos, 80)
			temp0++
		loop
	end if
end event

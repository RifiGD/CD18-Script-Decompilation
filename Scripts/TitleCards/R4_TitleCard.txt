event ObjectDraw
	object[0].value1 = 0
	switch object.state
	case 0
		object[0].controlMode = -1
		object[0].up = 0
		object[0].down = 0
		object[0].left = 0
		object[0].right = 0
		object[0].jumpPress = 0
		object[0].jumpHold = 0
		if warp.xpos < 1
			object[0].speed = 0
		end if
		if object.value3 > 0
			if object.value3 == 256
				PlayMusic(0)
				if warp.xpos > 0
					object[0].xpos = warp.xpos
					object[0].ypos = warp.ypos
					object[0].xvel = warp.xvel
					object[0].yvel = warp.yvel
					object[0].speed = warp.xvel
					object[0].state = warp.state
					stage.milliSeconds = recMilliSeconds
					stage.seconds = recSeconds
					stage.minutes = recMinutes
				end if
				warp.timer = 0
				warp.frame = 0
			end if
			SetScreenFade(object.value4, object.value5, object.value6, object.value3)
			object.value3 -= 8
		else
			fadeColor = 0
			CheckEqual(warp.xpos, 0)
			temp0 = checkResult
			CheckLower(lampPostID, 32)
			temp0 &= checkResult
			if temp0 == 1
				object.state = 1
				object.value3 = 0
			else
				object[0].controlMode = 0
				stage.timeEnabled = 1
				ResetObjectEntity(20, TypeName[Blank Object], 0, 0, 0)
				warp.xpos = 0
			end if
		end if
		break
	case 1
		temp0 = 144
		temp0 += object.value7
		if object.value1 > temp0
			object.value1 -= 8
		else
			if object.value0 == 0
				object.state = 2
			end if
		end if
		if object.value0 < 0
			object.value0 += 8
		end if
		break
	case 2
		if object.value3 == 90
			object.state = 3
		else
			object.value3++
		end if
		break
	case 3
		if object.value1 < 408
			object.value1 += 16
			object.value0 -= 16
		else
			object[0].controlMode = 0
			stage.timeEnabled = 1
			ResetObjectEntity(20, TypeName[Blank Object], 0, 0, 0)
			warp.xpos = 0
		end if
		break
	end switch
	if object.type > 0
		temp0 = 136
		temp0 += object.value7
		DrawSpriteScreenXY(0, temp0, object.value0)
		DrawSpriteScreenXY(stage.actNum, object.value1, 80)
		DrawSpriteScreenXY(4, object.value1, 80)
		DrawSpriteScreenXY(5, object.value1, 80)
		DrawSpriteScreenXY(6, object.value1, 80)
		DrawSpriteScreenXY(7, object.value1, 80)
		temp0 = 8
		while temp0 < object.value2
			DrawSpriteScreenXY(temp0, object.value1, 80)
			temp0++
		loop
	end if
end event


event ObjectStartup
	if stage.playerListPos == 0
		LoadSpriteSheet("Global/Display.gif")
	else
		LoadSpriteSheet("Global/Display_t.gif")
	end if
	if options.soundtrack == 0
		switch hud.currentTimePeriod
		case TIME_PRESENT
			SetMusicTrack("JP/R4A.ogg", 0, 1)
			break
		case TIME_PAST
			SetMusicTrack("R4B.ogg", 0, 1)
			break
		case TIME_GOOD_FUTURE
			SetMusicTrack("JP/R4C.ogg", 0, 1)
			break
		case TIME_BAD_FUTURE
			SetMusicTrack("JP/R4D.ogg", 0, 1)
			break
		end switch
	else
		switch hud.currentTimePeriod
		case TIME_PRESENT
			SetMusicTrack("US/R4A.ogg", 0, 415148)
			break

			break
		case TIME_BAD_FUTURE

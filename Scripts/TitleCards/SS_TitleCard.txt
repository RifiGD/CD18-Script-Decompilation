event ObjectDraw
	object[4].value1 = 100
	object[4].value2 = 0
	switch object.state
	case 0
		if object.value3 > 0
			if object.value3 == 256
				PlayMusic(0)
				object[2].controlMode = 0
			end if
			SetScreenFade(object.value4, object.value5, object.value6, object.value3)
			object.value3 -= 8
		else
			object.state = 1
			object.value3 = 0
		end if
		object[2].controlMode = -1
		break
	case 1
		temp0 = 144
		temp0 += object.value7
		if object.value1 > temp0
			object.value1 -= 8
		else
			if object.value0 == 0
				object.state = 2
			end if
		end if
		if object.value0 < 0
			object.value0 += 8
		end if
		break
	case 2
		if object.value3 == 160
			object.state = 3
		else
			object.value3++
		end if
		break
	case 3
		if object.value1 < 408
			object.value1 += 16
			object.value0 -= 16
		else
			ResetObjectEntity(20, TypeName[Blank Object], 0, 0, 0)
			object[2].controlMode = 0
			if options.gameMode == 2
				stage.timeEnabled = 1
			end if
		end if
		break
	end switch
	if object.type > TypeName[Blank Object]
		temp0 = 136
		temp0 += object.value7
		DrawSpriteScreenXY(0, temp0, object.value0)
		DrawSpriteScreenXY(5, object.value1, 64)
		DrawSpriteScreenXY(6, object.value1, 64)
		DrawSpriteScreenXY(7, object.value1, 64)
		temp0 = 8
		while temp0 < object.value2
			DrawSpriteScreenXY(temp0, object.value1, 64)
			temp0++
		loop
	end if
end event

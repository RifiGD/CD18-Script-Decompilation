			stage.newYBoundary2 *= -512
			stage.newYBoundary2 += object.iypos
			object.value1 = stage.newYBoundary2

			stage.curYBoundary2 *= -512
			stage.curYBoundary2 += object.iypos
			object.value1 = stage.curYBoundary2
		end if

			stage.curYBoundary2--

				stage.curYBoundary2--

	end if

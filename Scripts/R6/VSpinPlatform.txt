event ObjectUpdate
	object.angle = oscillation
	object.angle <<= 1
	if object.propertyValue == 0
		object.angle += 256
	end if
	Sin(object.value2, object.angle)
	object.value2 <<= 13
	object.value2 += object.ypos
	object.value2 &= -65536
	object.value2 -= object.value1
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		object.value3 = object.ypos
		object.ypos = object.value1
		BoxCollisionTest(C_PLATFORM, object.entityPos, -24, -12, 24, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[currentPlayer].ypos += object.value2
			if object[currentPlayer].state != Player_State_SpinningTop
				object[currentPlayer].speed = 0
				object.value4 = 1
				object[currentPlayer].value35 = 3072
				if object[currentPlayer].xpos > object.xpos
					object[currentPlayer].value1 = object[currentPlayer].xpos
					object[currentPlayer].value1 -= object.xpos
					object.value0 = 128
				else
					object[currentPlayer].value1 = object.xpos
					object[currentPlayer].value1 -= object[currentPlayer].xpos
					object.value0 = 384
				end if
			end if
			if object[currentPlayer].xpos > object.xpos
				if object[currentPlayer].left == true
					object[currentPlayer].value1 -= 2048
				end if
				if object[currentPlayer].right == true
					object[currentPlayer].value1 += 2048
				end if
			else
				if object[currentPlayer].left == true
					object[currentPlayer].value1 += 2048
				end if
				if object[currentPlayer].right == true
					object[currentPlayer].value1 -= 2048
				end if
			end if
			object[currentPlayer].state = Player_State_SpinningTop
			object[currentPlayer].animation = ANI_SPINNING_TOP
			Sin(object[currentPlayer].xpos, object.value0)
			object[currentPlayer].xpos *= object[currentPlayer].value1
			object[currentPlayer].xpos >>= 9
			object[currentPlayer].xpos += object.xpos
			temp0 = object.value0
			temp0 -= 128
			temp0 /= 86
			object[currentPlayer].frame = temp0
			object.value0 += 8
			if object.value0 > 639
				object.value0 -= 512
			end if
			if object[currentPlayer].jumpPress == true
				arrayPos1 = stage.entityPos
				stage.entityPos = currentPlayer
				CallFunction(Player_Action_Jump)
				stage.entityPos = arrayPos1
				object[currentPlayer].speed = object[currentPlayer].xpos
				object[currentPlayer].speed -= object.xpos
				object[currentPlayer].speed >>= 3
				object[currentPlayer].state = Player_State_Air
				object[currentPlayer].value1 = 0
				PlaySfx(SfxName[Jump], 0)
			end if
		else
			if object.value4 == 1
				if object[currentPlayer].state == Player_State_SpinningTop
					if object[currentPlayer].value43 == 0
						object[currentPlayer].state = Player_State_Ground
						object.value4 = 0
						object[currentPlayer].value1 = 0
					end if
				end if
			end if
		end if
		object.ypos = object.value3
	next
end event

event ObjectUpdate
	object.angle = oscillation
	object.angle <<= 1
	if object.propertyValue == 1
		object.angle += 256
	end if
	Sin(object.value2, object.angle)
	object.value2 <<= 13
	object.value2 += object.xpos
	object.value2 &= -65536
	object.value2 -= object.value1
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp0 = object.xpos
		object.xpos = object.value1
		BoxCollisionTest(C_PLATFORM, object.entityPos, -24, -12, 24, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[currentPlayer].xpos += object.value2
			if object[currentPlayer].state == Player_State_SpinningTop
				object[currentPlayer].state = Player_State_Ground
			end if
		end if
		object.xpos = temp0
	next
end event


event ObjectDraw
	Sin(object.value1, object.angle)
	object.value1 <<= 13
	object.value1 += object.xpos
	object.value1 &= -65536
	DrawSpriteXY(0, object.value1, object.ypos)
	DrawSpriteXY(1, object.value1, object.ypos)
end event

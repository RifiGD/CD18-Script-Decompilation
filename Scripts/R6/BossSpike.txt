event ObjectUpdate
	if object.state == 0
		object.ypos += 229376
		ObjectTileCollision(0, 0, 12, 0)
		if checkResult == true
			object.state = 1
		end if
	else
		object.value0++
		if object.value0 == 64
			object.type = TypeName[Blank Object]
		end if
	end if
	if object.state == 0
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, -4, -12, 4, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				CallFunction(Player_Hit)
			end if
		next
	end if
end event

	switch object.state
	case 0
		if object.value0 < 8
			object.value0++
			object.ypos -= 655360
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 1
		if object.value0 < 60
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 2
		if object.value0 < 80
			object.value0++
			object.ypos += 65536
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 3
		if object.value0 < 60
			object.value0++
		else
			object.value0 = 0
			object.state = 0
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -33, -40, 33, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[currentPlayer].xpos < object.xpos
				temp0 = object.xpos
				temp0 -= object[currentPlayer].xpos
				temp1 = 2359296
				temp1 -= object[currentPlayer].xvel
				if temp0 > temp1
					if object[currentPlayer].speed > -1
						object[currentPlayer].xpos = -32
						object[currentPlayer].xpos -= object[currentPlayer].collisionRight
						object[currentPlayer].xpos <<= 16
						object[currentPlayer].xpos += object.xpos
					end if
					if object[currentPlayer].speed > 0
						if object[currentPlayer].direction == 0
							object[currentPlayer].pushing = 2
						end if
						object[currentPlayer].speed = 0
						object[currentPlayer].xvel = 0
					end if
				else
					if object.state < 2
						if temp0 > 1572864
							object[currentPlayer].xpos += 1310720
							object[currentPlayer].speed = 0
							object[currentPlayer].xvel = 0
						end if
					end if
				end if
			else
				temp0 = object[currentPlayer].xpos
				temp0 -= object.xpos
				temp1 = 2359296
				temp1 += object[currentPlayer].xvel
				if temp0 > temp1
					if object[currentPlayer].speed < 1
						object[currentPlayer].xpos = 32
						object[currentPlayer].xpos -= object[currentPlayer].collisionLeft
						object[currentPlayer].xpos <<= 16
						object[currentPlayer].xpos += object.xpos
					end if
					if object[currentPlayer].speed < 0
						if object[currentPlayer].direction == 1
							object[currentPlayer].pushing = 2
						end if
						object[currentPlayer].speed = 0
						object[currentPlayer].xvel = 0
					end if
				else
					if object.state < 2
						if temp0 > 1572864
							object[currentPlayer].xpos -= 1310720
							object[currentPlayer].speed = 0
							object[currentPlayer].xvel = 0
						end if
					end if
				end if
			end if
		end if
		switch object.state
		case 0
			if object.value0 < 4
				BoxCollisionTest(C_PLATFORM, object.entityPos, -30, -40, 30, -20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			else
				BoxCollisionTest(C_TOUCH, object.entityPos, -16, -40, 16, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object[currentPlayer].speed >>= 1
					object[currentPlayer].xvel >>= 1
					BoxCollisionTest(C_SOLID, object.entityPos, -48, -80, -32, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
					BoxCollisionTest(C_SOLID, object.entityPos, 32, -80, 48, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
					if object[currentPlayer].state == Player_State_Fly
						object[currentPlayer].state = Player_State_Air
						object[currentPlayer].animation = ANI_WALKING
					end if
				end if
			end if
			break
		case 2
			BoxCollisionTest(C_PLATFORM, object.entityPos, -30, -41, 30, -20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object[currentPlayer].ypos += 65536
			end if
			break
		case 3
			BoxCollisionTest(C_PLATFORM, object.entityPos, -30, -40, 30, -20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

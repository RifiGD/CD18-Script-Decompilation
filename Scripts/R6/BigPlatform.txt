// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	switch object.state
	case 0
		BoxCollisionTest(C_TOUCH, object.entityPos, -96, -160, 128, 160, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			temp0 = object[0].xpos
			temp0 -= object.xpos
			if temp0 > 8388608
				object.state = 2
				object.priority = 1
			else
				object.state = 1
				object.priority = 1
			end if
		end if
		break
	case 1
		temp0 = object[0].xpos
		temp0 -= object.xpos
		if temp0 > 8388608
			object.state = 2
		end if
		if object.angle > 0
			object.angle -= 2
		end if
		break
	case 2
		if object[0].xpos < object.xpos
			object.state = 1
		end if
		if object.angle < 256
			object.angle += 2
		end if
		break
	end switch
	Cos(object.value2, object.angle)
	object.value2 <<= 13
	object.value2 += 4194304
	object.value2 += object.ypos
	object.value2 &= -65536
	object.value2 -= object.value1
	BoxCollisionTest(C_TOUCH, object.entityPos, -328, -256, 328, 256, 0, C_BOX, C_BOX, C_BOX, C_BOX)
	if checkResult == false
		object.state = 0
		object.priority = 0
		object.angle = 256
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp0 = object.ypos
		object.ypos = object.value1
		BoxCollisionTest(C_SOLID, object.entityPos, -48, -81, 48, 80, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[currentPlayer].ypos += object.value2
			object[currentPlayer].ypos += 65536
		end if
		object.ypos = temp0
	next
end event


event ObjectDraw
	Cos(object.value1, object.angle)
	object.value1 <<= 13
	object.value1 += 4194304
	object.value1 += object.ypos
	object.value1 &= -65536
	DrawSpriteXY(0, object.xpos, object.value1)
end event


event ObjectStartup
	LoadSpriteSheet("R6/Objects4.gif")
	if hud.currentTimePeriod == TIME_GOOD_FUTURE
		SpriteFrame(-48, -80, 96, 192, 98, 1)
	else
		SpriteFrame(-48, -80, 96, 192, 1, 1)
	end if
	foreach (TypeName[Big Platform], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].angle = 256
	next
end event

event RSDKDraw
	DrawSprite(0)
end event

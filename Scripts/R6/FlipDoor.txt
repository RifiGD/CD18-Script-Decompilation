event ObjectUpdate

		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			if object[currentPlayer].interaction == 1
				BoxCollisionTest(C_SOLID, object.entityPos, -4, -32, 4, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			end if
		next
	end if
	if object.direction == 0
		switch object.state
		case 0
			object.frame = 0
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				BoxCollisionTest(C_TOUCH, object.entityPos, -48, -32, 0, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == 1
					object.state = 1
					PlaySfx(SfxName[Door], 0)
				end if
			next
			break
		case 1
			if object.value0 < 4
				object.value0++
				object.frame = 1
			else
				object.value0 = 0
				object.state = 2
			end if
			break
		case 2
			object.frame = 2
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				BoxCollisionTest(C_TOUCH, object.entityPos, -48, -32, 64, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == 0
					object.state = 3
				end if
			next
			break
		case 3
			if object.value0 < 4
				object.value0++
				object.frame = 1
			else
				object.value0 = 0
				object.state = 0
			end if
			break
		end switch
	else
		switch object.state
		case 0
			object.frame = 0
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				BoxCollisionTest(C_TOUCH, object.entityPos, 0, -32, 48, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)

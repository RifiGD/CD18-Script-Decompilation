private alias 69 : TYPE_HANGINGTRIGGER



event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -8, -2, 8, 20, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if object[currentPlayer].yvel > -1
			if object[currentPlayer].xpos > object.xpos
				if checkResult == 1
					if object[currentPlayer].animation != ANI_HANGING
						object[currentPlayer].state = Player_State_HangBar
						object[currentPlayer].yvel = 0
						object[currentPlayer].animation = ANI_HANGING
						object[currentPlayer].ypos = object[currentPlayer].collisionTop
						object[currentPlayer].ypos *= -65536
						object[currentPlayer].ypos += object.ypos
						PlaySfx(SfxName[Catch], 0)
						screen[0].adjustCameraY = 0
					end if
				end if
			end if
		end if
	next
end event

	DrawSprite(0)
end event

	if object.state == 0
		if object[+1].propertyValue == 1
			object.state = 1
		end if
		if object.value0 > 0
			object.value0 -= 262144
			object.ypos += 262144
		end if
	else
		if object.value0 < 4194304
			object.value0 += 262144
			object.ypos -= 262144
		end if
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.propertyValue == 0
			if object.state == 1
				BoxCollisionTest(C_SOLID, object.entityPos, -8, -32, 8, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				BoxCollisionTest(C_TOUCH, object.entityPos, -64, -128, -32, 128, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object.state = 0
				end if
			else
				BoxCollisionTest(C_SOLID, object.entityPos, -8, -32, 8, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			end if
		else
			if object.state == 1
				BoxCollisionTest(C_SOLID, object.entityPos, -8, -32, 8, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				BoxCollisionTest(C_TOUCH, object.entityPos, 32, -128, 64, 128, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object.state = 0
				end if
			else
				BoxCollisionTest(C_SOLID, object.entityPos, -8, -32, 8, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			end if
		end if

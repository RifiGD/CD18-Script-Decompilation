event ObjectUpdate
	switch object.state
	case EGGMANSTATUE_DESTROYED
		temp0 = object.timer
		temp0 &= 3
		if temp0 == 0
			Rand(temp0, 32)
			temp0 -= 16
			temp0 <<= 16
			temp0 += object.xpos

			Rand(temp1, 48)
			temp1 -= 24
			temp1 <<= 16
			temp1 += object.ypos

			CreateTempObject(TypeName[Explosion], 0, temp0, temp1)
			PlaySfx(SfxName[Explosion], 0)
		end if

		object.timer++
		if object.timer == 40
			object.timer = 0
			object.state++
		end if
		break

	case EGGMANSTATUE_BOMB_DROP
		object.timer++
		if object.timer == 60
			object.timer = 0
			CallFunction(EggmanStatue_HandleBombDrop)
			object.type = TypeName[Blank Object]
		end if
		break

	end switch

	if object.state == EGGMANSTATUE_INACTIVE
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			if object[currentPlayer].animation == ANI_JUMPING
				BoxCollisionTest(C_TOUCH, object.entityPos, -16, -24, 16, 24, currentPlayer, object[currentPlayer].value41, object[currentPlayer].value39, object[currentPlayer].value42, object[currentPlayer].value40)
				if checkResult == true
					object.state = EGGMANSTATUE_DESTROYED
				end if
			end if
		next
#platform: USE_DECOMP
	// Strangely enough, the fact that an endif is missing doesn't crash on the original application (unlike the foreach bugs in R1)
	end if
#endplatform
end event

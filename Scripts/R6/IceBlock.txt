event ObjectUpdate
	switch object.propertyValue
	case 0
		if object.state == 0
			object.ypos += object.yvel
			object.yvel += object[0].value24
			if object.yvel > 1310720
				object.yvel = 1310720
			end if
			ObjectTileCollision(0, 0, 24, 0)
			if checkResult == 1
				object.state = 1
				object.yvel = 0
				PlaySfx(SfxName[Glass Smash], 0)
			end if
			object[0].xpos = object.xpos
			object[0].ypos = object.ypos
		else
			if object.value0 < 16
				object.value0++
				if object[0].type == TypeName[Player Object]
					if object[0].jumpPress == 1
						CallFunction(IceBlock_Function87)
						object.type = TypeName[Blank Object]
						arrayPos1 = stage.entityPos
						stage.entityPos = 0
						CallFunction(Player_Action_Jump)
						stage.entityPos = arrayPos1
						object[0].visible = 1
						object[0].interaction = 1
					end if
				end if
			else
				CallFunction(IceBlock_Function87)
				object.type = TypeName[Blank Object]
				object[0].state = 13
				CallFunction(Player_Hit)
				object[0].visible = 1
				object[0].interaction = 1
			end if
		end if
		break
	case 1
		if object.value0 < 10
			object.value0++
			object.xpos += object.xvel
			object.ypos += object.yvel
		else
			object.type = TypeName[Blank Object]
			CreateTempObject(TypeName[Ice Block], 4, object.xpos, object.ypos)
			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].xvel = 0
			object[tempObjectPos].yvel = -65536
			CreateTempObject(TypeName[Ice Block], 4, object.xpos, object.ypos)
			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].xvel = -65536
			object[tempObjectPos].yvel = 0
			CreateTempObject(TypeName[Ice Block], 4, object.xpos, object.ypos)
			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].xvel = 65536
			object[tempObjectPos].yvel = 0
			CreateTempObject(TypeName[Ice Block], 4, object.xpos, object.ypos)
			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].xvel = 0
			object[tempObjectPos].yvel = 65536
		end if
		break
	case 2
		if object.value0 < 16
			object.value0++
			object.xpos += object.xvel
			object.ypos += object.yvel
		else
			object.type = TypeName[Blank Object]

			object[tempObjectPos].drawOrder = 4
			object[tempObjectPos].xvel = 65536
			object[tempObjectPos].yvel = 65536
		end if
		break
	case 3
		if object.value0 < 1
			object.value0++
		else
			object.type = TypeName[Blank Object]
		end if
		break
	case 4
		if object.value0 < 12

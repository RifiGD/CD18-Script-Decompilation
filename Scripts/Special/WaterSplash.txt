private alias 9 : TYPE_WATERSPLASH



event ObjectDraw
	if object[2].ypos > 0
		ResetObjectEntity(3, TypeName[Blank Object], 0, 0, 0)
	else
		temp0 = object[2].xpos
		temp0 >>= 16
		temp1 = object[2].value0
		temp1 >>= 16
		Get16x16TileInfo(checkResult, temp0, temp1, 6)
		if checkResult == 2
			if object.frame == 0
				PlaySfx(SfxName[Water Splash], 0)
				if stage.timeEnabled == 0
					object[4].value7 = 40
				end if
			end if
			object.frame++
			if object.frame == 40
				object.frame = 28
			end if
			temp0 = object.frame
			temp0 >>= 2
			DrawSpriteScreenXY(temp0, screen.xcenter, 224)
		else
			ResetObjectEntity(3, TypeName[Blank Object], 0, 0, 0)
			object[4].value7 = 0
		end if
	end if
end event


event ObjectStartup
	LoadSpriteSheet("Special/Objects.gif")
	SpriteFrame(-20, -32, 40, 32, 1, 84)
	SpriteFrame(-20, -32, 40, 32, 42, 84)
	SpriteFrame(-20, -32, 40, 32, 83, 84)
	SpriteFrame(-20, -48, 40, 48, 124, 58)
	SpriteFrame(-20, -64, 40, 64, 165, 58)
	SpriteFrame(-20, -8, 40, 8, 124, 107)
	SpriteFrame(-20, -8, 40, 8, 124, 116)
	SpriteFrame(-20, -32, 40, 32, 42, 84)
	SpriteFrame(-20, -32, 40, 32, 83, 84)
	SpriteFrame(-20, -32, 40, 32, 1, 84)
end event

end event

event ObjectUpdate
	object.xpos += object.value1
	object.ypos += object.value2
	object.value0 += object.value3
	object.value2 -= 8192
	if object.ypos < 0
		object.ypos = 0
		FlipSign(object.value2)
	end if
	temp0 = object.xpos
	temp0 -= tileLayer[0].xpos
	temp0 >>= 8
	temp1 = object.value0
	temp1 -= tileLayer[0].zpos
	temp1 >>= 8
	Sin(temp2, tileLayer[0].angle)
	temp2 *= temp1
	Cos(temp3, tileLayer[0].angle)
	temp3 *= temp0
	object.value6 = temp2
	object.value6 -= temp3
	object.value6 >>= 9
	Cos(temp2, tileLayer[0].angle)
	temp2 *= temp1
	Sin(temp3, tileLayer[0].angle)
	temp3 *= temp0
	object.value5 = temp2
	object.value5 += temp3
	object.value5 >>= 9
	object.value4++
	if object.value4 == 160
		object.type = TypeName[Blank Object]
	end if
end event


event ObjectDraw
	if object.value5 > 256
		object.scale = 12582912
		object.scale /= object.value5
		temp0 = 768
		temp0 -= object.scale
		temp0 /= 6
		temp1 = object.value5
		temp1 >>= 12
		if temp0 > 0
			object.scale += temp0
			object.scale -= temp1
		end if
		temp0 = object.value6
		temp0 *= 192
		temp0 /= object.value5
		temp0 += screen.xcenter
		temp1 = tileLayer[0].ypos
		temp1 -= object.ypos
		temp1 >>= 8
		temp1 *= 96
		temp1 /= object.value5
		temp1 += 128
		temp2 = object.value4
		temp2 &= 31
		temp2 >>= 3
		DrawSpriteScreenFX(temp2, FX_SCALE, temp0, temp1)
		temp1 = tileLayer[0].ypos
		temp1 >>= 8
		temp1 *= 96
		temp1 /= object.value5
		temp1 += 128
		DrawSpriteScreenFX(4, FX_SCALE, temp0, temp1)
	end if
end event

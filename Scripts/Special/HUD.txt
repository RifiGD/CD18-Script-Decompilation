private alias 4 : TYPE_HUD



event ObjectDraw
	temp1 = screen.xcenter
	temp1 -= 128
	DrawSpriteScreenXY(10, temp1, 10)
	temp0 = object.value0
	temp0 /= 10
	temp1 += 24
	DrawSpriteScreenXY(temp0, temp1, 11)
	temp0 = object.value0
	temp0 %= 10
	temp1 += 8
	DrawSpriteScreenXY(temp0, temp1, 11)
	temp1 += 48
	DrawSpriteScreenXY(11, temp1, 16)
	if options.gameMode < 2
		temp0 = object.value1
		temp0 /= 100
		temp1 += 40
		DrawSpriteScreenXY(temp0, temp1, 11)
		temp0 = object.value1
		temp0 %= 100
		temp0 /= 10
		temp1 += 8
		DrawSpriteScreenXY(temp0, temp1, 11)
		temp0 = object.value1
		temp0 %= 10
		temp1 += 8
		DrawSpriteScreenXY(temp0, temp1, 11)
		temp1 += 64
	else
		if stage.minutes > 29
			stage.minutes = 30
			stage.seconds = 0
			stage.milliSeconds = 0
		end if
		temp1 += 40
		DrawNumbers(0, temp1, 11, stage.minutes, 2, 8, 1)
		temp1 += 8
		DrawSpriteScreenXY(13, temp1, 11)
		temp1 += 16
		DrawNumbers(0, temp1, 11, stage.seconds, 2, 8, 1)
		temp1 += 8
		DrawSpriteScreenXY(14, temp1, 11)
		temp1 += 16
		DrawNumbers(0, temp1, 11, stage.milliSeconds, 2, 8, 1)
		temp1 += 32
	end if
	DrawSpriteScreenXY(12, temp1, 16)
	temp0 = object.value3
	temp0 /= 100
	temp1 += 32
	DrawSpriteScreenXY(temp0, temp1, 11)
	temp0 = object.value3
	temp0 %= 100
	temp0 /= 10
	temp1 += 8
	DrawSpriteScreenXY(temp0, temp1, 11)
	temp0 = object.value3
	temp0 %= 10
	temp1 += 8
	DrawSpriteScreenXY(temp0, temp1, 11)
end event

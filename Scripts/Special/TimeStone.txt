event ObjectDraw
	temp0 = object.frame
	temp0 /= 12
	object.frame++
	if object.frame > 47
		object.frame = 0
	end if
	if object.state == 0
		temp1 = object.ypos
		temp1 -= 2097152
		DrawSpriteFX(5, FX_FLIP, object.xpos, temp1)
		temp1 += 786432
		DrawSpriteFX(4, FX_FLIP, object.xpos, temp1)
		object.direction = object.frame
		object.direction %= 24
		object.direction /= 6
		if object.iypos < 200
			object.ypos += 98304
		else
			object.iypos = 200
			object.state = 1
			PlaySfx(SfxName[Achievement], 0)
			object[2].state = 10
			object[2].frame++
		end if
	end if
	DrawSprite(temp0)
end event


event ObjectStartup
	LoadSpriteSheet("Special/Objects.gif")
	SpriteFrame(-10, -12, 20, 24, 1, 232)
	SpriteFrame(-10, -12, 20, 24, 22, 232)
	SpriteFrame(-10, -12, 20, 24, 43, 232)
	SpriteFrame(-10, -12, 20, 24, 64, 232)
	SpriteFrame(-12, -12, 24, 24, 83, 143)
	SpriteFrame(-8, -8, 16, 16, 83, 126)
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad

end event

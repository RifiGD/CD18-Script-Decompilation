public function SSPlayer_HandleMovement
	if object.left == 1
		object.value7--
		if object.value7 < -8
			object.value7 = -8
		end if
	else
		if object.right == 1
			object.value7++
			if object.value7 > 8
				object.value7 = 8
			end if
		else
			if object.value7 > 0
				object.value7--
			end if
			if object.value7 < 0
				object.value7++
			end if
		end if
	end if
	if object.left == 1
		object.angle += 2
	end if
	if object.right == 1
		object.angle -= 2
	end if
	if object.angle < 0
		object.angle += 512
	end if
	object.angle &= 511
	Sin(temp0, object.angle)
	temp0 *= object.speed
	temp0 >>= 9
	object.xpos += temp0
	Cos(temp0, object.angle)
	temp0 *= object.speed
	temp0 >>= 9
	object.value0 += temp0
end function


public function SSPlayer_ProcessAnimation
	object.value3 += object.animationSpeed
	if object.value3 > 239
		object.value3 -= 240
		object.frame++
		if object.frame > object.value2
			object.frame = object.value1
		end if
	end if
end function

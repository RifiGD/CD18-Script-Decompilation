event ObjectUpdate
	if object[-playerCount].state == 26
		object.type = TypeName[Blank Object]
	end if
	if object[-playerCount].state == 27
		object.type = TypeName[Blank Object]
	end if
	object.drawOrder = object[-playerCount].drawOrder
	object.frame++
	if object.frame >= 7
		object.frame = 0
		object.type = TypeName[Blank Object]
	else
		object[-playerCount].value7 = 1
	end if
	if object.type == TypeName[Insta Shield]
		object[-playerCount].value39 = -24
		object[-playerCount].value40 = 24
		object[-playerCount].value41 = -24
		object[-playerCount].value42 = 24
	else
		object[-playerCount].value39 = 65536
		object[-playerCount].value40 = 65536
		object[-playerCount].value41 = 65536
		object[-playerCount].value42 = 65536
	end if
end event


event ObjectDraw
	DrawSpriteXY(object.frame, object[-playerCount].xpos, object[-playerCount].ypos)
end event

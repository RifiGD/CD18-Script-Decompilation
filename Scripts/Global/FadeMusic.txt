private alias 65 : TYPE_FADEMUSIC



event ObjectUpdate
	switch object.propertyValue
	case 0
		if object.value0 < 100
			object.value0++
			music.volume--
		else
			PlayMusic(4)
			object.type = TypeName[Blank Object]
		end if
		break
	case 1
		if object.value0 < 100
			object.value0++
			music.volume--
		else
			PlayMusic(0)
			object.type = TypeName[Blank Object]
		end if
		break
	case 2
		if object.value0 < 50
			object.value0++
			music.volume -= 2
		else
			PlayMusic(0)
			object.type = TypeName[Blank Object]
		end if
		break
	case 3
		if object.value0 < 100
			object.value0++
			music.volume--
		else
			PlayMusic(1)
			object.type = TypeName[Blank Object]
			object[30].type = TypeName[Act Finish]
			object[30].drawOrder = 6
			object[30].priority = 1
		end if
		break
	end switch
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

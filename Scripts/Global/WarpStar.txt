private alias 32 : TYPE_WARPSTAR


event ObjectUpdate
	switch object.state
	case 0
		warp.count++
		if warp.count > 2
			warp.count = 0
		end if
		object.value0--
		if object.value0 < 1
			if warp.timer > 0
				warp.frame = object.frame
				object.state = 1
				object.inkEffect = 2
				object.alpha = 255
				CreateTempObject(TypeName[Warp Star], 0, object[0].xpos, object[0].ypos)
				if object.speed < 0
					object[tempObjectPos].value0 = object.speed
					FlipSign(object[tempObjectPos].value0)
				else
					object[tempObjectPos].value0 = object.speed
				end if
				object[tempObjectPos].value0 >>= 19
				object[tempObjectPos].value0 += 7
				object[tempObjectPos].drawOrder = 4
				object[tempObjectPos].frame = object.frame
			else
				object.type = TypeName[Blank Object]
			end if
		else
			if warp.timer == 0
				object.type = TypeName[Blank Object]
			end if
		end if
		break
	case 1
		object.value0++
		object.alpha -= 8
		if object.value0 == 16
			object.type = TypeName[Blank Object]
		end if
		break
	case 2
		object.yvel += 12288
		object.ypos += object.yvel
		object.value0++
		if object.value0 == object.value1
			object.type = TypeName[Blank Object]
		end if
		break
	end switch
end event


event ObjectDraw
	switch object.state
	case 0
		if warp.timer > 0
			DrawSprite(object.frame)
		end if
		object.frame++
		object.frame &= 3
		break
	case 1
		if warp.count != 2
			if warp.timer > 0
				DrawSpriteFX(warp.frame, FX_INK, object.xpos, object.ypos)
			end if
		end if
		break
	case 2
		temp0 = object.frame
		temp0 >>= 1
		DrawSprite(temp0)
		object.frame++
		object.frame &= 7
		break
	end switch
end event


event ObjectStartup
	LoadSpriteSheet("Global/Items3.gif")
	SpriteFrame(-16, -16, 32, 32, 50, 133)
	SpriteFrame(-16, -16, 32, 32, 50, 166)
	SpriteFrame(-24, -24, 48, 48, 1, 197)
	SpriteFrame(-16, -16, 32, 32, 50, 199)
	SpriteFrame(-12, -12, 24, 24, 148, 152)
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

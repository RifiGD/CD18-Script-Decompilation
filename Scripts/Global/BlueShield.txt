private alias 25 : TYPE_BLUESHIELD


event ObjectUpdate
	object.drawOrder = object[-playerCount].drawOrder
	if object[-playerCount].state == 26
		object.type = TypeName[Blank Object]
	end if
	if object[-playerCount].state == 27
		object.type = TypeName[Blank Object]
	end if
	object.value0++
	if object.value0 == 2
		object.value0 = 0
		object.frame++
		if object.frame > 5
			object.frame = 0
		end if
	end if
end event


event ObjectDraw
	if warp.timer == 0
		if object[-playerCount].visible == 1
			DrawSpriteFX(object.frame, FX_INK, object[-playerCount].xpos, object[-playerCount].ypos)
		end if
	end if
end event


event ObjectStartup
	if stage.playerListPos == 0
		LoadSpriteSheet("Global/Items2.gif")
	else
		LoadSpriteSheet("Global/Items2_t.gif")
	end if
	SpriteFrame(-24, -24, 48, 48, 109, 196)
	SpriteFrame(0, 0, 0, 0, 109, 196)
	SpriteFrame(-24, -24, 48, 48, 158, 196)
	SpriteFrame(0, 0, 0, 0, 109, 196)
	SpriteFrame(-24, -24, 48, 48, 207, 196)
	SpriteFrame(0, 0, 0, 0, 109, 196)
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

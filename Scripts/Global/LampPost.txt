// ========================
// Function Definitions

public function LampPost_DebugDraw
	DrawSprite(0)
end function


public function LampPost_DebugSpawn
	CreateTempObject(TypeName[Lamp Post], 0, object.xpos, object.ypos)
end function


// ========================
// Events
// ========================

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state == 0
			BoxCollisionTest(C_TOUCH, object.entityPos, -8, -32, 8, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				lampPostID = object.entityPos
				if object[0].xpos < object.xpos
					lampPostSide = -1572864
					ObjectTileCollision(2, -24, 0, 0)
					if checkResult == true
						lampPostSide = 0
					end if
				else
					lampPostSide = 1572864
					ObjectTileCollision(1, 24, 0, 0)
					if checkResult == true
						lampPostSide = 0
					end if
				end if
				recMilliSeconds = stage.milliSeconds
				recSeconds = stage.seconds
				recMinutes = stage.minutes
				object.state = 1
				object.angle = 384
				PlaySfx(SfxName[Lamp Post], 0)
			end if
		end if
	next
end event


event ObjectDraw
	switch object.state
	case 0
		DrawSprite(0)
		break
	case 1
		DrawSprite(1)
		Cos(temp0, object.angle)
		temp0 <<= 10
		Sin(temp1, object.angle)
		temp1 <<= 10
		temp0 += object.xpos
		temp1 += object.ypos
		temp1 -= 1572864
		DrawSpriteXY(2, temp0, temp1)
		object.angle += 16
		if object.angle > 896
			object.angle = 0
			object.state = 2
		end if
		break
	case 2
		DrawSprite(1)
		object.angle++
		if object.angle == 8
			object.angle = 0
		end if
		temp0 = object.ypos
		temp0 -= 2097152
		if object.angle > 3
			DrawSpriteXY(4, object.xpos, temp0)
		else
			DrawSpriteXY(3, object.xpos, temp0)
		end if
		break
	end switch
end event


event ObjectStartup
	LoadSpriteSheet("Global/Items.gif")
	SpriteFrame(-8, -40, 16, 64, 1, 137)
	SpriteFrame(-8, -24, 16, 48, 1, 153)
	SpriteFrame(-8, -8, 16, 16, 1, 137)
	SpriteFrame(-8, -8, 16, 16, 1, 236)
	SpriteFrame(-8, -8, 16, 16, 18, 236)
	if warp.xpos == 0
		if hud.currentTimePeriod == TIME_PRESENT
			if lampPostID > 31
				arrayPos0 = lampPostID
				object[0].xpos = object[arrayPos0].xpos
				object[0].xpos += lampPostSide
				object[0].ypos = object[arrayPos0].ypos
				screen[0].cameraX = object[arrayPos0].ixpos
				screen[0].cameraY = object[arrayPos0].iypos
				object[arrayPos0].state = 2
				stage.milliSeconds = recMilliSeconds
				stage.seconds = recSeconds
				stage.minutes = recMinutes
			end if
		end if
	end if
	SetTableValue(TypeName[Lamp Post], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(LampPost_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(LampPost_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

end event

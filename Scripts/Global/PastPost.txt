event ObjectUpdate
	switch object.state
	case 0
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			if object[currentPlayer].collisionPlane == object.collisionPlane
				BoxCollisionTest(C_TOUCH, object.entityPos, -8, -24, 8, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					warp.destination = 1
					object.state = 1
					if object.priority != 4
						object.priority = 1
					end if
					object.frame = 1
					PlaySfx(SfxName[Past], 0)
				end if
			end if
		next
		break
	case 1
		object.value0++
		if object.value0 == 56
			object.value0 = 0
			object.state = 2
			if object.priority != 4
				object.priority = 0
			end if
		end if
		temp0 = object.value0
		temp0 &= 1
		object.frame += temp0
		if object.frame > 8
			object.frame = 1
		end if
		break
	end switch
end event


event ObjectDraw
	switch object.state
	case 0
		DrawSprite(0)
		DrawSprite(1)
		break
	case 1
		DrawSprite(0)
		DrawSprite(object.frame)
		break
	case 2
		DrawSprite(0)
		DrawSprite(5)
		break
	end switch
end event

event ObjectUpdate
	object[2].value3 = object[2].value2
	object[2].value2 = object[2].value1
	object[2].value1 = object[2].value0
	object[2].value0 = object[1].value7
	object[1].value7 = object[1].value6
	object[1].value6 = object[1].value5
	object[1].value5 = object[1].value4
	object[1].value4 = object[1].value3
	object[1].value3 = object[1].value2
	object[1].value2 = object[1].value1
	object[1].value1 = object[1].value0
	object[1].value0 = object.ypos
	switch object.state
	case 0
		object.ypos -= 524288
		if object.ypos < 7864320
			object.ypos = 7864320
			object.state = 1
			object.value1 = 0
		end if
		break
	case 1
		object.value1++
		if object.value1 == 120
			object.value1 = 0
			object.state = 2
			object[0].state = 4
		end if
		break
	case 2
		object.ypos -= 524288
		if object.outOfBounds == 1
			object.type = TypeName[Blank Object]
		end if
		break
	end switch
	object.value2++
	if object.value2 == 4
		object.value2 = 0
		Rand(temp0, 32)
		temp0 -= 16
		temp0 <<= 16
		temp0 += object.xpos
		Rand(temp1, 32)
		temp1 -= 16
		temp1 <<= 16
		temp1 += object.ypos
		CreateTempObject(TypeName[Warp Star], 0, temp0, temp1)
		object[tempObjectPos].state = 2
		Rand(object[tempObjectPos].value1, 32)
		object[tempObjectPos].value1 += 16
	end if
end event


event ObjectDraw
	temp0 = object.frame
	temp0++
	if temp0 > 4
		temp0 = 0
	end if
	if object.ypos < 17039360
		object.inkEffect = 2
		object.alpha = 160
		DrawSpriteFX(temp0, FX_INK, object.xpos, object[1].value3)
		object.alpha = 128
		DrawSpriteFX(temp0, FX_INK, object.xpos, object[1].value7)
		object.alpha = 96
		DrawSpriteFX(temp0, FX_INK, object.xpos, object[2].value3)
	end if
	DrawSprite(object.frame)
	object.value0++
	if object.value0 > 2
		object.value0 = 0
		object.frame++
		if object.frame > 4
			object.frame = 0
		end if
	end if
end event

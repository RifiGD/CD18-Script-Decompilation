event ObjectDraw
	if object[0].value0 == 0
		object.value0++
		if object.value0 == 16
			object.value0 = 0
		end if
	else
		if stage.minutes == 9
			object.value0++
			if object.value0 == 16
				object.value0 = 0
			end if
		end if
	end if
	DrawSpriteScreenXY(10, 17, 13)
	if stage.minutes == 9
		if object.value0 > 7
			DrawSpriteScreenXY(11, 17, 29)
		end if
	end if
	if object[0].value0 == 0
		if object.value0 > 7
			DrawSpriteScreenXY(12, 17, 45)
		end if
	end if
	DrawSpriteScreenXY(13, 67, 29)
	DrawNumbers(0, 104, 13, player.score, 6, 8, 0)
	DrawNumbers(0, 104, 29, stage.milliSeconds, 2, 8, 1)
	DrawNumbers(0, 80, 29, stage.seconds, 2, 8, 1)
	DrawNumbers(0, 56, 29, stage.minutes, 1, 8, 1)
	DrawNumbers(0, 80, 45, object[0].value0, 3, 8, 0)
	if stage.debugMode == true
		temp0 = object[0].xpos
		temp0 >>= 16
		temp1 = screen.xsize
		temp1 -= 24
		DrawNumbers(0, temp1, 13, temp0, 5, 8, 1)
		temp0 = object[0].ypos
		temp0 >>= 16
		if temp0 < 0
			FlipSign(temp0)
		end if
		DrawNumbers(0, temp1, 29, temp0, 5, 8, 1)
		temp1 -= 42
		DrawSpriteScreenXY(14, temp1, 15)
		DrawSpriteScreenXY(21, temp1, 31)
	end if
	temp0 = object.propertyValue
	temp0 += 17
	DrawSpriteScreenXY(temp0, 16, 212)
	DrawSpriteScreenXY(14, 32, 220)
	if player.lives < 10
		DrawNumbers(0, 40, 217, player.lives, 1, 8, 1)
		temp1 = 52
	else
		DrawNumbers(0, 48, 217, player.lives, 2, 8, 1)
		temp1 = 60
	end if
	switch warp.destination
	case 1
		if warp.timer < 100
			DrawSpriteScreenXY(15, temp1, 214)
		else
			temp0 = warp.timer
			temp0 &= 1
			if temp0 == 0
				DrawSpriteScreenXY(15, temp1, 214)
			end if
		end if
		break
	case 2
		if warp.timer < 100
			DrawSpriteScreenXY(16, temp1, 214)
		else
			temp0 = warp.timer
			temp0 &= 1
			if temp0 == 0
				DrawSpriteScreenXY(16, temp1, 214)
			end if
		end if
		break
	end switch
end event

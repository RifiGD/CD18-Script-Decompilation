event ObjectUpdate
	if object.state == 0
		if object[0].value0 > 49
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object[0].type = TypeName[Blank Object]
				object[playerCount].type = TypeName[Blank Object]
				object[0].controlMode = -1
				object[0].direction = 0
				ResetObjectEntity(4, TypeName[Blank Object], 0, 0, 0)
				object.drawOrder = 4
				object.state = 1
				object.frame = 32
				StopSfx(SfxName[Flying])
				StopSfx(SfxName[Tired])
				PlaySfx(SfxName[Special Ring], 0)
				object[0].left = 0
				object[0].right = 0
			end if
		end if
	end if
end event


event ObjectDraw
	switch object.state
	case 0
		if object[0].value0 > 49
			temp0 = object.frame
			temp0 >>= 2
			if temp0 > 3
				object.direction = 1
			else
				object.direction = 0
			end if
			DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
			object.frame++
			if object.frame > 31
				object.frame = 0
			end if
		end if
		break
	case 1
		temp0 = object.frame
		temp0 >>= 2
		if temp0 > 11
			object.direction = 1
		else
			object.direction = 0
		end if
		DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
		if temp0 == 11
			object.direction = 1
			DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
		else
			if temp0 == 15
				object.direction = 1
				DrawSpriteFX(temp0, FX_FLIP, object.xpos, object.ypos)
			end if
		end if
		object.frame++
		if object.frame > 63
			ResetObjectEntity(object.entityPos, TypeName[Blank Object], 0, 0, 0)
		end if
		break
	end switch
end event

event ObjectUpdate
	if object.state == GOALPOST_IDLE
		temp0 = object.ypos
		temp0 += 0x600000

		if object[0].ypos < temp0
			if object[0].xpos > object.xpos
				// Block the player from going back into the stage, reset time warps
				stage.newXBoundary1 = object.xpos
				stage.newXBoundary1 >>= 16
				stage.newXBoundary1 -= 160

				warp.destination = 0
				warp.timer = 0

				object.state = GOALPOST_BLANK
				if object.propertyValue == UPPERGOAL
					BoxCollisionTest(C_TOUCH, object.entityPos, -32, -256, 32, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
					if checkResult == true
						// Take The High Road Unlock Criteria
						if stage.playerListPos == PLAYER_SONIC
							if stage.debugMode == false
								// Another one in the Achievement Graveyard...
							end if
						end if
					end if
				end if
			end if
		end if
	end if
end event

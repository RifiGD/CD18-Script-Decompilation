event ObjectUpdate
	// if the timer is disabled
	if DebugMode_unusedTimer == 0
		temp0 = 0
		if keyDown[0].left == 1
			object.xpos -= object.yvel
			temp0 = 1
		end if
		if keyDown[0].right == 1
			object.xpos += object.yvel
			temp0 = 1
		end if
		if keyDown[0].up == 1
			object.ypos -= object.yvel
			temp0 = 1
		end if
		if keyDown[0].down == 1
			object.ypos += object.yvel
			temp0 = 1
		end if
		if temp0 == 1
			object.yvel += 3072
			if object.yvel > 1048576
				object.yvel = 1048576
			end if
		else
			object.yvel = 0
		end if
		object.value3 = 0
		object.value4 = 0
		if keyPress[0].buttonB == 1
			if DebugMode_unusedTimer == 0
				if stage.playerListPos == 1
					object.type = TypeName[Tails Object]
				else
					object.type = TypeName[Player Object]
				end if
				object.speed = 0
				object.xvel = 0
				object.yvel = 0
				object.state = 13
				object.animation = ANI_WALKING
				object.animationSpeed = 0
				object.frame = 0
				object.controlMode = 0
				object.drawOrder = -1
				object.drawOrder = 4
				object.interaction = 1
				object.tileCollisions = 1
				object.visible = 1
			end if
		else
			if keyPress[0].buttonA == 1
				object.value16++
				if object.value16 >= DebugMode_ObjCount
					object.value16 = 0
				end if
			end if
			if keyDown[0].buttonA == 1
				if keyPress[0].buttonC == 1
					object.value16--
					if object.value16 < 0
						object.value16 += DebugMode_ObjCount
					end if
				end if
			else
				if keyPress[0].buttonC == 1
					GetTableValue(temp0, object.value16, DebugMode_SpawnTable)
					CallFunction(temp0)
					object[tempObjectPos].priority = 4
				end if
			end if
		end if
	else
		// do whatever this is
		options.touchControls = 0
		DebugMode_unusedTimer += 8
		SetScreenFade(0, 0, 0, DebugMode_unusedTimer)
		
		// fun fact it never loads this stage, which is R12A lol.
		if DebugMode_unusedTimer == 384
			lampPostID = 0
			stage.activeList = PRESENTATION_STAGE
			stage.listPos = 5
			LoadStage()
		end if
	end if
end event


event ObjectDraw
	GetTableValue(object.type, object.value16, DebugMode_TypesTable)
	GetTableValue(temp0, object.value16, DebugMode_DrawTable)
	CallFunction(temp0)
	object.type = TypeName[Debug Mode]
end event


event ObjectStartup
	if stage.playerListPos == 0
		LoadSpriteSheet("Global/Display.gif")
	else
		LoadSpriteSheet("Global/Display_t.gif")
	end if
	SpriteFrame(-8, -8, 16, 16, 187, 189)
	DebugMode_ObjCount = 0
	temp0 = 0
	while temp0 < 256
		SetTableValue(TypeName[Debug Mode], temp0, DebugMode_TypesTable)
		SetTableValue(DebugMode_DrawBlank, temp0, DebugMode_DrawTable)
		SetTableValue(DebugMode_SpawnBlank, temp0, DebugMode_SpawnTable)
		temp0++
	loop

	// This disables its... special behavior
	DebugMode_unusedTimer = 0
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

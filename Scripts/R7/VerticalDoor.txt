private alias 55 : TYPE_VERTICALDOOR

// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	switch object.state
	case 1
		if object.value0 < 4194304
			object.value0 += 131072
			object.ypos -= 131072
		else
			object.state = 5
		end if
		break
	case 2
		if object.value0 > 0
			object.value0 -= 131072
			object.ypos += 131072
		else
			object.state = 6
		end if
		break
	case 3
		temp0 = object[0].collisionRight
		temp0 += 16
		temp0 <<= 16
		temp0 += object.xpos
		if object[0].xpos > temp0
			object.state = 2
			PlaySfx(SfxName[Door Close], 0)
		else
			if object[+2].xpos > object.xpos
				object.state = 4
				PlaySfx(SfxName[Door Close], 0)
			end if
		end if
		break
	case 4
		if object.value0 > 0
			object.value0 -= 131072
			object.ypos += 131072
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		switch object.state
		case 0
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -32, 16, 64, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			break
		case 1
		case 5
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -32, 16, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			break
		case 2
		case 6
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -32, 16, 128, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			temp0 = object[currentPlayer].collisionRight
			temp0 <<= 16
			temp0 += object[currentPlayer].xpos
			temp1 = stage.curXBoundary2
			temp1 <<= 16
			if temp0 > temp1
				object[currentPlayer].xvel = 0
				object[currentPlayer].speed = 0
				object[currentPlayer].xpos = temp1
				temp0 = object[currentPlayer].collisionRight
				temp0 <<= 16
				object[currentPlayer].xpos -= temp0
			end if
			break
		case 3
			BoxCollisionTest(C_SOLID, object.entityPos, -16, -32, 16, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			temp0 = object[currentPlayer].collisionRight
			temp0 <<= 16
			temp0 += object[currentPlayer].xpos
			temp1 = stage.curXBoundary2
			temp1 <<= 16
			if temp0 > temp1
				object[currentPlayer].xvel = 0
				object[currentPlayer].speed = 0
				object[currentPlayer].xpos = temp1
				temp0 = object[currentPlayer].collisionRight
				temp0 <<= 16
				object[currentPlayer].xpos -= temp0
			end if
			break
		case 4
			temp0 = object[currentPlayer].collisionRight
			temp0 <<= 16
			temp0 += object[currentPlayer].xpos
			temp1 = object.xpos
			temp1 -= 1048576
			if temp0 > temp1
				object[currentPlayer].xvel = 0
				object[currentPlayer].speed = 0
				object[currentPlayer].xpos = temp1
				temp0 = object[currentPlayer].collisionRight
				temp0 <<= 16
				object[currentPlayer].xpos -= temp0
			end if
			break
		end switch
	next
end event


event ObjectDraw
	temp0 = object.value0
	temp0 >>= 1
	temp0 += object.ypos
	DrawSpriteXY(0, object.xpos, temp0)
	DrawSprite(1)
end event


event ObjectStartup
	LoadSpriteSheet("R7/Objects.gif")
	if hud.currentTimePeriod == TIME_GOOD_FUTURE
		SpriteFrame(-16, -32, 32, 32, 117, 219)
		SpriteFrame(-16, 0, 32, 32, 117, 219)
	else
		SpriteFrame(-16, -32, 32, 32, 1, 1)
		SpriteFrame(-16, 0, 32, 32, 1, 1)
	end if
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

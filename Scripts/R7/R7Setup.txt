private value unusedValue1 = 0;
private value unusedValue2 = 0;


private function R7Setup_GetLoopTileInfo
	temp3 = player[currentPlayer].xpos
	temp3 >>= 16
	temp3 &= 255
	temp3 += 12544
	temp0 -= 16
	temp0 >>= 2
	temp0 <<= 8

	temp4 = player[currentPlayer].ypos
	temp4 >>= 16
	temp4 &= 255
	temp4 += temp0

	Get16x16TileInfo(temp0, temp3, temp4, CMODE_FLOOR)
end function


private function R7Setup_drawFunctionPresent
	object.timer++
	if object.timer == 5
		object.timer = 0
		object.currentPalette++
		object.currentPalette &= 1
		SetActivePalette(object.currentPalette, 0, 240)
	end if
end function


private function R7Setup_drawFunctionPast
	object.timer++
	if object.timer > 1
		tileLayer[1].deformationOffset++
		object.timer = 0
	end if
end function


private function R7Setup_drawFunctionGoodFuture
	// Nothing Lol
end function


private function R7Setup_drawFunctionBadFuture
	object.timer++
	if object.timer == 5
		object.timer = 0
		object.currentPalette++
		object.currentPalette %= 6
		SetActivePalette(object.currentPalette, 0, 240)
	end if
end function


private function R7Setup_drawFunctionAct3
	object.xpos = screen.xoffset
	object.xpos /= 420
	object.xpos *= 420
	object.xpos += 320
	object.xpos *= 3
	object.xpos >>= 1
	temp0 = screen.xoffset
	temp0 *= 3
	temp0 >>= 1
	object.xpos -= temp0
	object.ypos = 520

	temp0 = screen.yoffset
	temp0 *= 23
	temp0 >>= 4
	object.ypos -= temp0

	DrawSpriteScreenXY(0, object.xpos, object.ypos)
	DrawSpriteScreenXY(1, object.xpos, object.ypos)
	DrawSpriteScreenXY(2, object.xpos, object.ypos)
	DrawSpriteScreenXY(3, object.xpos, object.ypos)
end function


event ObjectUpdate
	CallFunction(object.drawFunction1)

	if object.flameTimer > 0
		object.flameTimer--
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		temp1 = player[currentPlayer].xpos
		temp1 >>= 23
		temp2 = player[currentPlayer].ypos
		temp2 >>= 23
		GetTileLayerEntry(temp0, 0, temp1, temp2)
		switch temp0
		case 16
		case 17
		case 18
		case 19
			CallFunction(R7Setup_GetLoopTileInfo)
			if player[currentPlayer].collisionMode == CMODE_FLOOR
				if temp0 == 1001
					temp1 &= 1022
					temp2 &= 1022
					SetTileLayerEntry(20, 0, temp1, temp2)
					temp1++
					SetTileLayerEntry(21, 0, temp1, temp2)
					temp1--
					temp2++
					SetTileLayerEntry(22, 0, temp1, temp2)
					temp1++
					SetTileLayerEntry(23, 0, temp1, temp2)
				end if
			else
				if player[currentPlayer].speed > 0
					if temp0 == 1003
						temp1 &= 1022
						temp2 &= 1022
						SetTileLayerEntry(20, 0, temp1, temp2)
						temp1++
						SetTileLayerEntry(21, 0, temp1, temp2)
						temp1--
						temp2++
						SetTileLayerEntry(22, 0, temp1, temp2)
						temp1++
						SetTileLayerEntry(23, 0, temp1, temp2)
					end if
				end if
			end if
			break

		case 20
		case 21
		case 22
		case 23
			CallFunction(R7Setup_GetLoopTileInfo)
			if player[currentPlayer].collisionMode == CMODE_FLOOR
				if temp0 == 1000
					temp1 &= 1022
					temp2 &= 1022
					SetTileLayerEntry(16, 0, temp1, temp2)
					temp1++
					SetTileLayerEntry(17, 0, temp1, temp2)
					temp1--
					temp2++
					SetTileLayerEntry(18, 0, temp1, temp2)
					temp1++
					SetTileLayerEntry(19, 0, temp1, temp2)
				end if
			else
				if player[currentPlayer].speed < 0
					if temp0 == 1002
						temp1 &= 1022
						temp2 &= 1022
						SetTileLayerEntry(16, 0, temp1, temp2)
						temp1++
						SetTileLayerEntry(17, 0, temp1, temp2)
						temp1--
						temp2++
						SetTileLayerEntry(18, 0, temp1, temp2)
						temp1++
						SetTileLayerEntry(19, 0, temp1, temp2)
					end if
				end if
			end if
			break

		case 24
		case 25
		case 26
		case 27
			CallFunction(R7Setup_GetLoopTileInfo)
			if temp0 == 1001
				temp1 &= 1022
				temp2 &= 1022
				SetTileLayerEntry(28, 0, temp1, temp2)
				temp1++
				SetTileLayerEntry(29, 0, temp1, temp2)
				temp1--
				temp2++
				SetTileLayerEntry(30, 0, temp1, temp2)
				temp1++
				SetTileLayerEntry(31, 0, temp1, temp2)
			end if
			break

		case 28
		case 29
		case 30
		case 31
			CallFunction(R7Setup_GetLoopTileInfo)
			if temp0 == 1000
				temp1 &= 1022
				temp2 &= 1022
				SetTileLayerEntry(24, 0, temp1, temp2)
				temp1++
				SetTileLayerEntry(25, 0, temp1, temp2)
				temp1--
				temp2++
				SetTileLayerEntry(26, 0, temp1, temp2)
				temp1++
				SetTileLayerEntry(27, 0, temp1, temp2)
			end if
			break

		end switch
		
		if object.flameTimer == 0

			temp0 = player[currentPlayer].speed
			if temp0 < 0
				FlipSign(temp0)
			end if

			if temp0 > 0xA 0000
				temp3 = player[currentPlayer].xpos
				temp3 >>= 16
				
				temp4 = player[currentPlayer].ypos
				temp4 >>= 16
				temp4 += 8

				Get16x16TileInfo(temp0, temp3, temp4, TILEINFO_ANGLEB)
				if temp0 == 1
					object.flameTimer = 4

					CreateTempObject(TypeName[Fire Trail], 0, player[currentPlayer].xpos, player[currentPlayer].ypos)
					if player[currentPlayer].speed < 0
						object[tempObjectPos].direction = 1
					else
						object[tempObjectPos].direction = 0
					end if
					object[tempObjectPos].drawOrder = 4
					object[tempObjectPos].iypos += player[currentPlayer].collisionBottom
				end if
			end if
		end if
	next
end event


event ObjectDraw
	CallFunction(object.drawFunction2)
end event


event ObjectStartup
	object[21].type = TypeName[R7 Setup]
	object[21].priority = 1
	object[21].drawFunction2 = R7Setup_drawFunctionGoodFuture // 86
	
	switch hud.currentTimePeriod
	case TIME_PRESENT
		object[21].drawFunction1 = R7Setup_drawFunctionPresent // 84

		CopyPalette(0, 0, 1, 0, 255)
		CopyPalette(0, 0, 2, 0, 255)
		CopyPalette(0, 0, 3, 0, 255)
		CopyPalette(0, 0, 4, 0, 255)
		CopyPalette(0, 0, 5, 0, 255)

		temp0 = 0
		temp1 = 0
		temp2 = 32
		while temp0 < 2
			LoadPalette("R7A_PalCycle.act", temp0, 160, temp1, temp2)
			temp0++
			temp1 += 32
			temp2 += 32
		loop
		break

	case TIME_PAST
		object[21].drawFunction1 = R7Setup_drawFunctionPast // 85

		SetLayerDeformation(DEFORM_BG, 128, 8, 0, 0, 0)
		break

	case TIME_GOOD_FUTURE
		object[21].drawFunction1 = R7Setup_drawFunctionGoodFuture // 86

		if stage.actNum == 3
			object[21].drawOrder = 5
			object[21].drawFunction2 = R7Setup_drawFunctionAct3 // 88
			
			LoadSpriteSheet("R7/Objects2.gif")
			SpriteFrame(-24, -24, 48, 24, 207, 1)
			SpriteFrame(-8, 0, 16, 224, 239, 26)
			SpriteFrame(-8, 224, 16, 224, 239, 26)
			SpriteFrame(-8, 448, 16, 224, 239, 26)
			stage.curXBoundary2 -= 288
		end if
		break

	case TIME_BAD_FUTURE
		object[21].drawFunction1 = R7Setup_drawFunctionBadFuture // 87

		CopyPalette(0, 0, 1, 0, 255)
		CopyPalette(0, 0, 2, 0, 255)
		CopyPalette(0, 0, 3, 0, 255)
		CopyPalette(0, 0, 4, 0, 255)
		CopyPalette(0, 0, 5, 0, 255)

		temp0 = 0
		temp1 = 0
		temp2 = 16
		while temp0 < 6
			LoadPalette("R7D_PalCycle.act", temp0, 160, temp1, temp2)
			temp0++
			temp1 += 16
			temp2 += 16
		loop

		if stage.actNum == 3
			object[21].drawOrder = 5
			object[21].drawFunction2 = R7Setup_drawFunctionAct3 // 88
			LoadSpriteSheet("R7/Objects2.gif")
			SpriteFrame(-24, -24, 48, 24, 173, 26)
			SpriteFrame(-8, 0, 16, 224, 222, 26)
			SpriteFrame(-8, 224, 16, 224, 222, 26)
			SpriteFrame(-8, 448, 16, 224, 222, 26)
			stage.curXBoundary2 -= 288
		end if
		break
	end switch
end event

event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			switch object.propertyValue
			case 0
				if object.value1 == 0
					if object[currentPlayer].state != Player_State_TubeAirRoll
						object[currentPlayer].state = Player_State_TubeAirRoll
						object[currentPlayer].gravity = 0
						object[currentPlayer].value43 = 0
						object[currentPlayer].animation = ANI_JUMPING
						object[currentPlayer].value17 = 851968
					else
						object[currentPlayer].state = Player_State_Roll
						object[currentPlayer].animation = ANI_JUMPING
					end if
				end if
				break
			case 1
				object[currentPlayer].state = Player_State_Air
				object[currentPlayer].animation = ANI_JUMPING
				if object[currentPlayer].gravity == 1
					if object[currentPlayer].yvel < 0
						object[currentPlayer].yvel = -851968
					else
						object[currentPlayer].yvel = 851968
					end if
				else
					switch object[currentPlayer].collisionMode
					case 1
						if object[currentPlayer].speed < 0
							object[currentPlayer].yvel = 851968
						else
							object[currentPlayer].yvel = -851968
						end if
						break
					case 3
						if object[currentPlayer].speed < 0
							object[currentPlayer].yvel = -851968
						else
							object[currentPlayer].yvel = 851968
						end if
						break
					end switch
				end if
				object[currentPlayer].gravity = 1
				object[currentPlayer].xpos = object.xpos
				object[currentPlayer].xvel = 0
				object[currentPlayer].speed = 0
				break
			case 2
				if object.value1 == 0
					if object[currentPlayer].state != Player_State_TubeAirRoll
						object[currentPlayer].xpos = object.xpos
						object[currentPlayer].state = Player_State_TubeAirRoll
						object[currentPlayer].animation = ANI_JUMPING
						object[currentPlayer].value17 = 851968
						object[currentPlayer].collisionMode = 3
						object[currentPlayer].angle = 64
						object[currentPlayer].gravity = 0
						object[currentPlayer].value43 = 0
						object[currentPlayer].xvel = 0
						if object[currentPlayer].yvel < 0
							object[currentPlayer].speed = -851968
						else
							object[currentPlayer].speed = 851968
						end if
						object[currentPlayer].yvel = 0
						object[currentPlayer].right = 0
						object[currentPlayer].left = 0
					else
						object[currentPlayer].state = Player_State_Air
						object[currentPlayer].animation = ANI_JUMPING
						switch object[currentPlayer].collisionMode
						case 1
							if object[currentPlayer].speed < 0
								object[currentPlayer].yvel = 851968
							else
								object[currentPlayer].yvel = -851968
							end if
							break
						case 3
							if object[currentPlayer].speed < 0
								object[currentPlayer].yvel = -851968
							else
								object[currentPlayer].yvel = 851968
							end if
							break
						end switch
						object[currentPlayer].gravity = 1
						object[currentPlayer].xpos = object.xpos
						object[currentPlayer].xvel = 0
						object[currentPlayer].speed = 0
					end if
				end if
				break
			case 3
				if object.value1 == 0
					if object[currentPlayer].collisionMode == 2
						object[currentPlayer].collisionMode = 0
						object[currentPlayer].angle = 0
						FlipSign(object[currentPlayer].speed)
						FlipSign(object[currentPlayer].xvel)
					end if
				end if
				break
			case 4
				if object.value1 == 0
					object[currentPlayer].state = Player_State_Roll
					object[currentPlayer].animation = ANI_JUMPING
					if object[currentPlayer].xpos < object.xpos
						if object[currentPlayer].speed < 655360
							object[currentPlayer].speed = 655360
						end if
						object[currentPlayer].direction = 0
					else
						if object[currentPlayer].speed > -655360
							object[currentPlayer].speed = -655360
						end if
						object[currentPlayer].direction = 1
					end if
				end if
				break
			case 5
				if object[currentPlayer].yvel < 0
					if object[currentPlayer].state != Player_State_TubeAirRoll
						if object[currentPlayer].left == true
							object[currentPlayer].state = Player_State_TubeAirRoll
							object[currentPlayer].gravity = 0
							object[currentPlayer].value43 = 0
							object[currentPlayer].collisionMode = 3
							object[currentPlayer].angle = 64
							object[currentPlayer].animation = ANI_JUMPING
							object[currentPlayer].value17 = 851968
							object[currentPlayer].speed = -851968
							object[currentPlayer].xvel = 0
							object[currentPlayer].yvel = 0
							object[currentPlayer].xpos -= 524288
						end if
						if object[currentPlayer].right == true
							object[currentPlayer].state = Player_State_TubeAirRoll
							object[currentPlayer].gravity = 0
							object[currentPlayer].value43 = 0
							object[currentPlayer].collisionMode = 1
							object[currentPlayer].angle = 192
							object[currentPlayer].animation = ANI_JUMPING
							object[currentPlayer].value17 = 851968
							object[currentPlayer].speed = 851968
							object[currentPlayer].xvel = 0
							object[currentPlayer].yvel = 0
							object[currentPlayer].xpos += 524288
						end if
					end if
				end if
				break
			case 6
				if object.value1 == 0
					if object[currentPlayer].state == Player_State_Air
						object[currentPlayer].xpos = object.xpos
						object[currentPlayer].xvel = 0
						object[currentPlayer].speed = 0
					end if
				end if
				break
			case 7
				if object.value1 == 0
					if object[currentPlayer].collisionMode == 2
						object[currentPlayer].collisionMode = 0
						object[currentPlayer].angle += 128
						FlipSign(object[currentPlayer].speed)
						FlipSign(object[currentPlayer].xvel)
					end if
				end if
				break
			end switch
			object.value1 = 1
		else
			object.value1 = 0
		end if
	next
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

		DrawSpriteFX(object.propertyValue, FX_FLIP, object.xpos, object.ypos)
	else
		object.value0++
		temp0 = object.value0
		temp0 >>= 1
		switch temp0
		case 0
		case 1
			DrawSpriteFX(object.propertyValue, FX_FLIP, object.xpos, object.ypos)
			break
		case 2
		case 6
			temp1 = object.propertyValue
			temp1 += 8
			DrawSpriteFX(temp1, FX_FLIP, object.xpos, object.ypos)
			break
		case 3
		case 4
		case 5
			temp1 = object.propertyValue
			temp1 += 16
			DrawSpriteFX(temp1, FX_FLIP, object.xpos, object.ypos)
			break
		end switch
		if object.value0 > 12
			object.value0 = 0
		end if
	end if
end event

private alias 65 : TYPE_LIGHTNING



event ObjectUpdate
	switch object.state
	case 0
		if object.value0 < 256
			object.value0++
		else
			object.value0 = 0
			object.alpha = 254
			object.value1 = 128
			object.state++
		end if
		break
	case 1
		if object.alpha > 0
			object.alpha -= 2
		else
			object.state = 0
			object.frame++
			object.frame &= 3
		end if
		if object.value1 > 0
			object.value1 -= 4
		end if
		break
	end switch
end event


event ObjectDraw
	if object.state > 0
		if object.drawOrder == 1
			object.ypos = screen.yoffset
			object.ypos *= object.propertyValue
			object.ypos >>= 7
			FlipSign(object.ypos)
			object.ypos += 312
			temp0 = object.value1
			temp0 >>= 1
			DrawRect(0, 0, screen.xsize, screen.ysize, 224, 192, 128, temp0)
			temp0 = screen.xoffset
			temp0 >>= 3
			temp0 &= 511
			FlipSign(temp0)
			temp1 = object.frame
			temp0 += 64
			if temp0 < -64
				temp0 += 512
				DrawSpriteScreenFX(temp1, FX_INK, temp0, object.ypos)
				temp0 -= 512
			else
				DrawSpriteScreenFX(temp1, FX_INK, temp0, object.ypos)
			end if
			temp1++
			temp1 &= 3
			temp0 += 128
			if temp0 < -64
				temp0 += 512
				DrawSpriteScreenFX(temp1, FX_INK, temp0, object.ypos)
				temp0 -= 512
			else
				DrawSpriteScreenFX(temp1, FX_INK, temp0, object.ypos)
			end if
			temp1++
			temp1 &= 3
			temp0 += 128
			if temp0 < -64
				temp0 += 512
				DrawSpriteScreenFX(temp1, FX_INK, temp0, object.ypos)
				temp0 -= 512
			else
				DrawSpriteScreenFX(temp1, FX_INK, temp0, object.ypos)
			end if
			temp1++
			temp1 &= 3
			temp0 += 96
			if temp0 < -64
				temp0 += 512
				DrawSpriteScreenFX(temp1, FX_INK, temp0, object.ypos)
				temp0 -= 512
			else
				DrawSpriteScreenFX(temp1, FX_INK, temp0, object.ypos)
			end if
			SetDrawListEntityRef(object.entityPos, 5, screen[5].drawListSize)
			screen[5].drawListSize++
			object.drawOrder = 5
		else
			DrawRect(0, 0, screen.xsize, screen.ysize, 224, 224, 192, object.value1)
			object.drawOrder = 1
		end if
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R7/Objects3.gif")
	SpriteFrame(-8, 0, 16, 40, 43, 140)
	SpriteFrame(-8, 0, 16, 24, 60, 140)
	SpriteFrame(-16, 0, 32, 40, 77, 132)
	SpriteFrame(-8, 0, 16, 24, 60, 140)
	foreach (TypeName[Lightning], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].priority = 1
		object[arrayPos0].drawOrder = 1
		object[arrayPos0].inkEffect = 2
	next
end event

event RSDKDraw
	DrawSprite(0)
end event

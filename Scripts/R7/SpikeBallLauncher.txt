event ObjectUpdate
	switch object.state
	case 0
		object.priority = 1
		if object.value0 < 150
			object.value0++
		else
			object.value0 = 0
			object.value1 = 0
			object.state++
		end if
		break
	case 1
		if object.value0 < 32
			object.value0++
			object.value1 -= 32768
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 2
		if object.value0 < 30
			object.value0++
		else
			object.value0 = 0
			object.value2 = -393216
			object.state++
		end if
		break
	case 3
		if object.value1 < 0
			object.value2 += 16384
			object.value1 += object.value2
		else
			object.state++
		end if
		break
	case 4
		if object.value0 < 40
			object.value0++
		else
			object.value0 = 0
			object.value1 = 0
			object.state = 1
		end if
		break
	end switch
	if object.outOfBounds == 1
		object.state = 0
		object.value0 = 0
		object.priority = 0
	end if
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		switch object.state
		case 1
		case 2
		case 3
			temp0 = object.xpos
			temp1 = object.ypos
			switch object.propertyValue
			case 0
				object.ypos += object.value1
				break
			case 1
				object.ypos -= object.value1
				break
			case 2
				object.xpos += object.value1
				break
			case 3
				object.xpos -= object.value1
				break
			end switch
			BoxCollisionTest(C_TOUCH, object.entityPos, -12, -12, 12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				CallFunction(Player_Hit)
			end if
			object.xpos = temp0
			object.ypos = temp1
		end switch
	next
end event


event ObjectDraw
	if object.drawOrder == 3
		switch object.state
		case 1
		case 2
		case 3
			switch object.propertyValue
			case 0
				temp0 = object.ypos
				temp0 += object.value1
				DrawSpriteXY(4, object.xpos, temp0)
				break
			case 1
				temp0 = object.ypos
				temp0 -= object.value1
				DrawSpriteXY(4, object.xpos, temp0)
				break
			case 2
				temp0 = object.xpos
				temp0 += object.value1
				DrawSpriteXY(4, temp0, object.ypos)
				break
			case 3
				temp0 = object.xpos
				temp0 -= object.value1
				DrawSpriteXY(4, temp0, object.ypos)
				break
			end switch
		end switch
		SetDrawListEntityRef(object.entityPos, 4, screen[4].drawListSize)
		screen[4].drawListSize++
		object.drawOrder = 4
	else
		DrawSprite(object.propertyValue)
		object.drawOrder = 3
	end if
end event

	SpriteFrame(-16, -16, 32, 32, 1, 199)
end event

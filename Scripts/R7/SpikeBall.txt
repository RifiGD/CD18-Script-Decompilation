// Time Period Aliases
private alias object[9].propertyValue : hud.currentTimePeriod

private alias 0 : TIME_PRESENT
private alias 1 : TIME_PAST
private alias 2 : TIME_GOOD_FUTURE
private alias 3 : TIME_BAD_FUTURE

event ObjectUpdate
	switch object.state
	case 0
		if object.value1 < object.value4
			object.value1++
			object.xpos += object.value2
		else
			object.state++
		end if
		break
	case 1
		if object.value0 < object.value6
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 2
		if object.value1 > object.value5
			object.value1--
			object.xpos -= object.value2
		else
			object.state++
		end if
		break
	case 3
		if object.value0 < object.value6
			object.value0++
		else
			object.value0 = 0
			object.state = 0
		end if
		break
	case 4
		if object.value1 < object.value4
			object.value1++
			object.ypos += object.value2
		else
			object.state++
		end if
		break
	case 5
		if object.value0 < object.value6
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 6
		if object.value1 > object.value5
			object.value1--
			object.ypos -= object.value2
		else
			object.state++
		end if
		break
	case 7
		if object.value0 < object.value6
			object.value0++
		else
			object.value0 = 0
			object.state = 4
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -12, -12, 12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			CallFunction(Player_Hit)
		end if
	next
end event


event ObjectDraw
	DrawSprite(0)
end event

	foreach (TypeName[Spike Ball], arrayPos0, ALL_ENTITIES)
		if object[arrayPos0].propertyValue == 1
			object[arrayPos0].state = 4
			switch hud.currentTimePeriod
			case TIME_PAST
				object[arrayPos0].value2 = 131072
				object[arrayPos0].value4 = 16
				object[arrayPos0].value5 = -16
				object[arrayPos0].value6 = 40

				object[arrayPos0].value6 = 60
				break
			case TIME_GOOD_FUTURE
			case TIME_BAD_FUTURE
				object[arrayPos0].value2 = 65536
				object[arrayPos0].value4 = 32
				object[arrayPos0].value5 = -32
				object[arrayPos0].value6 = 20
				break
			end switch
		else
			switch hud.currentTimePeriod
			case TIME_PRESENT
				object[arrayPos0].value2 = 131072
				object[arrayPos0].value4 = 24
				object[arrayPos0].value5 = -24

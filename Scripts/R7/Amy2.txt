	else
		object.direction = 0
	end if

			end if
			break

		case 7

			end if

							object.state = 1

							object.value0 = 0
							object.xvel = 0
						end if
					else
					end if
				end if
			end if
			if object.xpos < object.value4
				object.xpos += object.xvel
			else
				object.xvel = 0
				object.xpos = object.value4
				object.state = 4
				object.frame = 0
				object.value0 = 0
			end if
		else
			if object.xvel > -131072
				object.xvel -= 8192
			end if
			BoxCollisionTest(C_TOUCH, object.entityPos, -4, -360, 32, 360, 0, C_BOX, C_BOX, C_BOX, C_BOX)

				if object[0].gravity == 0
					if object[0].direction == object.direction

							object.value0 = 0

					else
					end if

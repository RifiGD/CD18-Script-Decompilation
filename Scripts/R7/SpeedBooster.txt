event ObjectUpdate
	switch object.state
	case 1
		if object.value0 < 6
			object.value1 -= 524288
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 2
		if object.value0 < 6
			object.value1 += 524288
			object.value0++
		else
			object.value0 = 0
			object.state = 0
			object.priority = 0
		end if
		break
	case 3
		if object.value0 < 6
			object.value1 += 524288
			object.value0++
		else
			object.value0 = 0
			object.state++
		end if
		break
	case 4
		if object.value0 < 6
			object.value1 -= 524288
			object.value0++
		else
			object.value0 = 0
			object.state = 0
			object.priority = 0
		end if
		break
	end switch
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state == 0
			if object[currentPlayer].state != Player_State_Air
				BoxCollisionTest(C_TOUCH, object.entityPos, -32, -12, 32, 4, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object.priority = 1
					PlaySfx(SfxName[Large Booster], 0)
					if object[currentPlayer].speed > 0
						object.state = 1
						object[currentPlayer].speed = 1048576
						object[currentPlayer].direction = 0
						CreateTempObject(TypeName[Fire Trail], 0, object.xpos, object.ypos)
						object[tempObjectPos].xpos -= 1572864
						object[tempObjectPos].ypos -= 393216
						CreateTempObject(TypeName[Blank Object], 0, object.xpos, object.ypos)
						object[tempObjectPos].xvel = -131072
						object[tempObjectPos].yvel = -458752
					else
						object.state = 3
						object[currentPlayer].speed = -1048576
						object[currentPlayer].direction = 1
						CreateTempObject(TypeName[Fire Trail], 0, object.xpos, object.ypos)
						object[tempObjectPos].xpos += 1572864
						object[tempObjectPos].ypos -= 393216
						object[tempObjectPos].direction = 1
						CreateTempObject(TypeName[Blank Object], 0, object.xpos, object.ypos)
						object[tempObjectPos].xvel = 131072
						object[tempObjectPos].yvel = -458752
						object[tempObjectPos].direction = 1
					end if
				end if
			end if
		end if
	next
end event


event ObjectDraw
	temp0 = object.value1
	temp0 += object.xpos
	DrawSpriteXY(0, temp0, object.ypos)
end event


event ObjectStartup
	LoadSpriteSheet("R7/Objects.gif")
	SpriteFrame(-32, -12, 64, 24, 34, 96)
	foreach (TypeName[Speed Booster], arrayPos0, ALL_ENTITIES)
		object[arrayPos0].drawOrder = 4
	next
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

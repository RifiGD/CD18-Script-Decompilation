private alias object.value3 : object.boundsLeft
private alias object.value4 : object.boundsRight

private alias 0 : Amy1_State_Caught
private alias 1 : Amy1_State_Falling

// Collision Sides
private alias 0  : CSIDE_FLOOR

// Game Mode Aliases
private alias 2 : MODE_TIMEATTACK

event ObjectUpdate
	if object.state == Amy1_State_Falling
		object.ypos += object.yvel
		object.yvel += 0x2000
		ObjectTileCollision(CSIDE_FLOOR, 0, 19, 0)
		if checkResult == true
			ResetObjectEntity(object.entityPos, TypeName[Amy 2], 0, object.xpos, object.ypos)
			object.frame = 4
			object.drawOrder = 4
			object.boundsLeft = 0x3EA4 0000
			object.boundsRight = 0x4000 0000
		end if
	end if

	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		if object.state == Amy1_State_Caught
			if object.type == TypeName[Amy 1]
				BoxCollisionTest(C_TOUCH, object.entityPos, -8, -16, 8, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					FlipSign(object[currentPlayer].yvel)
					object[currentPlayer].yvel >>= 1
					object.state = 1
					object.frame >>= 3
					CreateTempObject(TypeName[Explosion], 0, object.xpos, object.ypos)
					stage.timeEnabled = 0
					PlaySfx(SfxName[Explosion], 0)
				end if
			end if
		end if
	next
end event


event ObjectDraw
	if object.state == Amy1_State_Caught
		temp0 = object.frame
		temp0 >>= 3
		DrawSprite(temp0)

		object.frame++
		object.frame &= 31
	else
		DrawSprite(object.frame)
	end if
end event


event ObjectStartup
	LoadSpriteSheet("R7/Objects2.gif")

	foreach (TypeName[Amy 1], arrayPos0, ALL_ENTITIES)
		if stage.playerListPos == PLAYER_SONIC
			if options.gameMode == MODE_TIMEATTACK
				object[arrayPos0].type = TypeName[Blank Object]
			else
				object[arrayPos0].drawOrder = 4
			end if
		else
			object[arrayPos0].type = TypeName[Blank Object]
		end if
	next

	SpriteFrame(-12, -20, 24, 40, 1, 170)
	SpriteFrame(-12, -20, 24, 40, 1, 211)
	SpriteFrame(-12, -20, 24, 40, 124, 109)
	SpriteFrame(-12, -20, 24, 40, 1, 211)
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

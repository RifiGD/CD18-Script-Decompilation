//-------Aliases-------//

private alias 54 : TYPE_BOUNDARYADJUST



event ObjectUpdate
	if object.propertyValue == 0
		if object[0].xpos > object.xpos
			stage.newYBoundary2 = 520
		else
			stage.newYBoundary2 = 768
		end if
	else
		if object[0].xpos < object.xpos
			stage.newYBoundary2 = 520
		else
			stage.newYBoundary2 = 768
		end if
	end if
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event

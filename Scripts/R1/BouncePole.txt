event ObjectUpdate
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -8, 16, 8, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			if object[currentPlayer].gravity == 1
				if object[currentPlayer].yvel > 0
					object[currentPlayer].state = Player_State_Air
					object[currentPlayer].value1 = 0
					object[currentPlayer].animation = ANI_JUMPING
					FlipSign(object[currentPlayer].yvel)
					object[currentPlayer].yvel -= 131072
					if object[currentPlayer].yvel < -655360
						object[currentPlayer].yvel = -655360
					end if
					object.frame = 1
				end if
			end if
		end if
		if object.frame > 0
			object.value0++
			if object.value0 == 3
				object.value0 = 0
				object.frame++
				if object.frame > 20
					object.frame = 0
				end if
			end if
		end if
	next
end event

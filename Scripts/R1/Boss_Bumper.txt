event ObjectUpdate
	switch object.state
	case 0
		object.state = 1
	case 1
		arrayPos0 = object.value0
		object.xpos = object[arrayPos0].xpos
		object.xpos -= 1245184
		object.ypos = object[arrayPos0].ypos
		if object[arrayPos0].frame != 0
			if object[arrayPos0].frame == 1
				object.xpos += 524288
			else
				object.xpos += 1048576
			end if
		end if
		break
	case 2
		object.xvel -= 1280
		object.yvel += 6144
		object.xpos += object.xvel
		object.ypos += object.yvel
		if object.outOfBounds == 1
			object.type = TypeName[Blank Object]
			object.priority = 0
		end if
		break
	end switch
	if object.state < 2
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			if object[currentPlayer].animation != ANI_HURT
				arrayPos0 = object.value0
				if object[arrayPos0].frame == 0
					BoxCollisionTest(C_TOUCH, object.entityPos, -10, -16, 14, 16, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
					if checkResult == true
						object[currentPlayer].speed = -262144
						object[currentPlayer].xvel = -262144
						if object.frame == 0
							PlaySfx(SfxName[Spring], 0)
						end if
						object.frame = 2
					end if
				end if
			end if
		next
	end if
end event


event ObjectDraw
	DrawSprite(object.frame)
	if object.frame > 0
		object.frame--
	end if
end event

end event

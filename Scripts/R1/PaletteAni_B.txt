event ObjectUpdate
	object.timer++
	if object.timer == 8
		object.timer = 0
		object.currentPalette++
		object.currentPalette %= 3
		RotatePalette(0, 177, 179, 1)
		RotatePalette(0, 172, 174, 0)
	end if
end event

end event

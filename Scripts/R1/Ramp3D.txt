	switch object.propertyValue
	case RAMP3D_RUN
		if object[0].gravity == 0
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				if object[0].state != Player_State_Static
					object[0].speed = 0
					object[0].xvel = 0
					object[0].animationSpeed = 0
					object[0].scrollTracking = true
					object[0].yvel = -786432
					object[0].state = Player_State_Static
					object[0].tileCollisions = 0
				end if
			end if
		end if
		break
	case RAMP3D_EXIT_UNUSED
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[0].state = Player_State_Air
			object[0].tileCollisions = true
			object[0].speed = 524288
			object[0].gravity = 1
			object[0].xvel = 524288
			object[0].yvel = -917504
			object[0].rotation = 384
			stage.newYBoundary2 = 1280
			stage.newXBoundary1 = 2080
			if object[0].animation == ANI_ROLL3D
				object[0].animation = ANI_JUMPING
			else
				if object[0].animation != ANI_JUMPING
					object[0].animation = ANI_WALKING
				end if
			end if
			object[15].type = TypeName[Background FX]
			object[15].priority = 1
			object[15].drawOrder = 0
			arrayPos0 = 0
			while arrayPos0 < 2
				hParallax[arrayPos0].scrollPos = 0
				arrayPos0++
			loop
		end if
		break
	case RAMP3D_EXIT
		if object[0].state == Player_State_Static
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object[0].state = Player_State_Air
				object[0].tileCollisions = true
				object[0].speed = 262144
				object[0].gravity = 1
				object[0].xvel = 262144
				object[0].yvel = -786432
				object[0].rotation = 384
				stage.newYBoundary2 = 1280
				stage.deathBoundary = 83886080
				if object[0].animation == ANI_ROLL3D
					object[0].animation = ANI_JUMPING
				else
					if object[0].animation != ANI_JUMPING
						object[0].animation = ANI_WALKING
					end if
				end if
			end if
		end if
		break
	case RAMP3D_SET_YBOUND
	// Unlike v3, this script actually sets a new death boundary
		if object[0].xpos > object.xpos
			stage.curYBoundary2 = 1024
			stage.deathBoundary = 67108864
		else
			stage.curYBoundary2 = 1280
			stage.deathBoundary = 83886080
		end if
		break
	case RAMP3D_ENTER
		if object[0].xpos > object.xpos
			if object[0].gravity == 0
				if object[0].state != Player_State_Static
					if object[0].state == Player_State_Roll
						if object[0].angle > 216
							object[0].animation = ANI_JUMPING
						else
							if object[0].angle == 0
								object[0].animation = ANI_JUMPING
							else
								object[0].animation = ANI_ROLL3D
							end if
						end if
					else
						if object[0].speed != 0
							object[0].state = Player_State_Ramp3D
						end if
					end if
				end if
			end if
			object[+3].value0 = object[0].xpos
			object[+3].value0 -= object.xpos
			object[+5].value0 = object[+3].value0
			object[+5].value0 /= 3
			object[+3].value0 >>= 1
			object[+4].value0 = object[+3].value0
			object[+3].value0 += object[+3].xpos
			object[+4].value0 += object[+4].xpos
			object[+5].value0 += object[+5].xpos
			temp0 = object.ypos
			temp0 -= object[0].ypos
			if temp0 < 14680064
				temp0 = object[0].xpos
				temp0 -= object.value1
				hParallax[0].scrollPos += temp0
				arrayPos0 = 13
				temp2 = 128
				while arrayPos0 < 141
					temp1 = temp0
					temp1 *= temp2
					temp1 >>= 7
					hParallax[arrayPos0].scrollPos += temp1
					arrayPos0++
					temp2--
				loop
				temp0 <<= 1
				hParallax[1].scrollPos += temp0
			end if
		else
			object[+3].value0 = object[+3].xpos
			object[+4].value0 = object[+4].xpos
			object[+5].value0 = object[+5].xpos
			if object[0].gravity == 0
				if object[0].state == Player_State_Ramp3D
					object[0].state = 12
				end if
			end if
		end if
		object.value1 = object[0].xpos
		break
	case RAMP3D_DROP
		if object[0].state == Player_State_Air
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object[0].animation = ANI_DROPPING
				object.value1 = object[0].xpos
			end if
			if object[0].animation == ANI_DROPPING
				object[0].speed = 0
				object[0].xvel = 0
				object[0].xpos = object.value1
			end if
		end if
		break
	case RAMP3D_SET_YBOUND2
		if object[0].ypos < object.ypos
			if object[0].xpos > object.xpos

				stage.deathBoundary = 51904512
			else
				stage.curYBoundary2 = 1024
				stage.deathBoundary = 67108864
			end if
		end if
		break
	case RAMP3D_RUN_FROM_TOP
		if object[0].xpos > object.xpos
			stage.curYBoundary2 = 1024
			if object[0].gravity == 0
				if object[0].state == Player_State_Ramp3D
					object[0].state = 12
				end if
			end if
		else
			stage.curYBoundary2 = 792
			stage.deathBoundary = 51904512
		end if
		break
	case RAMP3D_EXIT_BOOST
		BoxCollisionTest(C_TOUCH, object.entityPos, -16, -16, 16, 16, 0, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == true
			object[0].value1 = 0
			object[0].state = 13
			object[0].tileCollisions = 1
			object[0].speed = 524288
			object[0].gravity = 1
			object[0].xvel = 524288
			object[0].yvel = -917504
			object[0].rotation = 384
			stage.newYBoundary2 = 1280
			stage.deathBoundary = 83886080
			stage.newXBoundary1 = 2080
			if object[0].animation == ANI_ROLL3D
				object[0].animation = ANI_JUMPING
			else
				if object[0].animation != ANI_JUMPING
					object[0].animation = ANI_WALKING
				end if
			end if
			object[15].type = TypeName[Background FX]
			object[15].priority = 1
			object[15].drawOrder = 0
			arrayPos0 = 0
			while arrayPos0 < 2
				hParallax[arrayPos0].scrollPos = 0
				arrayPos0++
			loop
		end if
		if object[0].yvel < 0
			object[0].controlMode = -1
			object[0].left = 0
			object[0].right = 0
			object.priority = 1
		else
			object[0].controlMode = 0
			object.priority = 0
		end if
		break
	end switch
end event

// Function declarations
reserve function SmallPlatform_DebugDraw
reserve function SmallPlatform_DebugSpawn


public function SmallPlatform_DebugDraw
	DrawSprite(0)
end function


public function SmallPlatform_DebugSpawn
	CreateTempObject(TypeName[Small Platform], 0, object.xpos, object.ypos)
end function


event ObjectUpdate
	if object.value0 == 1
		if object.value1 < 16
			object.value1++
			temp0 = object.value1
			temp0 &= 3
			if temp0 == 3
				object.value2 = 65536
			else
				object.value2 = 0
			end if
		else
			object.value2 = 0
		end if
	else
		if object.value1 > 0
			object.value1--
			temp0 = object.value1
			temp0 &= 3
			if temp0 == 3
				object.value2 = -65536
			else
				object.value2 = 0
			end if
		else
			object.value2 = 0
		end if
	end if
	object.value0 = 0
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -16, -9, 16, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			object.value0 = 1
			object[currentPlayer].ypos += object.value2
		end if
	next
	object.ypos += object.value2
end event


event ObjectDraw
	DrawSprite(0)
end event

	SetTableValue(SmallPlatform_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

event ObjectUpdate
	if object.state < 2
		if object.state == 0
			if object.value0 == 1
				if object.frame < 24
					object.frame++
				end if
			else
				if object.frame > 0
					object.frame--
				end if
			end if
		end if
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -12, -12, 12, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			object.value0 = checkResult
		next
	end if
	if hud.currentTimePeriod > 1
		switch object.state
		case 0
			if object.value0 == 1
				object.state = 1
				object.priority = 1
				object[0].ypos += 131072
			end if
			break
		case 1
			if object.value0 == 1
				object[0].ypos += 131072
			end if
			object.ypos += 131072
			if object.outOfBounds == 1
				object.state = 2
				object.ypos = object.value1
			end if
			break
		case 2
			if object.outOfBounds == 1
				object.state = 0
				object.priority = 0
			end if
			break
		end switch
	end if
end event


event ObjectDraw
	switch hud.currentTimePeriod
	case TIME_PRESENT
		DrawSprite(object.frame)
		break
	case TIME_PAST
		DrawSprite(5)
		break
	case TIME_GOOD_FUTURE
	case TIME_BAD_FUTURE
		if object.state < 2
			DrawSprite(5)
		end if
		break
	end switch
end event

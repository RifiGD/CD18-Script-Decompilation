reserve function FallPlatform_DebugDraw

public function FallPlatform_DebugDraw
	DrawSprite(0)
end function


public function FallPlatform_DebugSpawn
	CreateTempObject(TypeName[Fall Platform], 0, object.xpos, object.ypos)
	object[tempObjectPos].value3 = object[tempObjectPos].ypos
end function


event ObjectUpdate
	if object.value0 == 1
		if object.value2 == 0
			object.priority = 1
			object.value2 = 30
		end if
		if object.value4 < 64
			object.value4 += 4
		end if
	else
		if object.value4 > 0
			object.value4 -= 4
		end if
	end if
	temp0 = object.ypos
	temp0 &= -65536
	switch object.state
	case 0
		temp1 = object.value4
		temp1 <<= 1
		Sin(temp2, temp1)
		temp2 <<= 9
		object.ypos = object.value3
		object.ypos += temp2
		if object.propertyValue == 0
			if object.value2 != 0
				object.value2--
				if object.value2 == 0
					object.state++
					object.value2 = 64
				end if
			end if
		end if
		break
	case 1
		object.value2--
		if object.value2 == 0
			object.state++
			foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
				BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -8, 32, 0, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
				if checkResult == true
					object[currentPlayer].ypos += object.yvel
					object[currentPlayer].yvel = object.yvel
					object[currentPlayer].yvel -= 8192
					object[currentPlayer].gravity = 1
				end if
			next
		end if
	case 2
		object.ypos += object.yvel
		object.yvel += 8192
		if object.outOfBounds == 1
			object.ypos = object.value3
			object.yvel = 0
			object.value2 = 0
			object.state++
		end if
		break
	case 3
		if object.outOfBounds == 1
			object.state = 0
			object.priority = 0
		end if
		break
	end switch
	object.value1 = object.ypos
	object.value1 &= -65536
	object.value1 -= temp0
	object.value0 = 0
	if object.state < 2
		temp1 = object.ypos
		object.ypos = temp0
		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -9, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == true
				object.value0 = 1
				object[currentPlayer].ypos += object.value1
			end if
		next
		object.ypos = temp1
	end if
end event


event ObjectDraw
	if object.state < 3
		DrawSprite(0)
	end if
end event

	SetTableValue(TypeName[Fall Platform], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(FallPlatform_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(FallPlatform_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)

end event

		foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
			BoxCollisionTest(C_TOUCH, object.entityPos, -16, 0, 16, 32, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
			if checkResult == 1
				object.value0 = 4
				object.state = 1
				object.drawOrder = 4
				object.frame = 0
			end if
		next
	end if
end event


event ObjectDraw
	switch object.state
	case 1
		if object.value0 > 0
			object.value0--
		else
			object.state = 2
			PlaySfx(SfxName[Water Splash], 0)
		end if
		break
	case 2
		DrawSprite(object.frame)
		object.value0++
		if object.value0 > 4
			object.value0 = 0
			object.frame++
			if object.frame > 4
				object.state = 0
			end if
		end if
		break
	end switch
end event

event ObjectDraw
	if screen.yoffset > 512
		temp0 = screen.yoffset
		temp0 -= 512
		temp0 <<= 15
		FlipSign(temp0)
		tileLayer[1].scrollPos = temp0
	else
		tileLayer[1].scrollPos = 0
	end if
end event

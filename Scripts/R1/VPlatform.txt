private alias 50 : TYPE_VERTPLATFORM

// Function declarations
reserve function VertPlatform_DebugDraw
reserve function VertPlatform_DebugSpawn


public function VertPlatform_DebugDraw
	DrawSprite(0)
end function


public function VertPlatform_DebugSpawn
	CreateTempObject(TypeName[Vert Platform], 0, object.xpos, object.ypos)
	object[tempObjectPos].xpos &= -65536
	object[tempObjectPos].ypos &= -65536
end function


event ObjectUpdate
	object.value3 = oscillation
	object.value3 <<= 1
	if object.propertyValue == 1
		object.value3 += 256
	end if
	Sin(object.value2, object.value3)
	object.value2 *= 6144
	object.value2 += object.ypos
	object.value2 &= -65536
	object.value2 -= object.value1
	temp0 = object.ypos
	object.ypos = object.value1
	foreach (GROUP_PLAYERS, currentPlayer, ACTIVE_ENTITIES)
		BoxCollisionTest(C_PLATFORM, object.entityPos, -32, -9, 32, 12, currentPlayer, C_BOX, C_BOX, C_BOX, C_BOX)
		if checkResult == 1
			object[currentPlayer].ypos += object.value2
		end if
	next
	object.ypos = temp0
end event


event ObjectDraw
	Sin(object.value1, object.value3)
	object.value1 *= 6144
	object.value1 += object.ypos
	object.value1 &= -65536
	DrawSpriteXY(0, object.xpos, object.value1)
end event


event ObjectStartup
	LoadSpriteSheet("R1/Objects.gif")
	SpriteFrame(-32, -16, 64, 32, 101, 109)
	SetTableValue(TypeName[Vert Platform], DebugMode_ObjCount, DebugMode_TypesTable)
	SetTableValue(VertPlatform_DebugDraw, DebugMode_ObjCount, DebugMode_DrawTable)
	SetTableValue(VertPlatform_DebugSpawn, DebugMode_ObjCount, DebugMode_SpawnTable)
	DebugMode_ObjCount++
end event

event RSDKDraw
	DrawSprite(0)
end event

event RSDKLoad
	LoadSpriteSheet("Global/Display.gif")
	SpriteFrame(-16, -16, 32, 32, 1, 143)
end event
